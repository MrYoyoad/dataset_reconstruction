\section{The exact channel: from the quotient to the full release}
\label{sec:exact-channel}

A released LoRA adapter is a deterministic function of the private data, the
public recipe and the random initialisation. Throughout, $k$ is the
dimension of the chart the attacker searches in — the coordinates $w$ of
Definition~\ref{def:setup} — not an intrinsic dimension of the images; every
count below is a count of those coordinates. This section writes that
function out (Theorem~\ref{thm:traj}), shows that the initialisation enters
it only through $rN$ numbers (Corollary~\ref{cor:seed}), and then reads the
same function in two ways: projecting the initialisation away gives Rev~9's
certificate with $r-N$ equations per image (Theorem~\ref{thm:quot}); keeping
it and solving for the $rN$ numbers gives $m-1+r-N$ equations per image
(Theorems~\ref{thm:cap}--\ref{thm:suff}). The difference between the two is
one identifiable object — the coefficient matrix of the released $B_T$ —
and Corollary~\ref{cor:two} states it. Section~\ref{sec:measured} reports
what was measured, on the synthetic test-bed and on real digits, and shows
that the boundary is a statement about the coordinates the attacker searches
in, not about the images.

%----------------------------------------------------------------------------
\subsection{Objects, hypotheses, and the gradient}
\label{sec:objects}

\begin{definition}[the layer, the data, the adapter]\label{def:setup}
Fix integers $m,n,r,N$ with $N\le r\le n$. A frozen weight
$W_0\in\mathbb R^{m\times n}$ receives the representations
$H=[h_1,\dots,h_N]\in\mathbb R^{n\times N}$, $\operatorname{rank}H=N$, of the $N$ private examples;
$E\in\{0,1\}^{m\times N}$ holds their one-hot labels. A LoRA adapter is a
pair $A\in\mathbb R^{r\times n}$, $B\in\mathbb R^{m\times r}$ with a fixed
scale $s>0$; the layer computes logits and loss
\[
  Z(A,B):=(W_0+sBA)H\in\mathbb R^{m\times N},\qquad
  L(A,B):=\sum_{i=1}^N \ell\bigl(z_i,e_i\bigr),
\]
$\ell$ the softmax cross-entropy. Write $D(Z):=\operatorname{softmax}(Z)-E$
(column-wise softmax), $G:=H^{\!\top}H$, and for any $A_0$,
$Q:=(A_0H)^{\!\top}(A_0H)$; both Grams are $N\times N$.
\end{definition}

\begin{lemma}[gradients]\label{lem:grad}
$\nabla_B L=s\,D\,H^{\!\top}A^{\!\top}$ and $\nabla_A L=s\,B^{\!\top}D\,H^{\!\top}$,
with $D=D(Z(A,B))$.
\end{lemma}
\begin{proof}
$dL=\langle D,dZ\rangle=\langle D,\;s\,dB\,AH+s\,B\,dA\,H\rangle
 =\langle s\,D H^{\!\top}A^{\!\top},dB\rangle+\langle s\,B^{\!\top}DH^{\!\top},dA\rangle$.
\end{proof}

\paragraph{Hypotheses.} Each is a formula; each theorem lists the ones it uses.
\begin{itemize}\itemsep2pt
\item[(A1)] \emph{Fixed inputs.} The same $H$ enters the adapter at every
  step: no augmentation, no dropout, no trainable block upstream.
\item[(A2)] \emph{Initialisation.} $B_0=0$; $A_0$ is drawn from a law with a
  density on $\mathbb R^{r\times n}$ (so, with $r\le n$,
  $\operatorname{rank}A_0=r$ a.s.).
\item[(A3)] \emph{Recipe.} Plain gradient descent with step $\eta$ for $T$
  steps,
  \[
    B_{t+1}=B_t-\eta\,\nabla_BL(A_t,B_t),\qquad
    A_{t+1}=A_t-\eta\,\nabla_AL(A_t,B_t),\qquad t=0,\dots,T-1,
  \]
  and the release is $(A_T,B_T)$.\footnote{Everything below survives any
  update that is a scalar linear filter of past gradients and iterates
  (momentum, scalar weight decay — which multiplies $A_0$ by a scalar $c_t$
  in Theorem~\ref{thm:traj}, and nothing else changes) and any matrix map of
  the form $M\mapsto p(MM^{\!\top})M=M\,p(M^{\!\top}M)$ applied to the
  gradient. Adam's entrywise normalisation is not of this form; \S\ref{sec:opens}, item~3.}
\item[(A4)] $\operatorname{rank}P_T=N$ for the matrix $P_T$ of
  Theorem~\ref{thm:traj}. (Needs $m\ge N$ and $N$ linearly independent
  accumulated error trajectories; it fails for duplicated or already-fit
  examples.)
\item[(A5)] $r\ge N+1$.
\item[(A6)] \emph{Canonical basis.} For any full-column-rank $H$, $U(H)$
  denotes the $Q$-factor of the QR decomposition $H=UR_H$ with
  $\operatorname{diag}R_H>0$. This makes $H\mapsto U(H)$ real-analytic; a
  QR without the sign convention is discontinuous at every sign change of a
  diagonal entry.
\end{itemize}

\paragraph{The two coordinates of the seed.} Let $U=U(H)$ and let
$V\in\mathbb R^{n\times(n-N)}$ be an orthonormal basis of $\operatorname{col}(H)^\perp$.
Define
\begin{equation}\label{eq:XY}
\begin{aligned}
  X&:=A_0U\in\mathbb R^{r\times N},\qquad Y:=A_0V\in\mathbb R^{r\times(n-N)},\\
  \text{so that}\qquad A_0&=XU^{\!\top}+YV^{\!\top},\qquad A_0H=XR_H,\qquad
  Q=R_H^{\!\top}X^{\!\top}XR_H .
\end{aligned}
\end{equation}

\paragraph{The private data as a manifold.} Each private image is
$x_i=\psi(w_i)$ for a generator $\psi:\mathbb R^k\to\mathbb R^{d}$, and its
representation is $h_i=\varphi(x_i)$; write $H(w)$, $R_H(w)$, $U_c(w):=U(H(w))$
for a candidate $w=(w_1,\dots,w_N)\in\mathbb R^{Nk}$. Both $\varphi$ and
$\psi$ are real-analytic ($\tanh$ networks in the test-bed).

\subsection{The trajectory, written out}
\label{sec:trajectory}

\begin{theorem}[closure of the factor dynamics]\label{thm:traj}
Assume (A1)--(A3). Then for every $t\le T$
\begin{equation}\label{eq:closure}
  B_t=P_t\,(A_0H)^{\!\top},\qquad A_t=A_0\bigl(I_n+H\,M_t\,H^{\!\top}\bigr),
\end{equation}
where $P_t\in\mathbb R^{m\times N}$, $M_t\in\mathbb R^{N\times N}$ are given
by $P_0=0$, $M_0=0$ and
\begin{equation}\label{eq:rec}
  D_t=D\bigl(W_0H+s\,P_tQ(I_N+M_tG)\bigr),\qquad
  P_{t+1}=P_t-\eta s\,D_t\,(I_N+GM_t^{\!\top}),\qquad
  M_{t+1}=M_t-\eta s\,P_t^{\!\top}D_t .
\end{equation}
\end{theorem}
\begin{proof}
Induction on $t$; $t=0$ is (A2). Assume \eqref{eq:closure} at $t$. Then
$B_tA_tH=P_t(A_0H)^{\!\top}A_0(H+HM_tG)=P_t\,Q\,(I+M_tG)$, which gives $D_t$.
By Lemma~\ref{lem:grad},
\begin{align*}
  \nabla_BL&=sD_tH^{\!\top}A_t^{\!\top}=sD_t\bigl(H^{\!\top}+G^{\!\top}M_t^{\!\top}H^{\!\top}\bigr)A_0^{\!\top}
  =sD_t(I+GM_t^{\!\top})(A_0H)^{\!\top}\qquad(G^{\!\top}=G),\\
  \nabla_AL&=sB_t^{\!\top}D_tH^{\!\top}=A_0H\,(sP_t^{\!\top}D_t)\,H^{\!\top},
\end{align*}
so the update of (A3) keeps both shapes and produces \eqref{eq:rec}.
\end{proof}

\begin{corollary}[where the seed enters]\label{cor:seed}
Assume (A1)--(A3). The coefficient trajectory $(P_t,M_t,D_t)_{t\le T}$ is a
function of $W_0H$, $G$, $Q$, $E$ and $(\eta,s,T)$ only; the initialisation
$A_0$ enters it through $Q=R_H^{\!\top}X^{\!\top}XR_H$ and nowhere else. The release is
\begin{equation}\label{eq:release}
  B_T=P_T\,R_H^{\!\top}X^{\!\top},\qquad
  A_T=X\,\Omega\,U^{\!\top}+Y\,V^{\!\top},\qquad
  \Omega:=I_N+R_HM_TR_H^{\!\top},
\end{equation}
so $A_TU=X\Omega$ and $A_TV=Y$: the block $Y$ of the seed is released
unchanged, and $B_T$, $A_TU$ depend on the seed only through $X$. In
particular, for every orthogonal $O\in\mathrm O(r)$ the seed $OA_0$ produces
the same $(P_t,M_t,D_t)_{t\le T}$ and $\Omega$, and the release
$(OA_T,\;B_TO^{\!\top})$: the coefficient trajectory is invariant under the
LoRA gauge $BA=(BO^{\!\top})(OA)$, and the release transforms covariantly.
\end{corollary}
\begin{proof}
Substitute \eqref{eq:XY} into \eqref{eq:closure}--\eqref{eq:rec}:
$A_T=A_0+A_0HM_TH^{\!\top}=XU^{\!\top}+YV^{\!\top}+XR_HM_TR_H^{\!\top}U^{\!\top}$.
For the gauge, $Q$ is unchanged by $X\mapsto OX$, so \eqref{eq:rec} is
unchanged, and \eqref{eq:release} transforms as stated.
\end{proof}
The gauge statement was tested rather than assumed (job 487882 — a log,
not a jsonl row; the third shape ran at $\eta=0.03$):
at three shapes — $(n,k,N,T)$ equal to $(96,12,8,400)$, to $(32,6,4,200)$
and to $(96,10,12,150)$ — the relative discrepancies
\[
  \frac{\|A_T'-OA_T\|}{\|A_T\|},\qquad \frac{\|B_T'-B_TO^{\!\top}\|}{\|B_T\|},\qquad
  \frac{\|\Omega'-\Omega\|}{\|\Omega\|}
\]
are all between $5.8\times10^{-16}$ and $2.1\times10^{-15}$. The coefficients see only $Q$; the release sees $X$; the
gauge moves both, so nothing is lost by the attacker and the invariance is a
statement about the trajectory, not about identifiability — the release
still determines $X$, and $\rho$ is not degenerate.

\subsection{Two readings of the same release}
\label{sec:read}

\subsubsection*{First reading: project the seed away}

\begin{theorem}[the quotient certificate]\label{thm:quot}
Assume (A1)--(A5). Then $\operatorname{row}(B_T)=\operatorname{col}(A_0H)$
as subspaces of $\mathbb R^r$, and with $\Pi_S$ the orthogonal projector of
$\mathbb R^r$ onto $S^\perp$,
\begin{equation}\label{eq:C}
\begin{aligned}
  C&:=\Pi_{\operatorname{row}(B_T)}A_T=\Pi_{\operatorname{col}(A_0H)}A_0,\\
  CH&=0,\qquad \operatorname{rank}C=r-N,\qquad
  \ker C=\operatorname{col}(H)+\ker A_0 .
\end{aligned}
\end{equation}
\end{theorem}
\begin{proof}
By \eqref{eq:closure} the rows of $B_T$ are the $m$ combinations of the $N$
columns of $A_0H$ with coefficient rows $P_T$; under (A4) they span
$\operatorname{col}(A_0H)$. Projecting \eqref{eq:closure} for $A_T$:
$\Pi_{\operatorname{col}(A_0H)}A_T=\Pi_{\operatorname{col}(A_0H)}A_0+\Pi_{\operatorname{col}(A_0H)}(A_0H)\,M_TH^{\!\top}$,
and the second term is zero. Hence $CH=\Pi_{\operatorname{col}(A_0H)}(A_0H)=0$;
the image of $C$ is $\Pi_{\operatorname{col}(A_0H)}(\operatorname{col}A_0)=\operatorname{col}(A_0H)^\perp$
of dimension $r-N$ by (A2), (A5); and $Cx=0$ iff $A_0x\in\operatorname{col}(A_0H)$
iff $x\in\operatorname{col}(H)+\ker A_0$.
\end{proof}

What the first reading uses is visible in \eqref{eq:C}: of $B_T$ only the
subspace $\operatorname{row}(B_T)$ — a point of the Grassmannian
$\operatorname{Gr}(N,r)$, $N(r-N)$ numbers — and of $A_T$ only its projection
off that subspace. The coefficient matrix $P_T$ is removed by the projection.
The equations it imposes on candidate data are $CH(w)=0$: $C$ has rank $r-N$,
so these are $(r-N)N$ scalar equations, $r-N$ per image; Rev~9 Thm~3 and
Prop~4 show they pin $w$ locally iff $k<r-N$. No recipe and no solver are
used.

\subsubsection*{Second reading: keep the seed and solve for it}

\begin{definition}[the attacker's map]\label{def:rho}
The attacker knows $W_0$, $\varphi$, $\psi$, $E$, the recipe $(\eta,s,T)$
and the release $(A_T,B_T)$; the unknowns are $w\in\mathbb R^{Nk}$ and, by
Corollary~\ref{cor:seed}, $X\in\mathbb R^{r\times N}$.\footnote{$Y$ is not an
unknown: it enters neither $B_T$ nor $A_TU$, and \eqref{eq:release} shows it
passes to the release unchanged — in the true basis $V$, which the attacker
does not know and does not need.}
For a candidate $(w,X)$ let $P_T(w,X)$, $M_T(w,X)$ be the output of
\eqref{eq:rec} run with $H=H(w)$, $G=G(w)$, $Q=R_H(w)^{\!\top}X^{\!\top}XR_H(w)$,
and $\Omega(w,X)=I+R_H(w)M_TR_H(w)^{\!\top}$. The residual map is
\begin{equation}\label{eq:rho}
  \rho(w,X):=\Bigl(\;\frac{P_T(w,X)\,R_H(w)^{\!\top}X^{\!\top}-B_T}{\|B_T\|}\;,\;\;
  \frac{X\,\Omega(w,X)-A_T\,U_c(w)}{\|A_T\|}\;\Bigr)
  \in\mathbb R^{m\times r}\times\mathbb R^{r\times N}.
\end{equation}
The attack is $\min_{w,X}\|\rho(w,X)\|^2$. The true $(w^\star,X^\star)$ is a
zero of $\rho$; the second reading asks whether it is an isolated zero.
\end{definition}

The target $A_TU_c(w)$ in the second block moves with the candidate, because
the attacker does not know $U$ and must form $U_c(w)$; its $w$-derivative
$A_T\,\partial U_c/\partial w$ is part of $D\rho$.\footnote{Writing the target
as the fixed $A_TU$ instead drops this term and undercounts the
$w$-dependence; the budget one then obtains is refuted by the measured full
column rank at $k=24>m$ (Example~\ref{ex:T1}). The normalisers $\|B_T\|$,
$\|A_T\|$ are fixed numbers — deliberately not $\|A_TU_c(w)\|$, which the
optimiser could shrink — and scale the spectrum of $D\rho$, not its rank.}

\begin{proposition}[the softmax head constrains $B_T$]\label{prop:simplex}
Assume (A2), (A3). Then $\mathbf 1^{\!\top}P_t=0$ and hence
$\mathbf 1^{\!\top}B_t=0$ for every $t$.
\end{proposition}
\begin{proof}
Every column of $\operatorname{softmax}(Z)$ and of $E$ sums to one, so
$\mathbf 1^{\!\top}D(Z)=0$. In \eqref{eq:rec},
$\mathbf 1^{\!\top}P_{t+1}=\mathbf 1^{\!\top}P_t-\eta s\,(\mathbf 1^{\!\top}D_t)(I+GM_t^{\!\top})=\mathbf 1^{\!\top}P_t$,
and $P_0=0$. Then $\mathbf 1^{\!\top}B_t=\mathbf 1^{\!\top}P_t(A_0H)^{\!\top}=0$.
\end{proof}

\begin{theorem}[capacity: the count is necessary]\label{thm:cap}
Assume (A1)--(A3), (A6) and $N\le m-1$. Let
\[
  \mathcal B:=\{B\in\mathbb R^{m\times r}:\ \operatorname{rank}B=N,\ \mathbf 1^{\!\top}B=0\},
  \qquad \dim\mathcal B=N(r-N)+N(m-1)=N\bigl((m-1)+r-N\bigr).
\]
Then, on each connected component of $\{(w,X):\operatorname{rank}H(w)=N\}$
that contains a point with $\operatorname{rank}P_T(w,X)=N$, the maximal rank
of $D\rho$ is at most $N\bigl((m-1)+r-N\bigr)+rN$. Consequently, if
\begin{equation}\label{eq:cap}
  k\;\ge\;m+r-N\qquad\bigl(\text{equivalently } Nk+rN>\dim\mathcal B+rN\bigr),
\end{equation}
then for almost every true $(w^\star,X^\star)$ the set of candidates that
reproduce the release exactly, $\rho^{-1}(0)$, contains a submanifold through
$(w^\star,X^\star)$ of dimension at least $N\bigl(k-(m-1+r-N)\bigr)\ge N$.
\end{theorem}
\begin{proof}
\emph{Dimension of $\mathcal B$.} Let $\Theta\in\mathbb R^{m\times(m-1)}$ be
an orthonormal basis of $\mathbf 1^\perp$; $B\mapsto\Theta^{\!\top}B$ is a
rank-preserving isometry of $\{\mathbf 1^{\!\top}B=0\}$ onto
$\mathbb R^{(m-1)\times r}$. A rank-$N$ matrix there is determined by its row
space $S\in\operatorname{Gr}(N,r)$ and its coefficient matrix in a basis of
$S$, an element of $\mathbb R^{(m-1)\times N}$ of full rank; the dimensions
are $N(r-N)$ and $N(m-1)$.

\emph{Rank bound.} On the open set $\mathcal O$ where
$\operatorname{rank}P_T(w,X)=N$, Theorem~\ref{thm:traj} and
Proposition~\ref{prop:simplex} put the first block of $\rho$, plus the
constant $B_T/\|B_T\|$, in $\mathcal B/\|B_T\|$, a manifold of dimension
$\dim\mathcal B$, so the derivative of that block has rank at most
$\dim\mathcal B$ there; the second block has $rN$ rows; so
$\operatorname{rank}D\rho\le\dim\mathcal B+rN$ on $\mathcal O$. On a
connected component the maximal rank of $D\rho$ is attained on an open dense
set (the $\rho$ of \S\ref{sec:objects} is real-analytic there), which meets
the non-empty open set $\mathcal O$; hence the maximal rank on that component
is at most $\dim\mathcal B+rN$. (Under (A4) the truth lies in $\mathcal O$.)

\emph{Fibre.} On the component, the set where $\operatorname{rank}D\rho$ is
below its maximum is a proper analytic subvariety — closed, with empty
interior and Lebesgue measure zero. Off it the rank is locally constant and
the constant-rank theorem makes $\rho^{-1}(\rho(p))$ a submanifold through
$p$ of dimension $(Nk+rN)-\operatorname{rank}D\rho(p)\ge Nk+rN-\dim\mathcal
B-rN$. This holds at Lebesgue-almost every $(w^\star,X^\star)$; for the seed
that is a.e.\ under (A2), and for the data it is a.e.\ in the chart
coordinates, and $\rho(w^\star,X^\star)=0$ at the truth.
\end{proof}

\begin{remark}\label{rem:distance}
Theorem~\ref{thm:cap} says that past the line the truth is not an isolated
zero of $\rho$: a continuum of release-consistent candidates passes through
it. It says nothing about how far that continuum extends in image space:
non-isolation is not distance. And below the line, identifiability is not
fidelity: the coordinates $w$ are pinned, and the image returned, $\psi(\hat
w)$, is only as good as the chart $\psi$. Section~\ref{sec:measured} measures
both.
\footnote{Rank deficiency of $D\rho$ \emph{at} the truth alone would not give
non-isolation ($f(x)=x^2$ has $f'(0)=0$ and an isolated zero); the proof uses
that the rank bound holds at every candidate, and concludes only almost
everywhere.}
\end{remark}

\begin{theorem}[generic local identifiability below the line]\label{thm:suff}
Assume (A1)--(A3), (A6) and $k\le(m-1)+r-N$. Let
$\mathcal D:=\{(w,X):\operatorname{rank}H(w)=N\}$, an open set, and let
$\mathcal D_0$ be any of its connected components. Exactly one of the
following holds: $D\rho$ has full column rank $Nk+rN$ at almost every point
of $\mathcal D_0$, or at no point of $\mathcal D_0$. In the first case every
true $(w^\star,X^\star)\in\mathcal D_0$ outside a null set is an isolated
zero of $\rho$; so one point of full column rank in $\mathcal D_0$ gives
isolation for almost every seed (under (A2)) and Lebesgue-almost every chart
coordinate in $\mathcal D_0$.
\end{theorem}
\begin{proof}
The $(Nk+rN)$-minors of $D\rho$ are real-analytic on the connected
real-analytic manifold $\mathcal D_0$; their common zero set is an analytic
subvariety of $\mathcal D_0$, hence either all of $\mathcal D_0$ or of
measure zero (identity theorem). At a point of full column rank the
inverse-function theorem makes the zero isolated (Rev~9 Prop~8).
\end{proof}

The statement is per component because a real-analytic subvariety of
codimension one can disconnect: $\mathcal D$ is connected — and one witness
then serves globally — exactly when the rank-deficiency locus
$\{w:\operatorname{rank}H(w)<N\}$ has real codimension at least two, which is
the generic situation for $n\ge N+1$ but is not verified here. The theorem
reduces genericity to one witness of full column rank in the truth's
component, and at $T=1$ the map is explicit enough to exhibit one.

\begin{example}[$T=1$, closed form]\label{ex:T1}
From \eqref{eq:rec}, $P_1=-\eta s\,D_0$ and $M_1=0$, with
$D_0=D(W_0H(w))$ depending on $w$ only through the base model's logits.
Hence $A_1=A_0$, $\Omega=I$, and \eqref{eq:rho} reads (unnormalised)
\[
  \rho_{T=1}(w,X)=\bigl(-\eta s\,D_0(w)\,R_H(w)^{\!\top}X^{\!\top}-B_T,\;\;X-A_TU_c(w)\bigr).
\]
The second block gives $X=A_TU_c(w)$ exactly, and substituting,
$U_c(w)R_H(w)=H(w)$,
\begin{equation}\label{eq:T1}
  B_T\;=\;-\eta s\;D_0(w)\,H(w)^{\!\top}A_T^{\!\top}:
\end{equation}
an equation in $w$ alone whose right side ranges over $\mathcal B$ — the
$N((m-1)+r-N)$ constraints of Theorem~\ref{thm:cap} on the $Nk$ unknowns,
with no unrolling and the seed eliminated in closed form; the budget obtained
by eliminating $X$ here equals the one obtained by counting $X$ as an unknown
in Theorem~\ref{thm:cap}, as it must.\footnote{Eliminating
$X$ does not lower the count: the count is the dimension of the manifold that
$B_T$ lives on, not a tally of which block ``pays for'' which unknown. That is
why the intuition ``at $T=1$ the $A$-block hands $X$ over for free, so
$k\le m$'' is wrong; it is refuted by measurement at $k=24>m=20$.}
Measured$^\dagger$ (job 479587, $N=8$): $D\rho_{T=1}$ has full column rank
at $k=16,20,24$ with $\sigma_{\min}=7.26\times10^{-4},\,1.21\times10^{-4},\,3.15\times10^{-5}$,
identical to three figures with the $T=2$ values, and collapses at $k=28=m+r-N$
($3.3\times10^{-19}$). These are witnesses of Theorem~\ref{thm:suff} at
$T=1$ for the charts $\psi_k$ with $k=16,20,24$ — each $k$ is its own chart
and its own dichotomy. For the working point $T=400$ the witnesses are
direct: $D\rho$ at the truth has full column rank at every $k\le27$ (jobs
467914, 479587); the $T=1$/$T=2$ agreement to three figures is a consistency
check on the closed form, not the witness for $T>1$.
\end{example}

\subsubsection*{The two readings side by side}

\begin{corollary}[the two channels]\label{cor:two}
Under (A1)--(A3) the release $(A_T,B_T)$ has the form \eqref{eq:release}.
Reading it by Theorem~\ref{thm:quot} uses $\operatorname{row}(B_T)\in\operatorname{Gr}(N,r)$
and $\Pi_{\operatorname{row}(B_T)}A_T$, imposes $N(r-N)$ equations on $Nk$
unknowns, and needs (A4), (A5) but no recipe. Reading it by
Definition~\ref{def:rho} uses all of $B_T\in\mathcal B$ and $A_TU_c$,
imposes $N(r-N)+N(m-1)+rN$ equations on $Nk+rN$ unknowns, and needs (A6),
a solver, and the recipe — which \S\ref{sec:measured} shows can be checked
against the floor, fitted jointly, or probed. The $rN$ equations of the second block pay exactly
for the $rN$ unknowns of $X$; the difference between the channels is the
$N(m-1)$ coefficients of $P_T$ in a basis of $\operatorname{row}(B_T)$, which
the projection in \eqref{eq:C} removes and which \eqref{eq:rho} keeps:
\[
  \text{certificate: } k<r-N\qquad\qquad \text{exact: } k<m+r-N .
\]
\end{corollary}
\begin{proof}
Assemble Theorem~\ref{thm:quot}, the proof of Theorem~\ref{thm:cap}, and
Theorem~\ref{thm:suff}.
\end{proof}

\begin{center}\small
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{0.2\linewidth}>{\raggedright\arraybackslash}p{0.27\linewidth}>{\raggedright\arraybackslash}p{0.13\linewidth}>{\raggedright\arraybackslash}p{0.32\linewidth}@{}}\toprule
 & reads & equations per image & needs\\ \midrule
certificate (Thm~\ref{thm:quot}) & $\operatorname{row}(B_T)$; $\Pi_{\operatorname{row}(B_T)}A_T$ & $r-N$ & (A1)--(A5)\\
exact (Thms~\ref{thm:cap}, \ref{thm:suff}) & all of $B_T$; $A_TU_c(w)$ & $(m-1)+r-N$ & (A1)--(A3), (A6); the recipe, checked or fitted (\S\ref{sec:measured}); a solver and a start\\ \bottomrule
\end{tabular}

\begin{proposition}[chaining the two channels does not move the line]
\label{prop:chain}
Let $\mathcal S_\rho:=\{(w,X):\rho(w,X)=0\}$ be the release-consistent set of
Definition~\ref{def:rho}, let $Z_C:=\{w:C\varphi(\psi(w))=0\}$ be the
certificate manifold, and let
$\mathcal S_{\mathrm{chain}}:=\mathcal S_\rho\cap(Z_C^{\times N}\times\mathbb R^{rN})$
be the solution set of the chained problem --- replay restricted to candidates
whose images satisfy the certificate. Assume (A1)--(A6) hold for the
attacker's own simulated trajectory. Then
\[
  \mathcal S_\rho\subseteq Z_C^{\times N}\times\mathbb R^{rN},
  \qquad\text{hence}\qquad
  \mathcal S_{\mathrm{chain}}=\mathcal S_\rho ,
\]
and the boundary of exact identifiability is therefore unchanged at
$k<(m-1)+r-N'$.
\end{proposition}

\begin{proof}
Let $\rho(w,X)=0$ and write $\hat H$, $\hat A_T$, $\hat B_T$ for the
candidate's own features and released factors. Vanishing of the first block
gives $\hat B_T=B_T$; vanishing of the second gives
$\hat A_TU_c=A_TU_c$, and since each $\hat h_i$ lies in
$\operatorname{col}(U_c)$ by construction, $\hat A_T\hat h_i=A_T\hat h_i$.
The candidate's trajectory satisfies the hypotheses, so
Theorem~\ref{thm:quot} applies to it and gives
$\Pi_{\operatorname{row}(\hat B_T)^\perp}\hat A_T\hat h_i=0$. Substituting the
two equalities, $\Pi_{\operatorname{row}(B_T)^\perp}A_T\hat h_i=C\hat h_i=0$,
so $w\in Z_C^{\times N}$. The intersection defining
$\mathcal S_{\mathrm{chain}}$ is therefore all of $\mathcal S_\rho$, and
identifiability --- whether the truth is isolated in the solution set --- is a
property of that set alone, so Theorems~\ref{thm:cap} and \ref{thm:suff}
apply verbatim.
\end{proof}

A dimension count reaches the same place, which is worth recording because it
is the form in which the question is usually asked. On $Z_C$ each recorded
image retains $k-(r-N')$ free coordinates, and of the $B$-block's
$N'\bigl((m-1)+r-N'\bigr)$ conditions the certificate has already consumed
$N'(r-N')$, leaving $N'(m-1)$. Closure requires
$N'\bigl(k-(r-N')\bigr)\le N'(m-1)$, i.e. $k\le(m-1)+r-N'$ --- replay's own
line. Two independent routes to one boundary.

\begin{corollary}[the chain is a basin claim]\label{cor:chainbasin}
Chaining reduces the dimension of the search from $Nk$ to $N\bigl(k-(r-N')\bigr)$
coordinates, plus the seed block, while leaving the solution set unchanged.
Any advantage it confers is therefore a property of the optimisation and not
of identifiability. In particular a chained attack cannot reach charts richer
than $k=(m-1)+r-N'-1$, and it is only defined in the band
$r-N'\le k<(m-1)+r-N'$: below $r-N'$ the certificate already isolates the
recorded images and there is nothing to chain.
\end{corollary}

\noindent A corollary of the proposition worth stating separately: because
$C$ is a function of the release, no construction built from $C$ can add
information to $\rho$. Any derivation in which a chain \emph{widens} the
admissible $k$ has counted the release twice.
\end{center}
Both capacities are statements about the chart coordinates $w\in\mathbb R^{Nk}$
of Definition~\ref{def:setup}: the unknown is $w$, the theorems pin $w$, and
the attack returns $\psi(\hat w)$. Nothing in them constrains the chart
$\psi$, which is the attacker's to choose.

\begin{corollary}[the count for a general adapted layer]\label{cor:general}
The proof of Theorem~\ref{thm:cap} uses only that the simulated $B_T$ has
rank at most $\min(N,r)$ and, under a softmax head, zero column sums; it
does not use what the layer computes downstream. Hence, with $m$ the output
width of the adapted layer, the necessary condition on the chart dimension
reads
\begin{center}\small
\begin{tabular}{@{}lll@{}}\toprule
channel & $N\le r$ & $N\ge r$\\ \midrule
certificate (Theorem~\ref{thm:quot}) & $k<r-N$ & — (no certificate)\\
exact, softmax head & $k<m+r-N$ & $k\le (m-1)r/N$\\
exact, any other head & $k<m+r-N+1$ & $k\le mr/N$\\ \bottomrule
\end{tabular}
\end{center}
\end{corollary}
\begin{proof}
For $N\le r$ replace $\mathcal B$ by the rank-$N$ matrices in
$\mathbb R^{m\times r}$ (dimension $N(m+r-N)$) when there is no column-sum
constraint. For $N\ge r$ and $m\ge r$ (so that $\operatorname{rank}B_T=\min(\operatorname{rank}P_T,r)$
can reach $r$), $B_T=P_T(A_0H)^{\!\top}$ lies in $\mathbb R^{m\times r}$
(or in its $(m-1)r$-dimensional column-sum-zero subspace), while Corollary~\ref{cor:seed} still reduces the
seed to $X\in\mathbb R^{r\times N}$; the count is $mr+rN\ge Nk+rN$
(resp.\ $(m-1)r+rN\ge Nk+rN$). The two regimes agree at $N=r$.
\end{proof}
This is the realistic picture for a transformer: the first adapted
projection has $m$ equal to the layer's output width, hundreds to thousands
of coordinates per representation, against $r-N$ for the certificate; and
once $N$ exceeds $r$ — tokens, not images — the budget per representation is
$mr/N$. The $N\ge r$ regime is derived here and untested; every measurement
in \S\ref{sec:measured} has $N<r$.

\paragraph{Eliminating the seed in closed form.} By Corollary~\ref{cor:seed}
the coefficients depend on $X$ only through the Gram $S:=X^{\!\top}X\in
\mathrm{Sym}^+_N$, so $\Omega=\Omega(w,S)$, and the second block of
\eqref{eq:rho} solves for $X$ outright: $X=A_TU_c(w)\,\Omega(w,S)^{-1}$
wherever $\Omega$ is invertible. Substituting into the first block and
adding the consistency condition $X^{\!\top}X=S$ leaves unknowns
$(w,S)\in\mathbb R^{Nk}\times\mathrm{Sym}^+_N$ ($S$ is a Gram matrix, so it
lies in the positive-semidefinite cone, whose boundary
$\operatorname{rank}X<N$ is where the parametrisation degenerates) — $Nk+N(N+1)/2$ numbers, $36$
in place of $rN=128$ at $N=8$, $r=16$ — against $\dim\mathcal B+N(N+1)/2$
equations: the same capacity, a search space smaller by $rN-N(N+1)/2$. Since
\S\ref{sec:opens} finds the initialiser, not the information, to be what
stops the attack, this is the first thing to implement for efficiency. Two caveats: the
count transfers but the reduced problem's local identifiability near
$\partial\mathrm{Sym}^+_N$ is untested, and a matched cell measured no basin
widening \textnormal{[VERIFIED†]} (job 605718: $(w,X)$ recovers $0$ of $3$ at
a $1.5\times$ perturbation against $1$ of $3$ for the $(w,S)$ form, both $0$
of $3$ beyond), so it is a smaller search space, not a bigger basin.

\subsection{What was measured}
\label{sec:measured}

Test-bed: FP64, $m=20$, $r=16$, $n=96$, $\tanh$ generator $\psi$ to a $64$-d
image, $\tanh$ encoder $\varphi$, softmax head, plain SGD; the attack is
Levenberg--Marquardt on \eqref{eq:rho} with an autograd Jacobian. No
component is pretrained: $W_0$, the encoder $\varphi$ and the generator
$\psi$ are fixed random matrices (seeded), and the only training anywhere is
the adapter's $T$ steps on the $N$ private examples. The same holds for the
MNIST cells below, whose encoder is a fixed random network; the private
digits are ``unseen'' in the trivial sense that nothing was trained on
anything before the adapter. The trained-model paragraph below is the one cell on a learned encoder
and a learned head. Every run is
classified by two numbers, the final residual $\|\rho\|^2$ and the worst
image error: \emph{recovered} — residual at the $10^{-30}$ floor and error at
machine precision; \emph{alias} — residual at the floor at a point other than
the truth, with $\sigma_{\min}(D\rho)$ collapsed; \emph{search failure} —
residual above the floor. Two scopings hold for every capacity, alias and
recipe cell below: each is one seed, and each is solved from a start at the
truth plus $10\%$ latent noise, which an attacker cannot build (the starts an
attacker can build are the last item of \S\ref{sec:opens}). The residual and
$\sigma_{\min}$ are attacker-computable; the image-error column is not.

\paragraph{Where the law stops applying: a second channel-closing
mechanism.} Identifiability does fail below the line on a strong encoder,
and it is the imprint law of the next paragraph acting rather than a gap in
the theorems: $\operatorname{rank}B_T$ counts the examples recorded above the
floor, and when that count falls below $N$, hypothesis (A4) fails, $D\rho$
is rank-deficient, and the channel closes \emph{below its own capacity line}
— for reasons of the model rather than the count. On the $97.9\%$ encoder
this happens in nine of the ten batches measured, with
$\operatorname{rank}B_T$ between $1$ and $6$.

What decides which examples survive is measured, and it is the margin
\emph{ordering}. Across the nine rank-deficient batches$^\dagger$ (job
631392), the examples whose imprint lies below the floor are \emph{exactly}
the highest-margin ones, in all nine, with drop counts of $2,2,2,4,5,5,7,7$
and $7$ — so the match is not carried by the cheap two-of-eight cases. Under
random assignment the joint probability of that is $4\times10^{-13}$, and it
reproduces in a second measurement of the same batches (job 650891, nine of
nine).

There is no absolute cut. Below-floor margins run $30$ to $95$ while
above-floor margins reach $74$: an example at margin $44$ drops in one batch
while one at $74$ survives in another. Recording is therefore decided by an
example's \emph{rank within its batch}, not by any threshold on its own
confidence — which is both more surprising than a threshold and more
consequential, since it means a defender cannot secure an example by making
the model confident about it in isolation. Neither its label nor any
coupling between examples enters the ordering, and same-label batches settle
that directly: eight images sharing one label give full rank $8$ when the
model is wrong about all of them (the new-class cells) and rank $1$ when it
is wrong about one and confident about seven. Identical label multiset,
opposite outcomes, distinguished only by the margins.

One caution before that is read as a defence. The floor is relative to each
batch's largest imprint, so a badly fit example added to a batch demotes the
confident ones out of the \emph{count} — but the count is not the same as
recoverability, which turns on the absolute imprint against numerical
precision. And the absolute imprints point the other way: high-margin
examples sit at $10^{-41}$--$10^{-38}$ in an all-confident batch and at
$10^{-26}$--$10^{-25}$ in batches containing a badly fit one, fifteen orders
higher. Those are different image sets, so the comparison is uncontrolled and
proves nothing; it is exactly the measurement named as missing in
\S\ref{sec:opens}, and until it is made, the rank should be read as a count
of what dominates the release and not as a certificate about the rest.

That the identity is a measurement and not a definition matters here, and it
holds: the rank of $B_T$ read from its own singular spectrum, the rank of the
stacked imprints, and the count above the threshold are three separate
computations, and they agree in all forty batches. In the rank-deficient
cells the deficiency is visible in the release's own spectrum
($\sigma_N/\sigma_1$ of $1.5\times10^{-19}$ and $1.0\times10^{-16}$), so an
attacker sees it without the decomposition.

Two earlier readings of this paragraph were wrong and are worth recording,
because both were natural. Encoder quality alone was blamed, by quoting a
single strong rung of $2.0\times10^{-15}$ that was the geometric mean of two
populations seven orders apart — a value no measurement lies within three
orders of. Then the label multiset was blamed, on the observation that the
distinct-label draw keeps rank $8$ on-chart where the repeated-label draw
falls to $6$. That comparison is confounded: the two draws are not the same
images relabelled but different images, and the repeated one happens to
contain examples of margin $46$ and $37$ against a maximum of $23$ in the
other. On raw digits both draws give rank $6$. Chart projection is the one
factor that genuinely moves the count, and it moves it \emph{upward} for the
attacker: projecting onto a $16$-dimensional chart degrades the image, the
model becomes less certain, and more is recorded.

\paragraph{What the release records, per example.} The gradient in $B$ is
$\nabla_BL=s\,D\,(AH)^{\!\top}=s\sum_i D[:,i]\,(Ah_i)^{\!\top}$, a sum of
rank-one pieces, one per private example. Under (A2) and (A3) the release is
therefore
\begin{equation}\label{eq:imprint}
  B_T=\sum_{i=1}^N C_i,\qquad
  C_i=-\eta s\sum_{t<T} D_t[:,i]\,(A_th_i)^{\!\top},\qquad
  \|C_i\|\;\le\;\eta s\max_t\|A_th_i\|\;\sum_{t<T}\bigl\|D_t[:,i]\bigr\|,
\end{equation}
The split is additive with one term per example and no cross-index term —
exactly what the retracted reading of $P_T$ was not. Within the bound, the
error factor $\sum_t\|D_t[:,i]\|$ involves example $i$ alone, while the
scale factor $\max_t\|A_th_i\|$ contains the shared adapter and so depends
on the whole batch through the trajectory. That is not a caveat but the
content: the measurement below finds that common factor near-constant across
examples and batches, and its constancy is what earns the per-example
reading. Since $B_0=0$ starts the trajectory at $W_0$, an example the base
model already classifies with margin $M$ enters at the scale of its softmax
error there, $\mathrm e^{-M}$.

Two things are measured, and the second is the substantive one. First, in
all forty batches $\operatorname{rank}B_T$ equals the number of examples
whose imprint clears the floor$^\dagger$ (job 631392, gated at
$\|\sum_iC_i-B_T\|/\|B_T\|<4\times10^{-12}$): the rank of the release is a
count of the examples it records. Second, and this is the non-obvious part,
the \emph{ratio} $\|C_i\|/\sum_t\|D_t[:,i]\|$ is tight — median
$3.0\times10^{-3}$ overall, and within a single batch it varies by a median
of $0.4$ and at worst $0.95$ of a decade. Since $C_i$ is a sum over $T=400$
steps of terms each carrying $D_t[:,i]$, the per-step contributions could
have cancelled to any degree; that they do not, uniformly across encoders
and batches, is a dynamical fact and not an algebraic one.

The rank correlation between $\|C_i\|$ and $\sum_t\|D_t[:,i]\|$ is then
$971$ of $1120$ concordant pairs (Kendall $\tau\approx0.73$), but it should
be read per encoder rather than pooled,
because one population contributes nothing:

The random encoder is at chance for a reason that is itself quantitative
rather than an excuse: its eight examples differ in accumulated error by
only $4$ to $21\%$, which is smaller than the ratio's own within-batch
scatter of a third to a whole decade. So the ordering is resolvable exactly
when the spread in error exceeds the scatter in the ratio — a criterion that
says in advance where the law will and will not order examples, and which
the three trained encoders satisfy comfortably.\footnote{Because $C_i$
contains $D_t[:,i]$ as a factor, a log-log plot of $\|C_i\|$ against
$\sum_t\|D_t[:,i]\|$ is close to plotting a quantity against its own bound,
and a slope-one line is largely forced by \eqref{eq:imprint}. The evidence
is therefore the tightness of the ratio and the rank identity, not the
correlation; Figure~\ref{fig:imprint} is an illustration of the former, not

Two consequences follow, and both are basis-free. \emph{A private set the
model already fits is not recorded at all}$^\dagger$ (job 627166): for eight
digits of margin $56$--$95$ under the strong model, every imprint is below
$1.5\times10^{-24}$ in absolute terms — the release is numerically empty,
and a rank read off it at a relative threshold is meaningless. The same eight
digits through the random, weak and mid encoders give eight imprints of order
one. \emph{Where the model is wrong, the release is sharply selective}: in a
batch containing one misclassified digit (margin $-3.9$) the relative
imprints run $1$, $7.6\times10^{-2}$, $8.2\times10^{-7}$,
$4.8\times10^{-8}$ for margins $-3.9$, $4.1$, $15.6$, $17.7$, and the four
examples with margins above $39$ sit between $3\times10^{-17}$ and
$10^{-30}$. What leaks is what the model had to learn.

For the attacker, the capacity $m+r-N$ is an upper bound that a confident
model does not spend, and $\operatorname{rank}B_T$ read off the release says
how many examples were recorded at all. For the defender, the per-example
accumulated residual during their own fine-tuning — a quantity already
computed in the course of training — is the meter. Batch composition cannot
enter the \emph{form} of the bound in \eqref{eq:imprint}, and it enters its
\emph{value} only weakly: the accumulated error is trajectory-dependent and
the trajectory is shared, but holding the images fixed and permuting the
label multiset moves the accumulated error by $0.1$--$4\%$ and the imprint
by about $5\%$. That is composition-independence tested against label
permutation only; the case that would test it properly — the same example in
a batch of confident examples, then beside a badly fit one — is not in the
design (\S\ref{sec:opens}).\footnote{An earlier version of this section attributed per-example
content to the columns of $P_T$ and reported a coupling between batch-mates.
Column $i$ of $P_T=-\eta s(\sum_tD_t)R_H^{\!\top}$ mixes every example
$j\ge i$ in batch order through the triangular factor of $H=U_cR_H$: it is
basis- and order-dependent, and permuting the batch changes it. A gated
trace of the training loop refuted the coupling from the other side (job
628731: the margins it was supposed to move do not move — a margin-$71$
digit ends at $71.1$). Equation~\eqref{eq:imprint} is the basis-free
replacement, and every rank statement above was unaffected by the
retraction.}

\paragraph{A new category is the opposite extreme, and one shared label
costs conditioning but not identifiability.} Fine-tuning a class the model
has never had is the maximal case of the mechanism: the head is extended by
a zero row, so the model assigns the new examples no probability at all and
is maximally wrong about every one of them. Measured on a CIFAR-10
multilayer perceptron (best test accuracy $53\%$ — the weak regime) whose
head is extended to $m=11$, with eight CIFAR-100 flowers all labelled into
the new class$^\dagger$ (job 656205): every imprint lies within a factor
$2.5$ of the largest (relative $0.4$--$1.0$) and $\operatorname{rank}B_T=8$.
The new category is recorded in full.

The obvious objection is that the zero row \emph{constructs} the wrongness, so
the full recording might be an artefact of the initialisation rather than a
consequence of \eqref{eq:imprint}. It is not$^\dagger$ (job 763805). Replacing
the zero row with a random one leaves the conclusion unchanged: the initial
margins spread across $-10.5$ to $-0.41$ rather than sitting at a constructed
extreme, the release still has norm $0.95$, and the search still recovers
eight of eight from the same $38.4\%$ of random starts. Two details inside
that arm make the point better than the headline number does. None of the
letters is chance-correct even with a random row --- a model has no reason to
put an unseen category anywhere useful --- and the two letters whose margins
sit nearest zero carry the two lowest imprints ($0.07$ against $0.14$--$0.28$
for the rest), which is \eqref{eq:imprint} ordering examples \emph{within} the
arm that was built to test the construction. The zero row is the clean
statement of a class the model does not have; it is not what produces the
effect.

The eight imprints are also \emph{aligned}, which is a second mechanism and
not the same one: the Gram of the stacked imprints has
$\sigma_N/\sigma_1=0.145$ (raw) and $0.097$ (on-chart) with mean pairwise
cosine $0.50$--$0.57$, because every error points toward the same corner of
the simplex. That is the mirror image of the confident-batch case above,
where the imprints were \emph{absent}. Both depress $\sigma_{\min}(D\rho)$,
and what distinguishes them is magnitude against angle — which is why the
imprint Gram is worth recording beside the imprint norms. Alignment,
however, is not a wall: $\sigma_{\min}$ at the truth is $2.4\times10^{-6}$
and the inversion reaches the residual floor in $62$ iterations, recovering
the chart coordinates to $7\times10^{-14}$. A shared label costs
conditioning and nothing more.

Two details support the reading. The base model files the eight flowers as
frog, cat, truck, truck, frog, automobile, frog and deer — several classes,
so their errors are only moderately aligned, which is the within-cell version
of the same prediction: had they all been assigned to one class the
alignment would have been stronger. And an attacker's prior built from
public flowers draws them to $0.305$. This cell isolates the new-class and
shared-label structure rather than replicating the encoder-quality ladder,
since its base model is weak by construction.

That comparison has now landed, and it is the cleanest statement of the
mechanism in this section because both extremes sit on one encoder,
one eleven-row head and one recipe$^\dagger$ (job 658575, raw images).
Eight ordinary MNIST digits, at median margin $17.9$, leave imprints spanning
$7\times10^{-22}$ to $1$ with $\operatorname{rank}B_T=6$ of $8$ and mean
pairwise cosine $0.04$: mostly absent, and mutually unaligned. Eight EMNIST
\emph{a}'s through the same head, at median margin $-10.8$, leave imprints all
between $0.3$ and $1$ with $\operatorname{rank}B_T=8$ and cosine $0.52$:
present in full, and aligned. The two axes are therefore independent. Model
quality governs how much of \emph{ordinary} data is recorded --- that is the
encoder-quality ladder --- while novelty of the category governs whether that
suppression applies at all. A confident model records almost nothing of what
it already knows and everything of what it has never seen.

Two further measurements close this off$^\dagger$. The mechanism is visible
\emph{within} a single batch, not only between batches: on a font set the
strong model records the three characters it misreads (relative imprints
$3\times10^{-1}$ to $7\times10^{-1}$) and not the five it gets right
($10^{-17}$), so the split follows each example's own error and nothing else
(job 644064, three encoders against three foreign sets). And the new-class
effect does not depend on encoder quality, which was the pre-registered
prediction: a new class gives $\operatorname{rank}B_T=8$ with negative margins
on the strong, mid and weak encoders alike, while ordinary digits move with
quality --- rank $6$ on the strong model against $8$ on mid and weak (job
658575). Model quality and category novelty are therefore not merely
independent axes; the second is \emph{flat} in the first.

The practical reading is the uncomfortable one. The regime in which this
channel closes is the regime in which fine-tuning teaches a model more of what
it already does; the regime in which it stays fully open is the one where
fine-tuning adds a capability the base model lacks --- which is the canonical
reason to fine-tune at all. The defence found above is real but it does not
apply where adapters are actually used. (The on-chart row of the same job is
excluded: the chart there is fitted to letters and projects digits badly
enough that the model misreads them and records them all, which is a chart
artefact rather than the comparison.)

\paragraph{On a strong model, ordinary private data is dominated by one
imprint --- which is not the same as recording only one.} The imprint spectrum can be read off the release without
any private knowledge, and $\sigma_2/\sigma_1(B_T)<10^{-3}$ is the
attacker's test for ``one example carries this''. Its frequency over the
forty batches$^\dagger$ (job 650891): $9$ of $10$ on the strong encoder
($\sigma_2/\sigma_1$ from $4.8\times10^{-16}$ to $0.07$; ranks
$1,1,1,3,3,4,6,6,6,8$), and $0$ of $10$ on both the weak and the mid
encoders ($0.021$--$0.58$ and $0.005$--$0.69$). Both \emph{ordinary} random
draws trigger on the strong model ($2.1$ and $2.9\times10^{-5}$ raw), and
the single non-trigger there is the batch built to contain two low-margin
examples. What the test detects, though, is \emph{one dominant direction} and
not \emph{one recorded example}, and the two are different claims: of the
nine triggering batches the rank is $1$ in three, $3$ in two, $4$ in one,
$6$ in three and $8$ in one — a full-rank release with
$\sigma_2/\sigma_1=2.4\times10^{-4}$. The split by chart matters, because the
certificate and any chain built on it operate on-chart: raw, the trigger
fires in $5$ of $5$ and the ranks are $1,1,3,6,6$; on-chart it fires in $4$ of
$5$ with ranks $1,3,4,6,8$. Chart projection degrades the images, the model
becomes less certain, and \emph{more} is recorded --- so the on-chart figure,
which is the one an attacker meets, is the less favourable of the two for a
defender. So a single dominant imprint is what
ordinary private data produces on this $97.9\%$ model and never on the
weaker ones; a literal reduction to one recorded example happens in three of
the ten.\footnote{The test can fire for the wrong
reason. The random encoder produces one false positive
($\sigma_2/\sigma_1=8.7\times10^{-4}$ at full rank $8$) on a batch of eight
$1$s, all at residual $\approx0.95$: there the spectrum reflects the near
collinearity of the features, not a difference in what was recorded. A
spectral gap is evidence about the release, not automatically about the
margins, and a negative control that acts on a non-triggered read is what
makes the test load-bearing.}

That control has now run$^\dagger$ (job 652786) and the trigger behaves. On a
random encoder --- the case built so that a one-image reading would be
\emph{wrong} --- it correctly declines: $\sigma_2/\sigma_1=1.2\times10^{-3}$
with $\operatorname{rank}B_T=8$, where acting on the one-image reading would
have given a residual of $2.7\times10^{-6}$ against a predicted floor of
$0.76$. On the strong encoder it fires on raw digits, the label read is right,
and the image read still requires a chart (error $2.6$ without one). So the
trigger's failure mode is silence rather than a confident wrong answer, which
is the direction one wants of a read that an attacker would act on.

its provenance.} The mechanism predicts that data the base model
misclassifies is recorded in full, and a separate argument predicts that
data unlike the model's own domain is drawn badly by an attacker's prior.
Two foreign sets were used to test both$^\dagger$ (job 644062, strong
backbone, raw digits, $8$ private examples; the in-distribution control is a
random distinct-label draw from the MNIST test split through the identical
pipeline). The first prediction holds. The second was not tested at all, for
a reason worth recording.

holds on data of a kind the model has never seen.

withdrawn; the recording half stands on its own. A provenance test that would
survive this needs foreign digits at \emph{matched} resolution — USPS at
$16\times16$ upsampled exactly as the control is downsampled — and has not
been run.

\paragraph{The certificate is the recipe-free face of the same fact.}
Theorem~\ref{thm:quot} builds $C=\Pi_{\mathrm{row}(B_T)}A_T$ and gives
$CH=0$ with $\operatorname{rank}C=r-N$ under (A4). That is the
\emph{exact} form, and (A4) --- $\operatorname{rank}P_T=N$ --- is a rank
condition, hence a statement about a spectrum having a hard gap. It is worth
writing down what replaces it when the gap is soft, since nothing in the
construction requires exactness. Truncating instead at the top $j$ directions
of $B_T$ gives a certificate $C_j$ with $r-j$ conditions which are satisfied
not exactly but to the discarded tail, $\|C_jh_i\|/\|A_Th_i\|\sim
\sigma_{j+1}(B_T)/\sigma_1(B_T)$. So the general statement is
\emph{spectral}: at a tolerance $\varepsilon$ the usable width of the channel
is $r-j(\varepsilon)$ with $j(\varepsilon)=\min\{j:\sigma_{j+1}/\sigma_1<
\varepsilon\}$, and the exact theorem is the special case in which the
spectrum has a hard gap at $N$, so that $j(\varepsilon)=N$ for every
$\varepsilon$ above the numerical floor. Whether the two coincide in practice
is a property of the release's spectrum and not of the construction; it is
being measured, and \S\ref{sec:pixels}'s gap between the formal and usable
counts is the same distinction seen from the conditioning side. The imprint law says what
happens when (A4) fails, and it is not a degradation but a sharpening. By
Theorem~\ref{thm:traj}, $\mathrm{row}(B_T)$ is spanned by the vectors
$A_0h_i$ weighted by each example's imprint, so when only $N'$ examples are
recorded it is spanned by \emph{their} directions alone: $\operatorname{rank}
C=r-N'$, and $Ch_i=0$ for a recorded example while $Ch_i$ is $O(1)$ for an
invisible one. Measured$^\dagger$ (job 701679, strong backbone,
in-distribution draw): $\operatorname{rank}B_T=6$ with
$\operatorname{rank}C=10=r-N'$ exactly, and the per-image residuals
$\|Ch_i\|/\|A_Th_i\|$ are $8.8\times10^{-15}$, $\mathbf{0.84}$,
$3.7\times10^{-10}$, $6.1\times10^{-10}$, $9.9\times10^{-14}$,
$\mathbf{0.49}$, $3.7\times10^{-10}$, $8.5\times10^{-13}$ — the two bold
entries are exactly the two invisible examples, eight orders of magnitude
clear of the largest recorded one. On the chart-projected version, where all
eight are recorded, all eight residuals fall below $3\times10^{-8}$.

Two things make this worth having beyond elegance. It is \emph{recipe-free}:
$C$ needs no $\eta$, no $T$ and no labels, so an attacker can test any
candidate image for membership without knowing how the adapter was trained.
And the condition $C\varphi(\psi(w))=0$ is linear in the features and needs
no unrolled dynamics at all, a far simpler landscape than the full residual,
at the certificate line $k<r-N'$ rather than $k<m+r-N'$. That line comes from
the kernel rather than from $C$'s rank: $\ker C=\mathrm{span}\{h_i:\text{recorded}\}
\oplus\ker A_0$ has dimension $N'+(n-r)$, so it is of codimension $r-N'$ in
feature space, and a $k$-dimensional chart image meets it in generically
$k-(r-N')$ dimensions. Below the line the recorded examples are isolated
solutions and generically the only ones; above it each lies on a
positive-dimensional family that satisfies the certificate without being the
image. When several examples are recorded they are all equally valid
solutions, so a search started at random reaches \emph{one} of them with no
way to steer and no way to tell from the objective which — the $N'=1$ case is
the one where recovering a solution and recovering \emph{the} example
coincide.
\paragraph{From nothing: the certificate recovers recorded images from random
starts.} The two properties above --- recipe-free, and a landscape with no
unrolled dynamics --- combine into the first arm of this study that an
attacker could actually run. Measured$^\dagger$ (job 706721, strong backbone,
confident batch, chart-projected at $k=6$ with $N'=7$ recorded, so the
certificate line $k<r-N'=9$ is satisfied): from $2000$ random starts at public
scale, $300$ iterations each, with no recipe, no labels, no $N$ and no
proximity to the truth, $51.0\%$ of starts reach the objective's floor; the
fraction of starts landing on a recorded image is also $51.0\%$ --- the same
number, so \emph{every} start that reached the floor is on a private image and
none is spurious, as the kernel count requires below the line. Had the two
fractions differed, the count would be wrong. The two populations are cleanly
separated and no threshold is doing any work: over the $1020$ floor-reaching
starts the image error to the nearest recorded image runs from
$1.3\times10^{-15}$ (min) through $5.3\times10^{-15}$ (median) to
$3.7\times10^{-12}$ (max), while the $980$ that miss sit at objective
$1.1\times10^{-3}$ and image error $0.47$ at the median, with nothing in
between. All seven recorded images are found, the $\operatorname{argmin}$ over
starts lands on a recorded image at $5.5\times10^{-15}$ from its chart
projection, no start is degenerate, and the whole cell costs $389$\,s at
$0.2$\,s per start.

Which image a start finds is not uniform: the landings divide $414$, $383$,
$99$, $63$, $37$, $13$, $11$ across the seven, so the basins differ by a factor
of forty and the rarest example needed of order $180$ starts before it appeared.
Finding \emph{some} recorded example is therefore much cheaper than finding
\emph{all} of them, and the gap widens with the skew --- a scaling caveat that
belongs with any statement about recovering a whole batch. What sets the skew
is \emph{not} determined: four candidates were pre-registered and tested
against the landings over the seven recorded images (job 716233) --- the
imprint $\|C_i\|$, the margin, $\|A_T\varphi\|$ and the chart-space norm ---
and they order it at $17$, $16$, $15$ and $16$ concordant pairs of $21$, none
of them a discrimination at $n=7$, where the standard deviation of Kendall's
$\tau$ under the null is $0.32$ and the best of four correlated candidates is
expected to reach two of those by chance. The release-side predictors agree
with each other at $20/21$ --- one quantity wearing two hats --- while the
chart-space norm is independent of them at $12/21$ and orders the landings
just as well, and within the three largest imprints the ordering inverts
outright. Basin size is not a corollary of the
imprint law.

Pooling settles the direction, as the single cell could not$^\dagger$ (job
734584, per-cell $\tau$ over the seventeen on-chart cells with landings, $331$
pairs, weighted by pairs; the seven at- and above-line cells with no landings
are excluded rather than counted as $\tau=-1$). The chart-geometry
predictors lead --- the feature norm $\|A_T\varphi(x)\|$ at $+0.43$, positive
in $14$ of $17$ cells, and the chart-space norm at $+0.40$, positive in $15$
--- against the release-side pair at $+0.18$ and $+0.15$. Pooling per batch
first, which is the honest unit since cells within a batch share their images,
separates the two geometry predictors: the feature norm holds at $+0.44$ and
$+0.50$ across the two informative batches, while the chart-space norm is
$+0.55$ in the larger and $+0.17$ in the other. The feature norm is the one
that replicates and the one to name; the chart-space norm is the stronger
predictor in the batch supplying thirteen of the seventeen cells and does not
carry to the second. The two families are
nearly uncorrelated with each other ($+0.10$ and $-0.05$) while the imprint
and the margin agree at $+0.96$, confirming those two were one quantity
throughout. So \emph{presence} follows \eqref{eq:imprint} and search cost does not.

Two reservations keep this a strong lead rather than a result. The seventeen
cells come from three batches across several ranks and are not independent, so
the pooled $z$ of about five is inflated. Per batch there are only two
informative units --- the third contributes a single cell with at most one
landing per image --- so the $14$ of $17$ rests on two batches and is
dominated by one. And the norm predictors had an alternative
explanation --- starts are drawn at a fixed public scale, so a target whose
norm sits nearer that scale is nearer the starts --- which redrawing the
starts at half and twice that scale was run to exclude, and did not$^\dagger$
(job 737516). Three findings came back. The \emph{aggregate} yield is
scale-insensitive: $69.4$, $65.6$, $71.2$ per cent at $0.5\times$, $1\times$,
$2\times$ in one cell and $71.6$, $74.4$, $67.0$ in the other, so the public
scale is not privileged and sweeping it buys no extra recoveries. The
\emph{distribution} over targets is not: at $0.5\times$ the landings
concentrate on the dominant image and two of seven recorded images receive
none at all, while $2\times$ flattens the skew about fivefold and finds all
seven. And the feature-norm ordering is itself not scale-invariant --- $16$,
$23$, $18$ of $28$ concordant pairs at the three scales --- so the $+0.43$
above is a statement at one scale and not a property of the chart alone.

The honest form is therefore narrower than the pooled table suggests. The
\emph{dominant} target is the same at every scale, and that much is the
chart's; the ordering of the remainder moves with the start distribution,
which is the attacker's to choose. That converts the earlier
coupon-collection caveat into a technique rather than a limit: sweeping the
start scale does not raise the yield, but it raises the \emph{coverage}, and
a single scale can leave recorded images with no landings at all.

Three scopings, all of which matter, and the first is severe. The batch is
\emph{on-chart}: the adapter was trained on the $k=6$ projections, so those
projections are the private data and are what is recovered exactly. Against a
raw digit the error is the chart's, which is the factorisation of
\S\ref{sec:measured} and not a defect of this cell --- but at $k=6$ that
factor is doing much of the work, though less than the eye suggests. The
projections look like blurred blobs, and by inspection several appear to have
lost their digit; the base model disagrees, reading $6$ of the $8$ as their
own class ($4\to9$ and $5\to8$ are the failures), so $5$ of the $7$ recorded
examples keep their class through a six-coordinate chart. Two different
questions hide behind ``recognisable'' and they should be reported separately.
Class survival --- what a classifier reads off the projection --- is high even
here. Instance survival --- whether \emph{this} handwriting is identifiable
rather than a generic member of its class --- is what the representation error
measures, and it is large at $k=6$. Privacy cares about both, and they come
apart in exactly this regime, so the fidelity axis of \S\ref{sec:opens}'s rank
ladder carries the classification accuracy and the representation error side
by side rather than one standing for the other. This cell demonstrates the
mechanism end to end; fidelity is the ladder's question, since $k$ may run to
$r-N'-1$ and a rank-$64$ adapter admits a chart of forty-odd coordinates.

The fidelity axis is measurable without running any attack, because whether a
chart of dimension $k$ preserves a digit is a property of the chart and $k$
alone --- no adapter, no release, no rank, no search. Measured$^\dagger$ (job
719793, PCA chart fitted on $5\times10^4$ training images, evaluated on $2000$
held-out test digits, strong $98\%$ model): the base model's accuracy on the
$k$-projections runs $0.20$, $0.31$, $0.52$, $0.68$, $0.75$, $0.83$, $0.88$,
$0.94$, $0.95$, $0.97$ at $k=2,4,6,8,10,12,16,20,24,32$ and is flat at $0.977$
by $k=64$, against $0.979$ on the raw digits; the median representation error
falls from $0.73$ to $0.29$ over the same range. Two marks turn this into the
rank statement. At $r=16$ with seven recorded, the admissible charts $k\le8$
keep $52$--$68\%$ of digits classifiable; at $r=64$ with eight recorded the
admissible chart runs to $k=55$, where the projection is at the raw accuracy
of the model. \emph{The LoRA rank is a leakage dial}, and it is the knob
raised for utility. Note that the private batch's own $6$ of $8$ at $k=6$ sits
twenty points above the held-out $52\%$: the batch is the illustration, the
curve is the number, and quoting the first for the second would have
overstated the case.

\paragraph{Three disclosures, three thresholds.} ``Recognisable'' conceals
three different statements, and measuring them apart (job 721003, nearest
neighbour among the full $10^4$-digit raw test pool) is what makes the channel
interpretable.

\emph{Instance} --- is \emph{this} handwriting identifiable, rather than some
member of its class? Self top-$1$ retrieval by $k$: $0.001$, $0.01$, $0.04$,
$0.11$, $0.21$, $0.33$, $0.57$, $0.73$, $0.84$, $0.94$ at
$k=2,\dots,32$, reaching $0.99$ by $k=48$. \emph{Class} --- is the digit's
identity there? $0.52$ at $k=6$ and $0.68$ at $k=8$, as above. The two come
apart exactly where the rank-$16$ certificate lives: at $k=6$--$8$ the
projection reads as the right class half to two-thirds of the time while
identifying the specific digit only $4$--$11\%$ of the time, and retrieves a
\emph{different} member of the same class more often than its own ($0.53$ and
$0.58$). ``There was a $4$'', not ``this $4$''. By $k\ge32$, admissible at
$r=64$, both survive at $0.94$--$0.99$. So the rank does not merely sharpen
the picture; it moves the release across the boundary between a class
disclosure and an instance disclosure.

\emph{Membership} is the third and it has no threshold in $k$ at all. An
attacker holding a candidate pool need not render an image: they project each
candidate through the same chart and compare coordinates with the recovered
$\hat w$. Projection is deterministic, so this identifies the source among
$10^4$ candidates at every $k$ tested, $k=2$ included. The requirement is not
fidelity but recovery --- $k$ below the line and inside the basin --- so the
blurriness of the reconstruction is no defence against the membership
question, which is the one most privacy regimes actually ask. The obvious objection --- that this rests
on machine precision, since at small $k$ ten thousand projected candidates are
densely packed --- does not survive measurement (job 723107). Perturbing the
recovered coordinates by $\varepsilon$ times their standard deviation and
retrieving again, the source stays top-$1$ for \emph{every} start at
$\varepsilon\le3\times10^{-2}$ for all $k\ge8$, and at $96$--$100\%$ for
$\varepsilon=0.1$; only $k=2$ is fragile, holding $95\%$ at
$3\times10^{-3}$ and falling to $63\%$ at $10^{-2}$. Since solves land at
$10^{-15}$, a recovery that stops short of the floor by ten orders still
identifies its source. Membership is therefore the disclosure that survives
everything this section otherwise limits: it needs no fidelity, no recipe, no
labels, and no precision to speak of. The attacker needs the
release, the public base model, and a chart; that is all, and in particular
none of $(\eta,s,T)$, the labels, $N$, or a starting point near the answer.
And the price of needing nothing is the lower line: $k<r-N'$ here rather than
$k<m+r-N'$, six coordinates per image rather than twenty-two. At $k=8$, one unit further, the
basin falls to $18\%$ --- which raises the question of what the basin depends
on, and the answer decides whether the channel is practically narrow or not.

\paragraph{The basin is governed mostly by distance below the line, and only
weakly by $k$.}
Holding the chart, the images, $k=8$ and the $500$ starts fixed and moving
only the rank$^\dagger$ (job 721391, confident batch on-chart, $N'=7$): at
$r=16$, where $k=8$ sits one unit below the line, $16.6\%$ of starts land on a
private image and $6$ of the $7$ are found; at $r=32$, where the same $k=8$
sits seventeen units below, $74.4\%$ land and all seven are found with the
$\operatorname{argmin}$ on a private image; and at $r=64$, forty-nine units
below, $96.4\%$. A control batch runs $4\%$ (at its line), $73.4\%$, $89.0\%$
over the same three ranks. The basin therefore rises monotonically with the
slack and saturates. Neither below-line cell has a spurious zero.
The number of unknowns is identical in the two, so the slack below the line
and not the dimension of the search is what moves the basin over that range.
The mirror experiment bounds the other side: holding the slack roughly fixed
and varying $k$ instead, $r=64$ with $k=40$ ($16$ below) gives a per-image
basin of $5.7\%$ against $r=32$ with $k=8$ ($17$ below) at $10.6\%$, so
thirty-two additional unknowns cost about a factor of two. Distance dominates,
$k$ is not free, and the honest form of the claim needs both halves.

A second batch's ladder shows what happens on approach to the boundary. At
$r=64$ the per-image basin runs $71.6$, $51.6$, $18.2$, $6.6$ and $0.86$ per
cent at $k=24,32,40,48,56$, and the number of recorded images found runs
$7,7,7,6,5$ of seven. So the earlier reading --- that every admissible $k$ is
attackable and only the cost varies --- holds across most of the range and
fails within a few units of the line, where the basin becomes small enough
that some recorded images go unsampled at a fixed start budget. That is a
sampling limit rather than an identifiability one, but it is the attacker's
limit all the same.

One confound runs through the whole ladder and belongs with it. In the
on-chart construction the chart does not merely bound the attacker's search:
it \emph{defines} the private data, since the adapter is trained on the
projections. Raising $k$ therefore makes the projections more faithful, which
makes the base model more certain about them, which lowers what the release
records --- the same mechanism as the confident batches above, now running
along $k$ on a fixed batch. It is visible in a control batch whose $\|B_T\|$
falls from $3.7\times10^{-2}$ at $k=16$ to $1.7\times10^{-6}$ by $k=24$, with
$N'$ going $8\to7\to6$. So $k$ moves three things at once --- the dimension of
the search, the fidelity of what is recovered, and how much there is to
recover --- which is why $N'$ and the line are read per cell, and why no
single-$k$ comparison across batches means anything without them.

This makes the rank a leakage dial in two independent ways, which is the
sentence to carry: raising $r$ raises the admissible $k$, so the attacker may
use a richer chart and recover a sharper image; and at any fixed $k$ it
enlarges the basin, so the recovery is more likely to be found at all. Both
move the same way, and $r$ is raised for utility. The synthesis is no longer a prediction.

\paragraph{A second cap: the certificate needs $N'\le m-1$.} The line
$k<r-N'$ is necessary but not sufficient, and the missing condition comes
from the same simplex constraint that supplies the $-1$ in
Proposition~\ref{prop:simplex}. By \eqref{eq:imprint}, $B_T=\sum_i
q_i(A_0h_i)^{\!\top}$ with each $q_i$ carrying the softmax's zero column sum,
so at most $m-1$ of the $q_i$ are independent and
$\operatorname{rank}B_T\le\min(m-1,r,N')$. When $N'>m-1$ the row space is a
proper subspace of the recorded examples' feature span, aligned with none of
them individually, and $Ch_i$ vanishes for \emph{no} recorded example ---
the certificate fails for all of them at once rather than degrading.
Measured$^\dagger$ (the twenty-image cell, $10$-class head): twenty digits all
with $O(1)$ imprints give $\operatorname{rank}B_T=9$ with a clean spectral gap
($1$, $0.8$, $0.3$, $0.2$, $0.1$, $0.09$, $0.03$, $0.01$, $0.005$,
$3\times10^{-16}$) and certificate residuals of $10^{-2}$ to $0.5$ for every
one of the twenty in FP64. The head width therefore bounds the certificate
channel twice over --- it caps how many examples can be recorded in a
recoverable way, and $m$ is the defender's architecture rather than the
attacker's choice.

The cap was then tested where it predicts the channel switches back
\emph{on}, which is the stronger direction$^\dagger$ (job 725918). The same
network was retrained on the same MNIST data with a padded $26$-logit head,
sixteen logits never targets, reaching $98.28\%$ against the ten-logit model's
$97.9$--$98.2\%$: only $m$ changed, and data, encoder, batch, rank and chart
were held fixed. On the identical twenty digits at $k=8$, $r=64$,
$\operatorname{rank}B_T$ goes from $9$ to $19$ and $\operatorname{rank}C$ from
$55$ to $45$, and the per-image certificate residuals fall from $0.3$--$0.5$
for all twenty to between $2.5\times10^{-8}$ and $4.3\times10^{-3}$ for
nineteen of them. The twentieth stayed at $0.23$ at the default
truncation, making the counts $19$ and $45$ against the predicted $20$ and
$44$ --- and a sweep over the tolerance at which $\mathrm{row}(B_T)$ is
truncated (job 727654) resolves that in favour of the prediction: the
annihilated count runs $0,1,10,16,18,18,20$ as the tolerance tightens from
$10^{-8}$ to $10^{-14}$, reaching all twenty with $\operatorname{rank}C=44$
exactly. The shortfall was the truncation, not the release. The image is
nonetheless instructive, because its imprint is $3\times10^{-7}$ of the
largest while its singular value is $6\times10^{-14}$ --- seven orders of
collinearity with the other nineteen. So what decides whether an example is
individually recoverable is not the size of its imprint but the size of the
\emph{independent} direction it contributes, and the two can differ by orders.
That the same sweep returns zero at every tolerance and every precision on the
ten-class head confirms the $m-1$ cap is structural and not an artefact of any
threshold.

\paragraph{Precision of the release is a privacy parameter.} Because the
certificate's reach is set by which directions are numerically inside
$\mathrm{row}(B_T)$, and the imprint spectrum falls steeply, rounding the
released tensors removes the examples whose imprints fall below the rounding
noise. Measured$^\dagger$ (job 722663, counting images with
$\|Ch_i\|/\|A_Th_i\|<10^{-3}$) on a batch whose $B_T$ spectrum runs
$1$, $2\times10^{-4}$, $10^{-5}$, $9\times10^{-9}$, $4\times10^{-9}$,
$10^{-11}$, $2\times10^{-16}$, the counts fall $6\to3\to0\to0\to0$ from FP64
through FP32, TF32, FP16 to bfloat16, and fall monotonically in all four
batches tested. Quantisation noise is $3\times10^{-8}$ at FP32 and
$10^{-3}$--$2\times10^{-3}$ at bfloat16, which is what sets the cut.

The count must be read by the \emph{gap} and not by a fixed threshold, since
a threshold at $10^{-3}$ sits at bfloat16's own noise and would return zero by
construction. Counting an example as readable when its residual lies two
orders below the smallest invisible residual of the same release, bfloat16
still leaves separable examples: sweeping the truncation
tolerance as well (job 727654) the walls are $6,6,5,2,5$ and $3,3,3,1,3$ over
FP64, FP32, TF32, FP16 and bfloat16 for two eight-image batches, so the
single-tolerance counts above are a lower bound. The ordering is not
monotone in the mantissa and the reason is the exponent: FP16 carries ten
mantissa bits to bfloat16's seven but only five exponent bits to its eight, so
FP16 \emph{underflows} the small imprints to zero while bfloat16 keeps them
coarsely. Since the imprint spectrum spans many orders, dynamic range decides
and not precision --- with the awkward consequence that bfloat16, the format
deployment actually uses, is the most revealing of the low-precision formats
and FP16 the least. Where the count does fall to zero it falls for a
structural reason: in the twenty-image wide-head cell the walls are $20$
(FP64), $17$ (FP32) and none below, because with twenty comparable imprints
the rounding noise sets even the strongest residual. So the honest statement
is that \emph{low precision narrows the recipe-free channel and can close it
when many examples are comparably recorded, but a batch dominated by a few
strong imprints survives quantisation} --- the dominant direction survives and
everything beneath the rounding noise merges into the invisible band. That is
still a defender-side lever already in wide use for unrelated reasons: a
quantised adapter is not merely smaller, it records fewer examples
recoverably. The scope is the certificate channel; the full-residual channel
at low precision is a separate measurement.

Two caveats belong with it. The vanishing is approximate rather than exact
below full rank — the recorded residuals sit at $10^{-10}$, not at machine
precision, because the invisible examples' contribution to $M_T$ does not
identically vanish — so the test needs a threshold, and the gap it must
straddle is eight orders wide here. And $Ch=0$ is membership of a subspace,
not of the private set: any candidate whose features lie in the span of the
recorded ones passes, which is tight when $N'=1$ and a $6$-dimensional
subspace of $\mathbb R^{1000}$ when $N'=6$. Under Adam
$\operatorname{rank}B_T=r$, $C\equiv0$, and none of this applies.

\subsubsection*{What an attacker gets from it}

\paragraph{The three ingredients combine.} Budget, basin and fidelity were
measured separately; the cell that puts them together is $r=64$ with
$k=32$, where the chart is admissible ($N'=8$, line $56$), sits twenty-four
units below the line, and is the dimension at which a projection identifies
its own source among $10^4$ candidates $94\%$ of the time. Measured$^\dagger$
(job 728592, confident batch, chart-projected, $500$ random public-scale
starts of $300$ iterations, no recipe, no labels, no $N$): $65.6\%$ of starts
land on a recorded image, \emph{all eight} are found, and the
$\operatorname{argmin}$ over starts lands on a private digit at
$2.2\times10^{-14}$. No start is degenerate and no floor-reaching start is
spurious. The cell reproduces independently inside the ladder job ($0.660$,
eight of eight, $\operatorname{argmin}$ correct). At this $k$ the base model
reads all eight projections as their own digit and the representation error
is $0.400$.

The chart is public: PCA on the first $5\times10^4$ images of the train
split, while every private batch is drawn from the test split --- two disjoint
index files, verified in the code path that produced these rows, not inferred
from convention. The per-image best errors are $10^{-14}$ for five of the
eight digits and $10^{-5}$ to $9\times10^{-5}$ for the other three, the latter
still converging at the $300$-iteration cap.

So on a $98\%$ MNIST model, from the released rank-$64$ adapter, the public
base model and a public chart --- and no recipe, no labels, no batch size and
no starting point near the answer --- an attacker recovers every recorded
private example, and at that chart dimension the recovered object identifies
its source rather than merely its class. That is the claim the whole construction was built to test.

\paragraph{The scope that must travel with it: the release it was recovered
from has norm $7.6\times10^{-18}$.} Measuring $\|B_T\|$ against $k$ on the
same batch$^\dagger$ (job 752500) shows no window but a monotone collapse:
$0.47$, $0.23$, $0.081$, $1.6\times10^{-13}$, $7.6\times10^{-18}$,
$1.9\times10^{-22}$, $2.8\times10^{-24}$ at $k=8,12,16,24,32,40,56$, tracking
the projections' minimum margin as it rises from $0.02$ to $56$. Twenty-three
orders. The rank's apparent dip and return is a relative artefact of that
collapse: past $k=24$ all eight imprints shrink together, so their ratios
compress against a vanishing $\sigma_1$. In FP64 the certificate is
scale-free --- it is an angle --- so the recoveries above are exact and the
numbers stand. But they were obtained from a release that is numerically
nothing, and that has a consequence the file's storage precision does not
capture.

What that collapse does \emph{not} do is close the channel, and the reason is
worth following because the obvious argument gets it backwards. An imprint is
an accumulated residual $1-p_y\approx e^{-\text{margin}}$, and at margin $42$
that is $5.7\times10^{-19}$ --- far below FP32's unit roundoff, so
$p_y$ rounds to exactly $1$ and the own-class entry of $D$ is identically
zero. But $D$'s other entries are the off-class probabilities themselves,
$e^{z_j-z_y}/Z$, represented directly rather than as a difference from one,
and those survive to the format's \emph{subnormal floor}: $1.4\times10^{-45}$
in FP32 and $1.2\times10^{-38}$ in bfloat16. Unit roundoff zeroes one row of
the imprint; the rest is intact. So a release of norm $7.6\times10^{-18}$ is
produced, and stored, by FP32 or bfloat16 training without difficulty, and
since the certificate is an angle rather than a magnitude it is exactly as
informative as a release of norm one. Only FP16 erases it, and by range rather
than precision --- its smallest subnormal is $6\times10^{-8}$, so both the
residuals during training and the released file underflow to zero.

Two things follow that are stronger than the storage question. At this scale
the adapter never feeds back: $10^{-18}$ against logits of order $40$ is below
the last place even in FP64, so $A_T=A_0$ and $B_T$ is $T$ times the one-step
gradient in closed form. \emph{The fine-tune changed nothing about the model
and recorded every private example anyway} --- leakage is not conditional on
the update having had an effect, which is the opposite of the intuition that a
fine-tune too small to matter is a fine-tune too small to leak. And
confidence, which collapsed $\|B_T\|$ by twenty-three orders, does not
therefore protect: it makes the release numerically tiny without making it
less identifiable, in every format with FP32's exponent range.

Measured$^\dagger$ (job 753371, the headline release round-tripped through
each format and re-attacked with the same $500$-start search). Of the eight
images FP64 recovers: FP32 recovers five, TF32 five, and FP16 none. The three
lost under FP32 are exactly those recorded at $7\times10^{-11}$,
$3\times10^{-11}$ and $2\times10^{-12}$ of the strongest, and they are
destroyed rather than merely hard --- the certificate residual at their own
truths is $0.9$. Two things the pre-registration got wrong, both in the
attacker's favour: a quantised file's spectrum floor sits about three orders
\emph{below} the format's unit roundoff, so TF32 keeps five where two were
predicted; and the attacker's truncation tolerance should be tight rather than
matched to the noise, since noise-matching discards a recoverable image. And
one structural observation: the aggregate basin stays between $48\%$ and
$67\%$ across every format, so precision changes \emph{which} images are
reachable and not how hard they are to reach.

The FP16 zero is this cell's own, and must not be generalised. Two thresholds
govern: the format's unit roundoff, which is relative, and its subnormal
floor, which is absolute. Which binds depends on $\|B_T\|$. Here the release
is $7.6\times10^{-18}$ and FP16's smallest subnormal is $6\times10^{-8}$, so
the entire file underflows to zero --- a statement about a release that had
collapsed twenty-three orders, not about FP16. For a release of order
$10^{-6}$ to $1$, which is what data the model gets wrong produces, the floor
does not bind and the roundoff does. So the honest form of the storage
statement is that \emph{FP32 and TF32 preserve everything recorded above about
$10^{-10}$ of the strongest imprint} --- ten orders of dynamic range, which is
essentially no protection --- while FP16 protects only when the release is
already too small to have moved the model.

\paragraph{The canonical case has none of the corner's caveats.} Everything
above is a demonstration on a batch the model was certain of. The case that
matters is the opposite one, and it is also what people fine-tune for: a class
the base model does not have. Measured$^\dagger$ (job 760909, EMNIST
\emph{a} as an eleventh class on the $98\%$ digit model, its head row
initialised to zero, $r=64$, $k=32$ on the letters' own public chart, FP64).
The margins at $t=1$ are $-2.65$ to $-10.3$: the model is wrong about every
letter, and unlike the digits no amount of chart fidelity can make it right
about a class it does not have, so the collapse mechanism has nothing to act
on. By $t=T$ the margins are $+6.0$ to $+13.7$ --- the model learned the
class. The adapter genuinely moved: $A_T$ is $9.3\%$ from $A_0$ and the
feedback $\|sB_TA_TH\|/\|z\|$ is $0.43$, against $10^{-19}$ in the confident
cell. The release has norm $1.00$, the eight imprints are $0.12$ to $0.26$ ---
comparable, all recorded --- and the certificate residuals at the truths are
$2\times10^{-13}$ to $7\times10^{-12}$ \emph{with a moving adapter}. From
$500$ random starts, $38.4\%$ land on a letter, every one at the floor, and
all eight are found; the $\operatorname{argmin}$ sits on a letter at
$3\times10^{-26}$. The chart's own error is $0.235$.

That cell answers each objection the flagship one invites. It is not a
vanishing release ($1.00$ against $7.6\times10^{-18}$), so no format's
subnormal floor touches it. It is not a frozen adapter, so it is not one
gradient step in disguise and the trajectory is doing real work. It is not a
low-dimensional chart, so the recovery is instance-level. And it is the
canonical reason to fine-tune rather than a corner. One qualification stands:
under FP32 \emph{training} the release is unchanged to $4\times10^{-7}$ but
the certificate residuals at the truths rise to $10^{-4}$--$3\times10^{-3}$,
because four hundred steps of accumulated rounding act on a \emph{moving}
adapter --- the opposite regime from the confident cell, where a frozen
adapter made FP32 exact. At a noise-matched tolerance that recovers six of
eight --- but the tolerance was hiding them, not the release. At the tight
tolerance$^\dagger$ (job 764976; the tight read reports $\operatorname{rank}
B_T=11$, which is the inflation described below --- the true $N'$ is $8$ and
the line $56$, and $k=32$ is below it either way) the same FP32-trained
release gives $32.8\%$ of $500$ starts landing on a letter and \emph{all
eight found}, at residuals of $3\times10^{-7}$ to $1.3\times10^{-5}$, with the
two previously missing letters at $5\times10^{-6}$ and $6\times10^{-6}$ and
the $\operatorname{argmin}$ on a letter at image error $1.1\times10^{-6}$.
The qualification is therefore withdrawn: \emph{from an adapter trained in
FP32 on a class the base model does not have, every private example is
recoverable from random starts with no recipe, no labels and no knowledge of
the batch, at a chart dimension where the recovered object identifies its
source.}

At $k=16$ the qualification is already gone$^\dagger$. On the same letters,
with the release both \emph{trained and stored} in FP32 --- ordinary
arithmetic throughout --- $82.6\%$ of $500$ random starts land on a private
letter and \emph{all eight are recovered}, at residuals of $2\times10^{-5}$ to
$6\times10^{-4}$. The FP64 run of the same cell finds seven of eight, and the
eighth is missing for a reason worth separating: its certificate residual at
its own truth is $2.5\times10^{-14}$, so it is perfectly identifiable and
simply received no start in $500$ --- a sampling null, not a precision one,
and the FP32 run happens to sample it. The chart error there is $0.32$, so
this is the class-level statement: on the canonical fine-tuning task, in the
arithmetic everyone uses, every private example is recoverable from random
starts with no recipe, no labels and no knowledge of the batch. Whether the
same holds at the instance-level $k=32$ is the one open row. The result does
not depend on how the new head row is initialised: with a random row instead
of a zero one the margins are still negative throughout at $t=1$, the release
is $0.95$, and FP64 recovers eight of eight at the same $38.4\%$.

\paragraph{Half-precision \emph{training} defeats this attacker, and that is
not the same as protecting the data.} Trained in bfloat16 or FP16, the letter
release keeps its norm (within $12\%$ and $3\%$), its rank ($8$), and its
effect --- the model still learns the class, margins $+4$ to $+14$. What it
loses is \emph{direction}: four hundred steps of accumulated half-precision
rounding move the row space by $2$ to $25\%$ per letter, the certificate
residuals at the truths rise to $0.01$--$0.25$, and the search finds none of
the eight. The certificate route needs the released directions to about
$10^{-3}$, which FP32 supplies and half precision does not. But the release
still \emph{records} all eight --- full rank, order-one imprints, a class the
model demonstrably learned --- so the correct statement is which attacker was
run, not that the data is safe. A route that uses more of the release than the
row space alone is not excluded by any of this, and the recipe route against
these same half-precision releases was run to say so$^\dagger$ (job 771329).

Against the FP32-trained release it succeeds outright: the recipe route ends
$2\times10^{-6}$ from the on-chart truth for every letter, so half of the
statement is settled --- FP32 training protects against neither route.
Against the bfloat16-trained release it fails, and it fails in the more
interesting of the two ways. Not a search failure but an \emph{alias}: the
solver reaches residual $5.8\times10^{-4}$, two hundred and forty times
\emph{below} the $0.138$ that the true images themselves achieve, at per-letter
errors of $0.09$ to $0.77$. An FP64 simulator explains a bfloat16-trained
release better with the wrong images than with the right ones. That is the
standing rule --- consistency with the release is not evidence of correctness
--- arriving through arithmetic mismatch rather than through
non-identifiability, and it is the first place in this work where an
attacker's own objective actively prefers a wrong answer. The per-letter
errors do track how far each letter's row space moved, but not cleanly: two
letters perturbed by $0.08$ and $0.055$ still end at $0.50$ and $0.44$.

So the half-precision statement is: the release records all eight, and neither
of the two routes we ran reads it --- one because it needs directions the
arithmetic destroyed, the other because its forward model is the wrong
arithmetic. An attacker who simulates in the same precision as the defender
trained had not been run, and measuring the landscape it would have to descend
settles most of the question$^\dagger$ (jobs 779207, 779969).

The bfloat16 training map is not the rugged one its rounding suggests. It
responds to a $10^{-6}$ perturbation of the latents with $4.4\times10^{-6}$
(FP64: $1.25\times10^{-6}$), to $10^{-4}$ with $4.4\times10^{-3}$, and to
$10^{-2}$ with $1.6\times10^{-2}$: smooth above a rounding floor of $2$--$4
\times10^{-3}$ and linear beyond it. The release's $12\%$ deviation from FP64
is a systematic bias shared by nearby inputs, not noise. And along the segment
from a $0.1$ start to the truth the \emph{matched} residual falls
monotonically --- $0.19$, $0.022$, $5.9\times10^{-3}$,
$2.3\times10^{-3}$, and exactly zero at the truth --- with the windowed means
of a twenty-one-point scan monotone at every radius tested. There is no flat
region and no needle: a descent path exists. So the honest word for
half-precision training is \emph{cost}, not ruggedness, and the earlier
prediction of a noise floor is withdrawn.

What remains is a genuine floor of a different kind. The attacker must
reconstruct $A_0$ from the released $A_T$, and one unit in the last place on
every entry moves a bfloat16 release by $10^{-2}$, so the achievable residual
is bounded by how well $A_0$ can be recovered --- predicted at $10^{-3}$ to
$10^{-2}$, against the mismatched FP64 simulator's $0.138$. And the matched solver reaches
it$^\dagger$ (job 782682, letters at $k=16$, release trained in bfloat16,
simulator in bfloat16 with $A_0$ carried as an unknown). The floor is
$0.0231$ --- the rounded updates leave $2.05\%$ of $A_0$ outside the feature
span --- where the mismatched FP64 simulator sits at $0.173$ on the same
truth. From a $0.1$ start the solver reaches $0.0141$ in twenty iterations and
stops, and the per-letter image errors are $0.022$ to $0.069$, a median of
\textbf{three per cent}. Against the raw letters the median is $0.317$, which
is the chart's own floor of $0.316$: the arithmetic mismatch costs the
attacker three per cent on-chart, and that is invisible beside the thirty-one
per cent the chart costs anyway. Half-precision training is not protection,
demonstrated rather than argued; its price to the attacker is a floor of
$2\times10^{-2}$ in image error where FP64 gives $10^{-15}$.

That price is set by the mantissa, and it orders as one would then expect:
median image error $3.0\%$ from a bfloat16-trained release, $0.97\%$ from an
FP16-trained one (its $A_0$ floor is $2.05\times10^{-3}$ against bfloat16's
$0.0231$, a third of the unexplained fraction), $3.0\times10^{-7}$ from FP32
and $2\times10^{-15}$ from FP64 --- the same start, the same solver, the same
release but for the arithmetic it was trained in. Note that this reverses the ordering of the storage
result above, and for a reason worth stating: reconstructing $A_0$ inside a
matched simulator is limited by \emph{precision}, so FP16's ten mantissa bits
beat bfloat16's seven, while reading a stored release through the certificate
is limited by \emph{range}, so bfloat16's eight exponent bits beat FP16's
five. The same two formats swap places depending on which of the two the
attacker is up against.

One further mechanism governs how much an arithmetic mismatch costs, and it is
the chart's conditioning rather than the mismatch's size. The same $2.8\%$
mismatch --- an FP64 simulator against an FP16-trained release --- displaces
the solution by $0.074$ at $k=16$ and by $0.47$ at $k=32$, where
$\sigma_{\min}$ of the pull-back is $10^{-5}$. A richer chart buys fidelity in
exact arithmetic and pays for it by amplifying every arithmetic error into a
displacement, so under imperfect arithmetic the attacker's best $k$ is bounded
by conditioning and not only by the line. Scaling the $k=16$ column by that amplification
predicted roughly $6\%$ for FP16 and $19\%$ for bfloat16 at $k=32$. The rows
give $7.5\%$ and $4.6\%$ at full descent --- $4.20\%$ and $4.53\%$ at each
format's knee, for reasons the next paragraphs give --- so the prediction was
right for FP16 and wrong by a factor of four for bfloat16, and the reason is
worth more than the prediction was.

Image error is the larger of two terms, not one. A \emph{floor} term set by
how well $A_0$ can be reconstructed, which does not depend on the chart; and a
\emph{conditioning} term, the arithmetic mismatch amplified by
$1/\sigma_{\min}$ of the pull-back, which does. The measured amplification
from $k=16$ to $k=32$ separates them cleanly: $8.9\times$ for FP64,
$19\times$ for FP32, $7.7\times$ for FP16 --- and $1.5\times$ for bfloat16,
whose error was already sitting on its own floor ($3.0\%$ against a floor of
$2.3\%$) and so had nothing to amplify. My prediction applied a conditioning
factor to a quantity that was floor-dominated.

FP16's $7.5\%$ at $k=32$ against bfloat16's $4.6\%$ looks like a reversal of
the precision ordering, and is not one. It is \emph{over-descent}, and the
early-stop test settles it$^\dagger$: FP16 halted at bfloat16's residual gives
$4.72\%$ against bfloat16's $4.56\%$, where running to its own floor gives
$7.5\%$. Information the solver can recover by stopping sooner was never
destroyed. At each format's best stopping point the ordering holds at both
charts --- FP16 $0.97\%$ against bfloat16's $3.0\%$ at $k=16$, and $4.20\%$
against $4.53\%$ at $k=32$.

This is the same phenomenon as \S\ref{sec:measured}'s observation that the
image finishes long before the residual does, in its sharpest form. Because
$A_0$ cannot be reconstructed exactly, the matched objective's minimiser is
not the truth, so image error falls as the residual falls, reaches a
\emph{knee}, and then rises again as the solver closes on a displaced
minimiser. At $k=32$ FP16 passes the knee --- $4.2\%$ at residual
$4.2\times10^{-3}$ becoming $7.5\%$ at $2.8\times10^{-3}$, a factor of $1.5$
in residual costing a factor of $1.8$ in image error --- while bfloat16's
coarser floor of $1.2\times10^{-2}$ sits above the knee, so it cannot
over-descend and cannot reach the optimum either. FP32's minimiser is
essentially the truth, so it has no knee and descends monotonically to
$5.8\times10^{-6}$. The knee lifts with conditioning, which is why
over-descent appears only at the harder chart: at $k=16$ it sits below FP16's
floor and the descent is monotone. \emph{The attacker's rule is to stop at the
knee, not at the floor.} The curve is now swept rather than inferred from two
points$^\dagger$ (jobs 86888, 87369, 88743), and every part of the
pre-registration holds. All three formats coincide on the left arm --- at a
stopping residual of $0.08$ they give $5.25$, $5.24$ and $5.26$ per cent ---
and separate only past the knee. FP16 traces the full U: $5.25\%$ falling to a
minimum of $4.20\%$ at a stop of $5\times10^{-3}$, then rising to $7.47\%$ at
its floor. Bfloat16 is flat rather than U-shaped, $4.53\%$ at its best stop
against $4.56\%$ at its floor, because that floor sits at the knee and it
cannot over-descend past what it cannot resolve. And FP32 has \emph{no} right
arm at all, descending monotonically to $10^{-4}$ --- the falsifier, since an
upturn there with no displaced minimiser to chase would have refuted the
reading. At each format's best stop the precision ordering is the ordinary one,
$4.20\%$ against $4.53\%$.

The cause is underflow specifically, not inaccuracy in general --- bfloat16 is
the less accurate format and barely suffers, because it barely underflows
($1$--$2\%$ of residual entries flushed against FP16's $28$--$35\%$), while
FP32 rounds nothing away and has no knee at all. My earlier reading named
underflow correctly and then drew the wrong conclusion from it: flushed
entries displace the minimiser, they do not destroy the information, and the
loss is the solver chasing that displacement. Stopping restores the ordering
but not the margin --- FP16 recovers about threefold better than bfloat16 on
the easy chart and only slightly better on the hard one --- so the honest
claim needs its access model attached, because every matched-replay cell in
this study starts from the truth perturbed by $10\%$. What is established is
that \emph{half-precision training does not destroy the information}: matched
replay from a near start recovers to a few per cent, so the release still
contains the images. What it does do is break the \emph{certificate} route ---
the only route here that runs from random starts --- which returns none of the
eight on a half-precision-trained release. The replay route that succeeds
needs a starting point no attacker has: none of $20$ attacker-buildable starts
reaches the floor in any replay cell. So for a bfloat16-trained adapter the
search has, today, no attacker-available beginning, and the honest sentence is
\emph{recorded, recoverable in principle, and out of reach of every start we
can actually build}. That is an extraction cost and not an information bound,
but it is a cost the attacker cannot currently pay. Any precision-based
defence would still have to act on numerical \emph{range} rather than
accuracy. Two caveats: the dose-response rests on two FP16 points, a
trend and not a formula, and no format that rounds away more than FP16 has
been tested.

What survives all of it is the claim: at $k=32$, the instance-identifying
chart, a matched attacker recovers the private letters to $5.8\times10^{-6}$
from an FP32-trained release and, at each format's best stopping point, to
$4.53\%$ and $4.20\%$ from the half-precision ones --- against a chart whose own error is $0.235$. Even the
worst of those is a third of what the chart costs anyway.

Two details in the same rows are worth keeping. The conditioning term is
visible with no arithmetic mismatch at all: the FP64 cell's own error rises
from $1.7\times10^{-15}$ at $k=16$ to $1.5\times10^{-14}$ at $k=32$, the same
$8.9\times$, so the amplification is a property of the chart rather than an
artefact of quantisation. And the half-precision releases report
$\operatorname{rank}B_T=11$ at both $k$ against FP64's $8$ --- not merely
inflated but \emph{above the cap} of $m-1=10$ that
Proposition~\ref{prop:simplex} imposes. Rounding has broken the zero column
sum: unit roundoff sets the own-class entry of $D$ to exactly zero while the
off-class entries survive, so the columns no longer sum to zero and the
rank-$N$ locus is no longer confined to $\mathbf 1^{\perp}\otimes\mathbb R^r$.
Every $N'$ and every certificate line read off a half-precision release
inherits that inflation --- one more reason the certificate route misreads
these releases while the matched recipe route does not.

\paragraph{How $N'$ should be cut, and why the cap is the check.} $N'$ is
read as $\operatorname{rank}B_T$ at some tolerance, and the choice is not
cosmetic: it sets the line $k<r-N'$ and therefore the attacker's budget. Two
tolerances pull opposite ways, and they are answering different questions.
For \emph{building the projector} a tight cut is right, since a loose one
discards a recoverable image. For \emph{counting} $N'$ a tight cut
over-reads, because rounding lifts directions that carry no example above it.
Proposition~\ref{prop:simplex} settles the second, and it is worth stating as
a rule rather than left to judgement:
\[
  N'\;\le\;\min(m-1,\;r,\;N)\quad\text{is a theorem, so any tolerance
  returning }N'>m-1\text{ is provably too tight.}
\]
The cap is thus a self-check on the tolerance, not only a limit on the
channel. It is not academic: on the FP32-trained letters cell ($m=11$, $N=8$,
$r=64$) the $10^{-12}$ read gives $N'=11$, which exceeds $m-1=10$ and is
therefore invalid, while the noise-matched read gives $N'=8$ --- the true
number of private images. The invalid read costs three dimensions of budget,
$k<53$ against the correct $k<56$, by projecting away directions that are
rounding noise. Where the two disagree, the cap says which to believe.

\emph{The three attackers on this one release are the clearest lesson in the
section.} The certificate recovers none of the eight. The recipe route with an
FP64 simulator recovers none, and worse, aliases --- confidently, at a
residual two hundred and forty times below the truth's own. The recipe route
simulating in the release's own arithmetic recovers all eight to three per
cent. The access models differ and the comparison must say so: the certificate
runs from random public-scale starts, while both replay routes are given the
truth perturbed by $10\%$. So the clean contrast is the second against the
third --- same start, same solver, same release, differing only in the
arithmetic of the simulator, and the difference between a confident alias and
a full recovery. The certificate's failure is a separate fact about a
different and weaker attacker. Two attackers failed, their failure looked
exactly like protection, and nothing about the release differed between the
three runs. Any statement of the form
``this defence works'' is a statement about the attacker who was run, and here
the interval between the first attempt and the successful one was a few
hours.

Two supports keep it from resting on a single cell. The per-image basin does
not decay with $k$ once the slack is held: at $r=64$ it runs $13.8\%$,
$28.6\%$, $11.9\%$, $8.3\%$ at $k=8,16,24,32$, and at $r=32$ it runs $10.6\%$,
$15.8\%$, $12.4\%$ at $k=8,12,16$, with every recorded image found and the
$\operatorname{argmin}$ correct in every row. And the head-width cap has an
operational face: twenty private digits on the padded $26$-logit head give
$88.6\%$ of starts on a private digit with $18$ of the $20$ found, while the
\emph{identical} cell on the ten-class head finds nothing at all --- the same
data, the same rank, the same chart, separated only by whether $N'\le m-1$.

The third regime completes the picture. A control batch sitting exactly
\emph{at} its line ($r=16$, $k=8$, all eight recorded, line $8$) reaches the
floor on $34\%$ of starts but lands on a private image on only $4\%$, and its
$\operatorname{argmin}$ is wrong --- spurious solutions present, selection
unreliable, exactly as the kernel count predicts at and above the boundary.
Below the line, at it, and above it are three different regimes for the
attacker, and only the first is safe to act on.

The above-line regime has now been measured at high $k$ rather than inferred,
and the line is sharp$^\dagger$ (job 753886, FP64, $5000$ starts per cell).
One unit above ($k=58$, $N'=7$, line $57$): $14.3\%$ of starts reach an exact
zero, $0.04\%$ land on a recorded image, and the $\operatorname{argmin}$ is a
spurious solution at residual $4\times10^{-31}$ sitting $0.84$ away from any
private image. Three units above ($k=60$): no landings at all, $36.4\%$ at the
floor, $\operatorname{argmin}$ error $1.18$. So above the line the objective
is not merely uninformative --- it is confidently wrong, and the fraction of
starts reaching an exact zero \emph{rises} while the fraction reaching a
private image falls to nothing. A defender reading only ``the attacker's
solver converged'' would draw the opposite conclusion from the truth.

The corresponding half-precision cells cannot be placed relative to their line
at all, and that is itself the finding. The same bfloat16 release reads
$N'=3$ (line $61$) at tolerance $0.08$ and $N'=11$ (line $53$) at $10^{-12}$,
and $11$ breaks the $m-1$ cap --- so whether $k=58$ sits above or below its
line is a choice of tolerance on a quantity we have shown to be inflated by
rounding. What survives without that choice: of $5000$ starts the best is a
near-miss $7.8$--$10.3\%$ from a recorded image, nothing lands within $1\%$,
and the residual at the three strongest truths is $2\times10^{-3}$.
Approximately, and rarely.

\paragraph{The image finishes long before the residual does.} On the
$95.1\%$ backbone every below-line cell stops at residual
$10^{-13}$--$10^{-15}$ after $300$ iterations — fifteen orders above the
reproduction floor, so by the classification of \S\ref{sec:measured} these
are search failures. Yet the median image error already equals the chart's
own representation error to within $10^{-6}$--$10^{-3}$ at
$k=6,10,14,16$$^\dagger$ (job 614344). So on this backbone, at this budget, the median image error stood within
$0.4\%$ of the chart's floor while the residual stood fifteen orders above
its own — which is the attacker-relevant statement: no tight residual was
needed for the best available image here. Three things it is not. Neither
quantity finished: all five cells stopped at the iteration cap, so the image
was never shown to have \emph{stopped} moving either. The gap is not a
function of the residual, so it does not lift off this backbone — pooling
the twenty-four cells on disk that record both, at residual
$\approx2\times10^{-14}$ the gap runs from $4\times10^{-8}$ to
$1.2\times10^{-3}$, five orders at matched residual, with rank correlation
only $+0.43$; the obvious first-order model $\sqrt{\text{residual}}/
\sigma_{\min}$ lifts that only to $+0.53$, so conditioning does not explain
the scatter either. And what those last fifteen orders would do was not
observed at all, since no cell reached the floor — and off-chart there is reason to think they could do harm rather
than nothing. Driving the residual down moves the coordinates toward the
release-optimal point, which for data not lying in the chart is not the
representation-optimal point; continuing could therefore move the image away
from the chart's best. That is the same distinction that organises the whole
section, now on the convergence axis, and it is untested: one cell driven to
the floor off-chart would settle whether an attacker should stop early, and
recording the gap and the residual every twenty-five iterations along a
single cell would settle the plateau question directly, without the
cross-cell scatter. For
locating the boundary the residual remains the right instrument, since it is
what separates an alias from a search failure; for judging what an attacker
obtains it is not, and the image-space gap of this section is.



\paragraph{A free convergence certificate in image space.} The identity above
is not only a result; it is an instrument. On an on-chart cell a converged
inversion \emph{must} land on the chart's own best point, so
$\bigl|\,\text{median image error}-\text{chart representation error}\,\bigr|$
is zero at convergence and grows as the solver falls short — an image-space
diagnostic that costs nothing, needs no floor known in advance, and is
independent of the residual. Measured across twenty-two on-chart rows it
tracks the residual over fourteen orders$^\dagger$: $0$ at residuals
$10^{-31}$, $10^{-13}$ at $5\times10^{-17}$, $10^{-7}$ at $10^{-14}$,
$1.6\times10^{-3}$ at $2.9\times10^{-9}$. Two uses. At the fixed $80$-iteration
budget of the trained-model sweep it gives the shortfall as a curve in $k$ —
$3.8\times10^{-8}$, $7.2\times10^{-5}$, $4.1\times10^{-4}$,
$1.5\times10^{-3}$, $2.4\times10^{-3}$ at $k=14,16,17,18,20$ — for cells that
never reached the floor at all. And it flags the least converged member of a
comparison independently of the residual: among the four charts of
\S\ref{sec:opens} it singles out the ReLU-decoder VAE ($1.6\times10^{-3}$
against $10^{-13}$--$10^{-6}$ for the rest), the same cell whose
solved $\sigma_{\min}$ is three times its value at the truth. Separately,
the VAE charts' conditioning penalty is largely their decoder's own geometry
rather than anything about the encoder: the VAE decoder's Jacobian has
$\sigma_{\min}=0.008$ against $\sigma_{\max}=6.7$ (condition number
${\sim}830$) where a PCA chart's is exactly $1$, of the right order to
account for most of the ${\sim}2000\times$ gap in $\sigma_{\min}(D\rho)$
between them$^\dagger$ (job 626565). That verdict
carries the same caveat as the comparison itself: for a VAE chart the
reference $c_i$ comes from the inner solve, so the flag conflates outer
shortfall with inner slack and is unresolved in the same way.\footnote{An asymmetry to record while using it: for a PCA chart the
projection onto the chart is exact (orthonormal basis, closed form), while
for a VAE chart it is a $300$-step Adam solve with no convergence criterion
(vae\_chart.py:85). Between families this biases conservatively for the
richness ordering — an under-converged inner projection overstates a VAE's
error and so understates its richness. \emph{Within} the VAE family it need
not, because that same inner solve returns $W_{\mathrm{true}}$, which is both
the point the adapter is trained on and the point at which $\sigma_{\min}$ is
evaluated. Measured$^\dagger$ (job 626565), the concern does not bite for the
GELU decoder: at $300$, $1000$ and $3000$ inner iterations the inner loss is
$90.5$ to four digits and $\sigma_{\min}$ at the truth is
$1.13\times10^{-11}$ at every level and at three perturbations of the truth,
so the truth does not move with the inner budget there. Untested for the
ReLU decoder and the $\beta$ arms. The GELU/ReLU pair
therefore compares two separately fitted models differing in encoder,
decoder and inner starting point at once, not one activation swap, and its
$1.04\times$ in richness against $1.15\times$ in $\sigma_{\min}$ is
unresolved on three counts: a single $A_0$ draw, below the replicate spread,
and an unmeasured inner solve in both coordinates.}

boundary is chart-independent: all three charts collapse at $k=18$ exactly,
$\sigma_{\min}$ falling from $10^{-5}$ to $10^{-18}$, because the count is
blind to how the chart is built. What comes back is not: at $k=17$ every
chart recovers its own representable image to $10^{-14}$, and against the
real digit the three differ by thirteen orders. The boundary is therefore a
statement about the chart coordinates, not about the images. Below it the
coordinates $w$ are locally identifiable and the attack returns
$\psi(\hat w)$; how close that is to the private image depends entirely on
whether the chart can represent it. The right reading is a capacity one, and it is chart-\emph{independent}:
the release exposes exactly $m+r-N$ independent coordinates per example, in
any basis whatever. Reconstruction then factorises as
\emph{capacity} $\times$ \emph{chart quality} — this law fixes the first
exactly, and a prior supplies the second. Equivalently, and this is the
honest conditional form: exact reconstruction holds if and only if the
private data lies in an attacker-buildable chart of dimension below
$m+r-N$; the capacity is proved, and building that chart is the open
problem. What the law does not bound is the fraction of the image an
attacker can reach, so it is not a leakage bound: nothing constrains the
attacker's chart to be data-agnostic, and the oracle-built chart — which
spans the private images and so has $k=N=8$, far below the line — returns
them at $10^{-14}$ (last rows); an attacker does not have that chart, and
what the row shows is what a chart that happened to contain the images would
give. What the boundary limits is the search space, and
the search space is the attacker's to choose — which is the counting
argument handing the generative-prior direction of Rev~9 its role: a prior
is a chart-builder (Figure~\ref{fig:chart}).

\paragraph{A richer chart is not harder to invert.} Since the count caps the
chart's dimension and not its quality, the natural worry is that a chart
which draws better is harder to invert — that expressiveness is paid for in
conditioning. It is not, within a family. Varying only the $\beta$ of a VAE
at fixed latent dimension, architecture and training data$^\dagger$ (job
626565, weak encoder): $\beta=0.25$ represents the private digits markedly
better than $\beta=1$ (error $0.222$ against $0.339$) at the same
conditioning ($\sigma_{\min}$ at the truth $1.15\times10^{-11}$ against
$1.13\times10^{-11}$; three $A_0$ draws span $0.99$--$1.28\times10^{-11}$).
Across \emph{families} the differences are large — a VAE chart is
${\sim}2000\times$ worse conditioned than a PCA chart on this encoder — but
that is accounted for by the decoder's own Jacobian (condition number
${\sim}830$ against exactly $1$ for a linear chart), not by what the chart
can draw. An ordering across four charts that appeared monotone in richness
was withdrawn on this evidence: it does not survive a within-family sweep,
and the linear pair inverts on two of the four encoders. The practical
reading is favourable to the attacker and to the defender's
threat model alike: \emph{spend the $k$ coordinates on the best chart
available}, because richness is free in conditioning terms provided the
decoder itself is well conditioned.

A budget rerun qualifies that comparison$^\dagger$ (jobs 624463, 624465, at
$3000$ iterations). Only the PCA charts actually reach the floor --- PCA at
$1.8\times10^{-29}$ in $1808$ iterations, the global chart at $10^{-30}$ in
$1175$, $\beta=16$ at $8\times10^{-31}$ in $616$ but with a useless $0.72$
ceiling. Every learned chart is \emph{still descending} at the cap, so no
fidelity ranking among them is claimed here and the earlier ordering should
not be read as one. What stands is the trade: the learned charts have better
ceilings ($0.22$--$0.34$ against PCA's $0.42$) and worse conditioning
($\sigma_{\min}$ $4\times10^{-12}$ to $6\times10^{-11}$ against
$10^{-8}$ to $2\times10^{-8}$), and do not reach in $3000$ iterations what PCA
reaches in about $1500$.

\subsubsection*{The recipe}

\paragraph{How much of the recipe the release itself gives up.}
Definition~\ref{def:rho} takes $(\eta,s,T)$ and the labels as known. The first
thing to say about that hypothesis is which channel carries it.
\emph{The certificate is recipe-free by construction} --- $C$ is assembled
from $(A_T,B_T)$ alone, with no $\eta$, no $T$, no schedule and no labels ---
and it is the only route here that runs from random starts. So recipe
knowledge is a hypothesis of the \textbf{wide} channel and not of the narrow
one, and the narrow one needs no argument about what a release discloses.

A second distinction is worth drawing precisely, because the two halves are
usually conflated and only one of them is free. A released adapter ships its
\emph{architecture} --- the rank $r$, the scale $\alpha$ (hence $s$), and
which modules were adapted --- in the configuration file that accompanies the
weights. Those are exactly hypotheses (A2) and (A5) and the $s$ of every
recurrence, so that much of the setup is disclosed as a matter of format. The
\emph{recipe} --- optimiser, learning rate, epoch count --- is not: it appears
in a model card when an author chooses to put it there, which is a property of
practice and not of the format. We assume, and do not measure, that default
recipes dominate real fine-tunes; a survey of public adapter configurations
would make that empirical and is noted in \S\ref{sec:opens} as available work.

With that scoping, three measurements$^\dagger$ at $k=12$, $N=8$, $T=400$,
$\eta=0.01$ (job 484255) show that a wrong recipe is detected (R1) and that
$\eta$, the $(\eta,T)$ split and the labels are fitted from the release alone
(R2, R3). A fourth (R4) needs more access and is scoped as such. One
qualification governs how R1 may be read, and it is not a small one: those
probes start from the truth perturbed by $10\%$. The rejection test is
therefore decisive \emph{in the regime where the inversion already succeeds}.
From attacker-buildable starts no replay cell reaches the floor at all, and
there a wrong recipe and a correct recipe from a poor start produce the same
observation --- a high residual. So R1 establishes that the release
distinguishes recipes, not that an attacker lacking a start can use it to find
one.

\emph{(R1) None of seven wrong recipes reached the floor.} With the control first:

The true recipe is the unique floor-reacher, separated from the nearest
wrong hypothesis — one step in four hundred — by twenty-three orders. The
residual is \emph{not} monotone in the size of the recipe error
($\eta\times2$ ends lower than $\eta/2$; $T+25\%$ is eighteen times lower
than $T-25\%$), so it is a detector, not a graded objective; it is computed
from the release and the candidate alone, never from the private images.
Seven hypotheses, one cell, one seed.

\emph{(R2) The learning rate can be fitted jointly with the data.} Carrying
$\eta$ as a free unknown beside $(w,X)$ from starts at $1\times$, $2\times$,
$0.5\times$ the truth, the fit returns $\eta$ to relative error
$5.2\times10^{-16}$ and the images to $3\times10^{-15}$ at the floor in all
three cases — one extra unknown against a slack of $120$ in \eqref{eq:cap}.

\emph{(R3) $\eta T$ is not a degeneracy of discrete SGD, and labels are
identifiable.} Holding $\eta T=4$ and varying the split, $(T,\eta)=(100,0.04)$,
$(200,0.02)$, $(800,0.005)$ stop at residuals $3.1\times10^{-7}$,
$3.5\times10^{-8}$, $8.8\times10^{-9}$ while the true $(400,0.01)$ reaches
$6.4\times10^{-31}$: the product does not determine the release, so $\eta$ and
$T$ are separately identifiable — the degeneracy is a gradient-flow statement
that finite steps break, and this is the weakest separation in the study
(twenty-two orders), as it should be if the degeneracy is approached in the
small-step limit. The wrong splits still return images within
$0.1$--$0.7\%$. Labels: the true assignment reaches the floor
($7.2\times10^{-31}$); swapping two examples ($2.5\times10^{-3}$) or shifting
all of them ($7.5\times10^{-6}$) does not.

\emph{(R5) The batch size is absorbed into the step, and is not
identifiable.} This one is a corollary of \eqref{eq:rec}, and then a
measurement. If the recipe minimises the \emph{mean} rather than the sum, as
implementations normally do, then Lemma~\ref{lem:grad} carries a factor $1/N$
and every recurrence in \eqref{eq:rec} depends on $(\eta,s,N)$ only through
the single scalar $\eta s/N$. Three consequences follow. First, a simulation
handed a subset of $N'$ examples reproduces the release only at step $\eta
N'/N$: at the original step it is a different recipe, and its residual at the
truth will not reach the floor, which is (R1) turned on the analyst. The
subset simulation is legitimate because the omitted examples contribute no
gradient --- that is what invisibility means in \eqref{eq:imprint} --- but the
normalisation must be corrected for the ones that remain. Second, the attacker
never needs $N$: they fit one scalar, which is precisely the fit (R2) measures
succeeding to $5.2\times10^{-16}$. Third, and this is the defender-side
reading, $N$ cannot be recovered from the release at all. Only $N'=\operatorname{rank}B_T$
can be --- and at a layer whose \emph{input is frozen}, which is the setting
throughout this section, that rank is the number of examples recorded. At a
layer whose input drifts because an adapted layer sits below it, the same rank
counts recorded \emph{directions}, over images and steps together, and is no
longer a headcount (\S\ref{sec:opens});
and the two halves of that statement hold to different strengths. That $\eta$
and $N$ are inseparable, and that neither is needed, is exact and
unconditional: it is a statement about the parametrisation of a fixed batch.
That a batch of eight of which two are invisible is indistinguishable from a
batch of six at a proportionally smaller step is exact only if the two
contribute \emph{identically} zero gradient at every step, and invisibility is
a floor rather than a zero, so the two releases agree to the floor and not
beyond. They differ through two channels, not one: the omitted imprints
$\sum_{i\,\mathrm{inv}}C_i$ directly, and the invisible rows of $M_T$, which
enter $A_t$ and so perturb the \emph{recorded} examples' imprints as well ---
the same second channel that keeps the certificate residuals at $10^{-10}$
rather than at machine precision.

Measured$^\dagger$ (job 709507, strong backbone, three batches raw and
chart-projected). Comparing the full-batch release against its recorded
members alone at step $\eta N'/N$, with the omitted imprints' relative scale
in brackets: $1.8\times10^{-15}$ $[1.9\times10^{-15}]$ and
$4.9\times10^{-16}$ $[4.0\times10^{-16}]$ for the repeated batch at $N'=6$;
$1.3\times10^{-15}$ $[1.5\times10^{-15}]$ at $N'=1$ and $7.5\times10^{-17}$
$[3.8\times10^{-17}]$ at $N'=4$ for \texttt{hard1\_diff}; $4.6\times10^{-14}$
$[4.6\times10^{-14}]$ and $6.8\times10^{-16}$ $[6.7\times10^{-16}]$ for the
confident batch at $N'=3$. Every cell sits on its own bracket to within a
factor of two, so in these six the trajectory channel is at or below the
omitted imprints and the two releases agree to machine precision. Two
controls separate this from a tautology (job 710597). The same recorded
subset at the \emph{unscaled} step differs by $2.9\times10^{-2}$ to $1.7$ ---
that is the error itself, cell by cell. A subset built from the complement,
taking all the invisible members and filling from the weakest recorded ones,
differs by $1.0$ in all six cells. And swapping a single member, the weakest
recorded one for the strongest invisible one, differs by $1.1\times10^{-11}$,
$6.2\times10^{-10}$, $1.0$, $5.7\times10^{-8}$, $1.3\times10^{-8}$ and
$1.4\times10^{-6}$ across the six.

That last row is worth reading as more than a control, because it is
\eqref{eq:imprint} measured from a direction the imprint experiments never
took: exchanging one member moves the release by that member's own imprint and
by nothing else, cell by cell, over five orders of magnitude. The $1.0$ is the
$N'=1$ cell, where the single recorded example \emph{is} the entire record, so
swapping it replaces the release rather than perturbing it. The exception is
instructive. Under
decoupled weight decay the step contributes $-\eta\,\mathrm{wd}\,B_t$ as well,
so the release determines the two combinations $\eta s/N$ and
$\eta\,\mathrm{wd}$; a defender whose weight decay is nonzero \emph{and}
published thereby publishes their batch size.

\emph{(R6) Which subset was recorded is read off the level of the residual
floor, not off reaching it.} The natural test --- simulate a candidate subset
and see whether it reaches the floor --- does not discriminate, because a
wrong subset reaches \emph{its own} floor just as exactly. Measured$^\dagger$
(job 706597, three recorded of a repeated batch): the true recorded subset has
a predicted floor of $1.1\times10^{-16}$ and achieves it (over-floor $1.03$),
while a subset with one member swapped has a predicted floor of
$2.1\times10^{-10}$, sits at $2.1\times10^{-10}$ at its own truth, and
achieves $2.1\times10^{-10}$ (over-floor $1.00$). Both are exact; they differ
by six orders in \emph{where} exactness lies, because the floor is set by the
imprints the subset omits. So the selection rule is to minimise the achievable
residual over candidate subsets, and it is available to an attacker, who
observes each candidate's achievable residual without knowing any truth. The
rule has a boundary, and it is not a value of $N'$: it discriminates while the
omitted imprint exceeds the achievable residual. Across three batches the
one-swapped subset's residual exceeds the recorded subset's by $3.9\times$,
$2.9\times10^{3}$ and $1.9\times10^{6}$, and by only $1.6\times$ in the cell
where the swapped-in image's imprint is $4\times10^{-19}$ of the release ---
below the $5\times10^{-17}$ the solver itself reaches, so there is nothing
left to separate. The coarse discrimination never fails: a candidate subset
containing \emph{no} recorded image separates by $10^{5}$ to $2\times10^{16}$
in every cell tested. What degrades is the fine distinction between two
subsets that both contain most of the release. (The batches confound $N'$ with
which image was swapped, so the ordering by $N'$ is not clean; the ordering by
the swapped-in imprint against the achievable residual is.)

\emph{(R4) With continued-training access, a decaying schedule leaks the
step count that a constant one hides.} This result assumes an access model
the others do not: the attacker submits a batch of their own choosing,
$H_p=\varphi(\psi(W_p))$ with labels they choose, and observes the
\emph{victim's} training process take one more step on it with its hidden
$\eta$, weight decay and scheduler state — a checkpoint that includes
optimiser and scheduler state, or a fine-tuning service that continues the
job. An attacker who runs the step themselves sets $\eta$ and learns nothing;
under weights-only release (R4) does not apply, and neither the schedule nor
$T$ is recoverable by it. Within that access model no private data enters:
every regressor is computable from $(A_T,B_T)$, $W_0$ and the probe batch.
One step under decoupled weight decay is $\Delta B=-\eta\nabla_BL-\eta\,\mathrm{wd}\,B_T$, linear in
$(\eta,\eta\,\mathrm{wd})$: the two-parameter least squares returns $\eta$ to
$5\times10^{-16}$ and $\mathrm{wd}=10^{-3},10^{-2}$ to $10^{-11}$ relative
(job 487290), and
$\cos(\Delta B,-\nabla_BL)=1$ under SGD against $0.43$ under Adam separates
the optimiser family (job 485912). The same probe returns the per-step $\eta$ at the continuation steps
to $3\times10^{-14}$ or better (jobs 487290, 488314). Under a constant rate that is all it
returns, and $T$ stays hidden. Under a cosine schedule of known family, a
global fit of $(\text{base},T_{\max},T)$ to six probe steps (eliminate the
base by ratios, grid the two integers, solve the base in closed form) returns
all three exactly — $T=400$, $T_{\max}=1000$, base $0.0100$ — and identically
from windows of $20$, $60$ and $150$ steps, and for releases at $T=200$ and
$T=700$ as well$^\dagger$ (job 591848, a global grid over $(t,T_{\max})$
with the base in closed form, fit residual $\le1.3\times10^{-16}$, applied to
the probed values of job 488314). A schedule encodes position on its own
curve — and with checkpoint access $T$ is usually in the metadata anyway, so
the content of (R4) is the fine-tuning-service case.\footnote{A gradient fit
of the same three parameters had landed $30$--$40\%$ off and was retracted;
it was a local minimum of a multimodal objective, not a degeneracy, and the
global fit was reproduced independently by an auditor before the script was
committed.} The family is assumed known;
selecting it is a model-selection problem not run here.

\paragraph{Why the certificate cannot reconstruct, stated structurally.} The
foregoing reports that reconstruction needs a chart; the reason is visible in
the certificate's defining property and is worth stating, because it is the
same property twice.

The measurements are generic. $A_0$ is Gaussian, so for a feature direction
$h$ orthogonal to the recorded ones, $A_0h$ is \emph{independent} of
$\{A_0h_i\}$ --- for a Gaussian matrix $A_0u$ and $A_0v$ are independent
exactly when $u\perp v$ --- hence independent of $\operatorname{row}(B_T)$, so
$\Pi A_0h$ is a fresh Gaussian in a random subspace. Off the training span the
certificate is a random projection, which is the setting sparse recovery is
built for, and $\ker A_0$ is a uniformly random $(n-r)$-dimensional subspace,
incoherent with any fixed sparsity basis with high probability.

What is not standard is where the private image sits. Its feature lies in
$\operatorname{col}(H)$ --- and the certificate \emph{annihilates}
$\operatorname{col}(H)$, because $Ch_i=0$ \emph{is} the membership property.
So the certificate takes no measurement of the recorded image at all. It
supplies $r-N'$ generic constraints on the ambient space and places the image
in $\ker C$; where in $\ker C$ it sits, the certificate does not say.
Reconstruction is therefore not ``invert the measurements'' but ``find the
sparsest box-feasible point of $\ker C$'', with an effective budget of $r-N'$
and a recoverable sparsity of about $(r-N')/\log n$:
\[
\begin{array}{r|ccccc}
 r \ (N'=8) & 16 & 32 & 64 & 128 & 256\\ \hline
 \text{recoverable pixels of }784 & 1.2 & 4.7 & 13.9 & 40.5 & 149.9
\end{array}
\]
against roughly $150$ non-zero pixels in an MNIST digit. The channel reaches
sparse-recoverability at $r=256$ for $28\times28$ images and, since the
requirement scales with the image's sparsity while $r$ does not, at no
practical rank for anything larger.

\emph{Membership and reconstruction are opposite readings of one annihilation.}
A test that returns exactly zero for every recorded example is for that reason
unable to distinguish among them. The property that makes the certificate an
exact membership instrument is the property that makes it useless as a
measurement of what it detects --- which is why every reconstruction route in
this work has funnelled into a prior, and why a prior is not a convenience
here but the only remaining source of the information the release withholds.

\subsection{Beyond one adapted layer: what a release constrains in pixel space}
\label{sec:pixels}

Everything above concerns one adapted layer with frozen inputs. Real adapters
sit on several layers, and the question a practitioner asks is not about chart
coordinates but about the image: \emph{how many independent constraints does a
released multi-layer adapter place on the raw pixels?} The theorems answer it
with a bound, and the bound turns out to be the binding one.

\paragraph{The bound.} Stack every adapted layer's certificate conditions
against the pixels: $K(x)=[\,C^\ell\,DF_\ell(x)\,]_\ell$, whose rank is the
number of independent constraints. Every adapted layer's map factors through
the first, $DF_\ell=\Phi_\ell DF_1$, so the stack shares that right factor and
\[
  \operatorname{rank}K\;\le\;\min\Bigl(\textstyle\sum_\ell\min\bigl(r_\ell-N'_\ell,\;\operatorname{rank}DF_\ell\bigr),\;\;\operatorname{rank}DF_1\Bigr).
\]
The last term is the one with teeth: however many layers are adapted and
whatever their ranks, the total cannot exceed the Jacobian rank of the frozen
path beneath the first adapted layer. That is a property of the stem, not of
the adapters.

\paragraph{Architecture gives an upper bound; training gives the real one.}
By widths alone the cap is the narrowest point of the stem: a pooled
convolutional stem with adapters on the head caps at $512$ or $2048$ against
$150{,}528$ pixels, while a ViT-B/16 stem has \emph{no} bottleneck at all ---
its patch embedding maps $16\times16\times3=768$ values to a $768$-dimensional
token, a square map, so $196\times768$ is exactly the pixel count. ViT-L/16
expands. So ``adapt late and you are safe'' is false for the family most
people adapt.

Measured on a trained network$^\dagger$, the cap falls far below the
architectural figure and keeps falling with depth: of $784$ input dimensions
the frozen path admits full rank at one and two layers, $220$ at four, and
$96$ at seven. The gap between $784$ and $96$ is not architecture --- the
widths permit $784$ throughout --- it is what training does to the map. A
classifier earns its accuracy by discarding variation, and the discarded
directions are exactly the ones no downstream certificate can constrain. The
encoder's learned invariances, which appear in \S\ref{sec:opens} as the reason
deep conditions are not independent, are the same phenomenon measured as a
rank.

\paragraph{Two consequences, and one withdrawal.} Past the cap, \emph{extra
adapter rank buys nothing}: at the eighth layer the count is $96$ at rank
$256$ and $96$ at rank $900$ alike. And conditioning collapses faster than the
count --- $\sigma_{\min}$ of $10^{-2}$, $7\times10^{-4}$, $10^{-10}$ at one,
two and four frozen layers --- so from four layers down the \emph{usable}
number of conditions sits below the formal one. The defender reading is that
\emph{adapting deeper layers leaks less in pixel space whatever rank is
chosen}, which is a design choice costing no accuracy and independent of the
model's confidence, its arithmetic and its training length. The scope line
must travel with it: this bounds the recipe-free channel, since the replay
route reads more of the release than $\operatorname{row}(B_T)$.

The withdrawal is a defence that looked available and is not. If the recorded
\emph{directions} at a drifting layer accumulated without limit, training
longer would eventually fill the rank and close the deep channel. They do not:
the count plateaus at $90$--$110$ directions and is flat from $10$ steps to
$3200$. Once the adapter's rank exceeds that plateau, no amount of further
training closes the channel, and ``train longer'' is not a lever.

\subsection{What is open}
\label{sec:opens}

\noindent\emph{Available and not on the critical path.} The threat model
assumes that default recipes dominate real fine-tunes. That is an assumption
about practice, not about the format, and it is the one place the recipe
discussion rests on something unmeasured. A survey of public adapter
configurations --- how many disclose optimiser, learning rate and epoch count,
and how concentrated the disclosed values are --- would convert it into an
empirical statement. It is self-contained, needs no compute, and is worth
doing when it is not competing with a measurement.

\begin{enumerate}\itemsep3pt
\item \textbf{Local, not global.} Theorem~\ref{thm:suff} isolates the truth
  in a neighbourhood; injectivity of the release map on the whole manifold
  is open, as it is for the certificate in Rev~9. No alias has appeared below
  the line$^\dagger$; that is evidence. The continuation walks one null
  direction per cell — a multi-direction version, which would say whether the
  extent is direction-dependent, has not been run — and the at-line cell was
  still rising when its budget ended. ``Recognisable'' is measured to $3.8\%$
  and no further. And since the boundary is about the chart, the question
  that matters for privacy is which chart an attacker can build
  \emph{without} the private images — a population prior — and how far below
  the line it lands; that is the generative-prior direction, now with a
  precise specification. Two priors compose here. A chart \emph{local to the
  label}, $\psi_y(w)=\mu_y+V_yw$ or a class-conditional generator, spends
  all $k$ coordinates on within-class variation (the label is given by the
  task or identifiable from the release, R3), and is better conditioned than
  a global chart. And the \emph{encoder} sets a ceiling above the count: two
  images with the same features $\varphi(x)$ are indistinguishable to any
  attack on this adapter, so the chart may as well be built in feature space
  and decoded by a separate feature-inversion prior; detail the encoder
  discards is lost whatever the chart. Cells comparing global PCA, local PCA
  and a conditional VAE at equal $k$ on the trained backbone are queued.
\item \textbf{The recipe family, and the access model.} What
  \S\ref{sec:measured} leaves open is the \emph{family}: the schedule fit
  assumes cosine, and choosing among families is model selection, which the
  floor test of (R1) would presumably rank but which has not been run; and
  (R4) holds only under continued-training access — under weights-only
  release the schedule and $T$ are not recovered by anything here. The count \eqref{eq:cap} bounds the recipe
  as it bounds the data: $p$ unknown recipe scalars are identifiable only if
  $p\le N\bigl((m-1)+r-N\bigr)-Nk$, which is $120$ at the standard cell. A
  parametric schedule ($p=2$--$4$) sits far inside; a free per-step sequence
  ($p=T$) is identifiable only for $T\le120$ and not at $T=400$ — a defence
  that works by counting, not by obscurity. The certificate remains the
  recipe-free channel at the price $k<r-N$; Rev~9 \S3.6--3.8's curvature
  mixture remains the recipe-unknown approximation.
\item \textbf{Adam.} Its entrywise normalisation is not of the form in
  (A3): Theorem~\ref{thm:traj} does not hold, Corollary~\ref{cor:seed} does
  not reduce the seed to $X$, and the unknown seed is all of $A_0$ —
  $Nk+rn$ unknowns against $mr+rn$ equations. Measured$^\dagger$ (jobs
  466915, 467622, 479684): $D\rho$ has $536=Nk+rn$ columns at $n=32$ and
  $1632$ at $n=96$ against SGD's $88$ and $224$, is full column rank at the
  truth in every Adam cell, and is $\sim\!400\times$ worse conditioned at
  $n=96$ ($\kappa=8\times10^{5}$ against $2\times10^{3}$) with a much smaller
  basin. Adam keeps identifiability and loses the $rN$ reduction; and because
  it also breaks Proposition~\ref{prop:simplex}, it hands the attacker the
  $N$ dimensions the softmax head had removed (the prediction above).
\item \textbf{One Jacobian, three readings.} $D\rho$ at the truth is one
  object: its rank is Theorem~\ref{thm:suff}; its spectrum is the local
  resolution ellipsoid $\{h:h^{\!\top}(J^{\!\top}\Sigma^{-1}J)h<\tau\}$ of image
  changes the release cannot distinguish; its whitened form is the
  detectability $d^2=2\,\mathrm{KL}$ of the dataset-sensitivity programme,
  whose valley widths are one-dimensional slices of the ellipsoid
  (Figure~\ref{fig:spectrum}). Rendering its leading directions as image
  edits is the natural next object.
\item \textbf{Encoder quality: measured, and it ends the channel before the
  count does.} This was an open item; it is now the ladder in
  \S\ref{sec:measured}. The conditioning cost grows monotonically with
  accuracy, and at $97.9\%$ hypothesis (A4) fails outright, so the capacity
  law has nothing to say about that regime — not because the count is wrong
  but because the adapter stops recording the easy examples at all. What is
  open now is the boundary between the two: at what accuracy, or what margin
  distribution, does $\operatorname{rank}P_T$ begin to fall? That a
  deliberately high-margin fine-tuning set suppresses the release is now
  measured (\S\ref{sec:measured}, the confident batch); what is not is how
  the suppression degrades as the batch is diluted with harder examples. The
  measurement that settles it is the same example's imprint in a confident
  batch and then beside a badly fit one, which no cell yet contains; the
  question is about \eqref{eq:imprint}'s value through the shared trajectory,
  not about the feature coupling that was withdrawn.
\item \textbf{LoRA on more than one layer.} Adapting the first layer moves
  its own successors' inputs, so (A1) fails downstream: the closure of
  Theorem~\ref{thm:traj} does not hold, several adapters are released, and
  neither the count nor identifiability is derived. Measured as an empirical
  cell$^\dagger$ (job 626564, all three layers): with the seeds \emph{known}
  as an oracle ($112$ unknowns) the solver reaches residual
  $2.2\times10^{-8}$; with them unknown ($22{,}384$ unknowns),
  $7.7\times10^{-7}$ — both twenty orders off the floor, and the oracle arm
  nearly as bad as the attack. The obstruction at this budget is the
  three-layer unroll itself, not the unknown seeds, so the cell supports no
  identifiability statement in either direction; the fact that more numbers
  are released than are unknown ($38{,}352$ against $22{,}384$) is not
  evidence either way: a count of released entries is not a count of
  independent constraints, which is the whole content of \eqref{eq:cap}. It stays outside
  the theorems.
\item \textbf{What stops the attack today is neither line.} Inside the
  capacity region, LM from perturbed-truth starts recovers up to $86\%$ start
  error, with one budget-limited failure at $81\%$ (residual $8\times10^{-5}$
  at the iteration cap, still descending) bracketed by successes at $77\%$
  and $86\%$\textsuperscript{$\dagger$} (job 459111). From the $20$ starts an attacker can build
  from the release alone (random, span, certificate, span-anchor; jobs
  408560--63) it reaches the floor in \emph{none}: one run ends inside the
  $10^{-2}$ image tolerance at residual $9\times10^{-7}$ with its restart
  budget exhausted — by the classification above a search failure that
  happened to land near the truth — and the other nineteen end at residuals
  $5\times10^{-3}$ to $0.4$. The binding constraint is the initialiser — the
  slot the framework reserves for a learned or population prior — not the
  information in the release.
\end{enumerate}

\subsection{What to say to Gal (replaces the lead of Rev~9 \S16)}
\label{sec:gal}

Lead with two theorems, each with a sharp measurement. The first says how
many numbers per private example a released adapter can hold; the second
says which examples are in it at all. Together they make the adapter a
\emph{measurable channel}: two computable quantities, one of which the
defender can evaluate during their own training without simulating any
attack. The reconstructions are consequences, and the $49/49$ grid should
not lead — it sits inside the capacity region and measures existence, not
rate.

\begin{enumerate}\itemsep3pt
\item \emph{The chain the theorems stand on.} Theorem~\ref{thm:traj} writes
  the adapter's whole trajectory in closed form; Corollary~\ref{cor:seed}
  shows the random initialisation enters it through only $rN$ numbers, the
  rest being released verbatim. Projecting those away is Rev~9's
  certificate, $r-N$ equations per image (Theorem~\ref{thm:quot}); solving
  for them is the exact channel, and the difference between the two is one
  identifiable object (Corollary~\ref{cor:two}).

\item \textbf{Theorem I — capacity.} The released $B_T$ lies on a manifold
  of dimension $N((m-1)+r-N)$, the $-1$ being the softmax head
  (Proposition~\ref{prop:simplex}), while $A_TU_c$ pays exactly for the
  seed. Hence $k<m+r-N$ coordinates per example (Theorem~\ref{thm:cap}), and
  full column rank almost everywhere below it (Theorem~\ref{thm:suff}, with
  the closed-form $T=1$ witness). Measured sharp to one unit of $k$ at
  $N=4,8,14$, at five head widths, at three ranks on real MNIST digits, and
  on a \emph{trained} network with unseen test digits — where the line holds
  unmoved while the conditioning needed to reach it worsens a thousandfold.
  Both terms of the cap are attained by the measured block ranks.

\item \textbf{Theorem II — imprint.} The gradient in $B$ is a sum of
  rank-one terms, one per private example, so
  $B_T=\sum_iC_i$ with $\|C_i\|$ bounded by \emph{that example's own}
  accumulated error and no cross term (\eqref{eq:imprint}). Measured on
  $320$ example-batch pairs across four encoders: one proportional line over
  forty orders of magnitude, $971$ of $1120$ concordant pairs,
  and $\operatorname{rank}B_T$ equal to the number of examples recorded above
  the floor in all forty batches (Figure~\ref{fig:imprint}) --- all at a
  layer whose input is frozen, which is what makes that rank a headcount.

\item \emph{What the two theorems give the defender — this is the tool.}
  What leaks is what the model had to learn. A private batch its base model
  already fits leaves every imprint below $1.5\times10^{-24}$: the release is
  numerically empty. Conditioning worsens ten orders from random weights to
  $97.9\%$ accuracy at fixed architecture, and at $97.9\%$ hypothesis (A4)
  fails outright — $\operatorname{rank}P_T<N$ — so the channel closes
  \emph{below its own capacity line}, for reasons of the model rather than
  the count. The exposure meter is therefore the per-example accumulated
  residual during the defender's own fine-tuning: already computed in the
  course of training, and requiring no attack to interpret. The limit of that
  defence is measured, and it is the sentence to say next: on one encoder, one
  eleven-row head and one recipe, eight ordinary digits give
  $\operatorname{rank}B_T=6$ of $8$ with imprints spanning twenty-two orders,
  while eight examples of a category the model has never had give rank $8$
  with every imprint of order one. Model quality and category novelty are
  independent axes, so the suppression is real but absent exactly where
  fine-tuning adds a capability the base model lacks --- the canonical reason
  to fine-tune.

\item \emph{What they give the attacker.} $\operatorname{rank}B_T$ is read
  off the release with no private knowledge and says how many examples were
  recorded. On the $97.9\%$ model one imprint dominates the spectrum in $9$ of
  $10$
  ordinary batches. Private data from a different distribution is recorded
  in full — scanned digits misclassified $4/8$ by that model give every
  imprint order one and full rank — foreign data tends to be
  recorded in full because a model tends to get it wrong.

\item \emph{Why a prior is the missing piece, precisely.} The theorems pin
  the chart \emph{coordinates}; the attack returns $\psi(\hat w)$. At one
  fixed $k$, three charts each recover their own representable image to
  $10^{-14}$ and differ by thirteen orders against the real digit. So
  reconstruction factorises as capacity $\times$ chart quality: this work
  fixes the first exactly, and a prior supplies the second. Richness is free in
  conditioning terms \emph{within a family at matched $\sigma_{\min}$}, which
  is where that was measured --- but not across families or across $k$: the
  learned charts have better ceilings and worse conditioning ($\sigma_{\min}$
  $10^{-12}$--$10^{-11}$ against PCA's $10^{-8}$), and conditioning worsens as
  $k$ grows. Since ill-conditioning amplifies any arithmetic mismatch into a
  displacement, under a quantised release the best $k$ is bounded by
  conditioning as well as by the line. Spend the coordinates on the best chart
  available in exact arithmetic, and stop short of that under a low-precision
  one.

\item \textbf{The attack, and the trade it makes.} There are two routes and
  they buy opposite things. The \emph{recipe route} inverts the full residual
  and has the wide line, $k<m+r-N$, twenty-two coordinates per image at these
  sizes --- but it needs the recipe, and at $N=8$ none of $20$
  attacker-buildable starts reaches its residual floor, while from perturbed
  truth the solver recovers from $86\%$ away, so the obstruction there is the
  initialiser and not the information. The \emph{certificate route} needs
  nothing but the release, the public model and a chart --- no recipe, no
  labels, no $N$ --- and pays for it with the narrow line, $k<r-N'$. Below
  that line it works from nothing, and at rank $64$ it works well: on the
  new-class cell above, from $500$ random starts with no recipe, no labels and
  no $N$, $38.4\%$ land on a private example and all eight are recovered ---
  and still all eight with the adapter \emph{trained} in FP32. The chart
  dimension that rank affords ($k=32$) is one at which a projection identifies
  its own source among $10^4$ candidates $94\%$ of the time. The rank is a leakage
  dial twice over: it raises the admissible $k$, and at fixed $k$ it enlarges
  the basin --- $16.6\%\to74.4\%\to96.4\%$ as the slack goes from one unit to
  forty-nine. Both are raised for utility. And the head width caps it
  independently at $N'\le m-1$: twenty digits recover on a $26$-logit head and
  none at all on a $10$-logit one, same data, same rank, same chart. Two scopings decide how this should be said.

  \emph{Lead with the new class, because it is both the strongest cell and
  the canonical use.} Adding a class the base model does not have --- the
  ordinary reason to fine-tune --- is the case where nothing protects. Eight
  unseen-category examples on a $98\%$ model, $r=64$, a $32$-dimensional
  public chart: the model is wrong about every one at the first step, learns
  the class by the last, the adapter moves $9.3\%$, and the release has norm
  $1.00$ with all eight imprints between $0.12$ and $0.26$. From $500$ random
  starts --- no recipe, no labels, no $N$, no start near the answer ---
  $38.4\%$ land on a private letter and \emph{all eight are recovered}, the
  best at $3\times10^{-26}$. This is not a vanishing release, not a frozen
  adapter, and not a coarse chart: at that dimension a projection identifies
  its own source rather than its class. And it is not an artefact of exact
  arithmetic: with the adapter \emph{trained} in FP32, all eight are still
  recovered, from $32.8\%$ of random starts, at residuals of $3\times10^{-7}$
  to $10^{-5}$. The one-line version: \emph{a released adapter, fine-tuned in
  ordinary arithmetic on a class the model did not have, hands back every
  private example it was trained on to an attacker who knows only the public
  model.}

  \emph{The confident cell is the extreme}, and it says something sharper at
  the cost of being a corner. There $\|B_T\|=7.6\times10^{-18}$ and the
  adapter never feeds back at all --- $A_T=A_0$, and $B_T$ is $T$ times one
  gradient step --- so the fine-tune changed nothing about the model and
  recorded every example anyway. Because the certificate is an angle and not
  a magnitude, a release of that size is not too small to read: FP32 recovers
  five of the eight, and FP16 none at all, since the whole file underflows.
  But bfloat16 is not the clean case the angle argument suggests --- it breaks
  the softmax's zero column sum and reports a rank \emph{above} the $m-1$ cap,
  corrupting the very directions the certificate reads, and it keeps only one
  of three on the comparable $r=16$ cell. Exponent range is necessary and not
  sufficient. This also settles the tension the earlier draft left open.
  Confidence \emph{does} reduce what is recorded --- by twenty-three orders of
  $\|B_T\|$ as the chart sharpens --- and it does \emph{not} protect, because
  a release too small to have moved the model is still large enough to
  identify every image in it.

\item \emph{Open, and said in the same breath.} Global uniqueness below the
  line; how far the release-consistent set extends past it (one direction of
  forty was walked, and one cell was still rising); the accuracy at which
  (A4) begins to fail; LoRA on more than one layer, where the closure fails
  and the measured cells are dominated by the unrolled optimisation rather
  than by identifiability; Adam, which keeps identifiability, loses the seed
  reduction, destroys the certificate and is predicted to sit one unit higher
  on the line; and the schedule family under a weights-only release.
\end{enumerate}
