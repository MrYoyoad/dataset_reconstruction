\section{The exact channel: from the quotient to the full release}
\label{sec:exact-channel}

A released LoRA adapter is a deterministic function of the private data, the
public recipe and the random initialisation. Throughout, $k$ is the
dimension of the chart the attacker searches in — the coordinates $w$ of
Definition~\ref{def:setup} — not an intrinsic dimension of the images; every
count below is a count of those coordinates. This section writes that
function out (Theorem~\ref{thm:traj}), shows that the initialisation enters
it only through $rN$ numbers (Corollary~\ref{cor:seed}), and then reads the
same function in two ways: projecting the initialisation away gives Rev~9's
certificate with $r-N$ equations per image (Theorem~\ref{thm:quot}); keeping
it and solving for the $rN$ numbers gives $m-1+r-N$ equations per image
(Theorems~\ref{thm:cap}--\ref{thm:suff}). The difference between the two is
one identifiable object — the coefficient matrix of the released $B_T$ —
and Corollary~\ref{cor:two} states it. Section~\ref{sec:measured} reports
what was measured, on the synthetic test-bed and on real digits, and shows
that the boundary is a statement about the coordinates the attacker searches
in, not about the images.

%----------------------------------------------------------------------------
\subsection{Objects, hypotheses, and the gradient}
\label{sec:objects}

\begin{definition}[the layer, the data, the adapter]\label{def:setup}
Fix integers $m,n,r,N$ with $N\le r\le n$. A frozen weight
$W_0\in\mathbb R^{m\times n}$ receives the representations
$H=[h_1,\dots,h_N]\in\mathbb R^{n\times N}$, $\operatorname{rank}H=N$, of the $N$ private examples;
$E\in\{0,1\}^{m\times N}$ holds their one-hot labels. A LoRA adapter is a
pair $A\in\mathbb R^{r\times n}$, $B\in\mathbb R^{m\times r}$ with a fixed
scale $s>0$; the layer computes logits and loss
\[
  Z(A,B):=(W_0+sBA)H\in\mathbb R^{m\times N},\qquad
  L(A,B):=\sum_{i=1}^N \ell\bigl(z_i,e_i\bigr),
\]
$\ell$ the softmax cross-entropy. Write $D(Z):=\operatorname{softmax}(Z)-E$
(column-wise softmax), $G:=H^{\!\top}H$, and for any $A_0$,
$Q:=(A_0H)^{\!\top}(A_0H)$; both Grams are $N\times N$.
\end{definition}

\begin{lemma}[gradients]\label{lem:grad}
$\nabla_B L=s\,D\,H^{\!\top}A^{\!\top}$ and $\nabla_A L=s\,B^{\!\top}D\,H^{\!\top}$,
with $D=D(Z(A,B))$.
\end{lemma}
\begin{proof}
$dL=\langle D,dZ\rangle=\langle D,\;s\,dB\,AH+s\,B\,dA\,H\rangle
 =\langle s\,D H^{\!\top}A^{\!\top},dB\rangle+\langle s\,B^{\!\top}DH^{\!\top},dA\rangle$.
\end{proof}

\paragraph{Hypotheses.} Each is a formula; each theorem lists the ones it uses.
\begin{itemize}\itemsep2pt
\item[(A1)] \emph{Fixed inputs.} The same $H$ enters the adapter at every
  step: no augmentation, no dropout, no trainable block upstream.
\item[(A2)] \emph{Initialisation.} $B_0=0$; $A_0$ is drawn from a law with a
  density on $\mathbb R^{r\times n}$ (so, with $r\le n$,
  $\operatorname{rank}A_0=r$ a.s.).
\item[(A3)] \emph{Recipe.} Plain gradient descent with step $\eta$ for $T$
  steps,
  \[
    B_{t+1}=B_t-\eta\,\nabla_BL(A_t,B_t),\qquad
    A_{t+1}=A_t-\eta\,\nabla_AL(A_t,B_t),\qquad t=0,\dots,T-1,
  \]
  and the release is $(A_T,B_T)$.\footnote{Everything below survives any
  update that is a scalar linear filter of past gradients and iterates
  (momentum, scalar weight decay — which multiplies $A_0$ by a scalar $c_t$
  in Theorem~\ref{thm:traj}, and nothing else changes) and any matrix map of
  the form $M\mapsto p(MM^{\!\top})M=M\,p(M^{\!\top}M)$ applied to the
  gradient. Adam's entrywise normalisation is not of this form; \S\ref{sec:opens}, item~3.}
\item[(A4)] $\operatorname{rank}P_T=N$ for the matrix $P_T$ of
  Theorem~\ref{thm:traj}. (Needs $m\ge N$ and $N$ linearly independent
  accumulated error trajectories; it fails for duplicated or already-fit
  examples.)
\item[(A5)] $r\ge N+1$.
\item[(A6)] \emph{Canonical basis.} For any full-column-rank $H$, $U(H)$
  denotes the $Q$-factor of the QR decomposition $H=UR_H$ with
  $\operatorname{diag}R_H>0$. This makes $H\mapsto U(H)$ real-analytic; a
  QR without the sign convention is discontinuous at every sign change of a
  diagonal entry.
\end{itemize}

\paragraph{The two coordinates of the seed.} Let $U=U(H)$ and let
$V\in\mathbb R^{n\times(n-N)}$ be an orthonormal basis of $\operatorname{col}(H)^\perp$.
Define
\begin{equation}\label{eq:XY}
\begin{aligned}
  X&:=A_0U\in\mathbb R^{r\times N},\qquad Y:=A_0V\in\mathbb R^{r\times(n-N)},\\
  \text{so that}\qquad A_0&=XU^{\!\top}+YV^{\!\top},\qquad A_0H=XR_H,\qquad
  Q=R_H^{\!\top}X^{\!\top}XR_H .
\end{aligned}
\end{equation}

\paragraph{The private data as a manifold.} Each private image is
$x_i=\psi(w_i)$ for a generator $\psi:\mathbb R^k\to\mathbb R^{d}$, and its
representation is $h_i=\varphi(x_i)$; write $H(w)$, $R_H(w)$, $U_c(w):=U(H(w))$
for a candidate $w=(w_1,\dots,w_N)\in\mathbb R^{Nk}$. Both $\varphi$ and
$\psi$ are real-analytic ($\tanh$ networks in the test-bed).

\subsection{The trajectory, written out}
\label{sec:trajectory}

\begin{theorem}[closure of the factor dynamics]\label{thm:traj}
Assume (A1)--(A3). Then for every $t\le T$
\begin{equation}\label{eq:closure}
  B_t=P_t\,(A_0H)^{\!\top},\qquad A_t=A_0\bigl(I_n+H\,M_t\,H^{\!\top}\bigr),
\end{equation}
where $P_t\in\mathbb R^{m\times N}$, $M_t\in\mathbb R^{N\times N}$ are given
by $P_0=0$, $M_0=0$ and
\begin{equation}\label{eq:rec}
  D_t=D\bigl(W_0H+s\,P_tQ(I_N+M_tG)\bigr),\qquad
  P_{t+1}=P_t-\eta s\,D_t\,(I_N+GM_t^{\!\top}),\qquad
  M_{t+1}=M_t-\eta s\,P_t^{\!\top}D_t .
\end{equation}
\end{theorem}
\begin{proof}
Induction on $t$; $t=0$ is (A2). Assume \eqref{eq:closure} at $t$. Then
$B_tA_tH=P_t(A_0H)^{\!\top}A_0(H+HM_tG)=P_t\,Q\,(I+M_tG)$, which gives $D_t$.
By Lemma~\ref{lem:grad},
\begin{align*}
  \nabla_BL&=sD_tH^{\!\top}A_t^{\!\top}=sD_t\bigl(H^{\!\top}+G^{\!\top}M_t^{\!\top}H^{\!\top}\bigr)A_0^{\!\top}
  =sD_t(I+GM_t^{\!\top})(A_0H)^{\!\top}\qquad(G^{\!\top}=G),\\
  \nabla_AL&=sB_t^{\!\top}D_tH^{\!\top}=A_0H\,(sP_t^{\!\top}D_t)\,H^{\!\top},
\end{align*}
so the update of (A3) keeps both shapes and produces \eqref{eq:rec}.
\end{proof}

\begin{corollary}[where the seed enters]\label{cor:seed}
Assume (A1)--(A3). The coefficient trajectory $(P_t,M_t,D_t)_{t\le T}$ is a
function of $W_0H$, $G$, $Q$, $E$ and $(\eta,s,T)$ only; the initialisation
$A_0$ enters it through $Q=R_H^{\!\top}X^{\!\top}XR_H$ and nowhere else. The release is
\begin{equation}\label{eq:release}
  B_T=P_T\,R_H^{\!\top}X^{\!\top},\qquad
  A_T=X\,\Omega\,U^{\!\top}+Y\,V^{\!\top},\qquad
  \Omega:=I_N+R_HM_TR_H^{\!\top},
\end{equation}
so $A_TU=X\Omega$ and $A_TV=Y$: the block $Y$ of the seed is released
unchanged, and $B_T$, $A_TU$ depend on the seed only through $X$. In
particular, for every orthogonal $O\in\mathrm O(r)$ the seed $OA_0$ produces
the same $(P_t,M_t,D_t)_{t\le T}$ and $\Omega$, and the release
$(OA_T,\;B_TO^{\!\top})$: the coefficient trajectory is invariant under the
LoRA gauge $BA=(BO^{\!\top})(OA)$, and the release transforms covariantly.
\end{corollary}
\begin{proof}
Substitute \eqref{eq:XY} into \eqref{eq:closure}--\eqref{eq:rec}:
$A_T=A_0+A_0HM_TH^{\!\top}=XU^{\!\top}+YV^{\!\top}+XR_HM_TR_H^{\!\top}U^{\!\top}$.
For the gauge, $Q$ is unchanged by $X\mapsto OX$, so \eqref{eq:rec} is
unchanged, and \eqref{eq:release} transforms as stated.
\end{proof}
The gauge statement was tested rather than assumed (job 487882 — a log,
not a jsonl row; the third shape ran at $\eta=0.03$):
at three shapes — $(n,k,N,T)$ equal to $(96,12,8,400)$, to $(32,6,4,200)$
and to $(96,10,12,150)$ — the relative discrepancies
\[
  \frac{\|A_T'-OA_T\|}{\|A_T\|},\qquad \frac{\|B_T'-B_TO^{\!\top}\|}{\|B_T\|},\qquad
  \frac{\|\Omega'-\Omega\|}{\|\Omega\|}
\]
are all between $5.8\times10^{-16}$ and $2.1\times10^{-15}$. The coefficients see only $Q$; the release sees $X$; the
gauge moves both, so nothing is lost by the attacker and the invariance is a
statement about the trajectory, not about identifiability — the release
still determines $X$, and $\rho$ is not degenerate.

\subsection{Two readings of the same release}
\label{sec:read}

\subsubsection*{First reading: project the seed away}

\begin{theorem}[the quotient certificate]\label{thm:quot}
Assume (A1)--(A5). Then $\operatorname{row}(B_T)=\operatorname{col}(A_0H)$
as subspaces of $\mathbb R^r$, and with $\Pi_S$ the orthogonal projector of
$\mathbb R^r$ onto $S^\perp$,
\begin{equation}\label{eq:C}
\begin{aligned}
  C&:=\Pi_{\operatorname{row}(B_T)}A_T=\Pi_{\operatorname{col}(A_0H)}A_0,\\
  CH&=0,\qquad \operatorname{rank}C=r-N,\qquad
  \ker C=\operatorname{col}(H)+\ker A_0 .
\end{aligned}
\end{equation}
\end{theorem}
\begin{proof}
By \eqref{eq:closure} the rows of $B_T$ are the $m$ combinations of the $N$
columns of $A_0H$ with coefficient rows $P_T$; under (A4) they span
$\operatorname{col}(A_0H)$. Projecting \eqref{eq:closure} for $A_T$:
$\Pi_{\operatorname{col}(A_0H)}A_T=\Pi_{\operatorname{col}(A_0H)}A_0+\Pi_{\operatorname{col}(A_0H)}(A_0H)\,M_TH^{\!\top}$,
and the second term is zero. Hence $CH=\Pi_{\operatorname{col}(A_0H)}(A_0H)=0$;
the image of $C$ is $\Pi_{\operatorname{col}(A_0H)}(\operatorname{col}A_0)=\operatorname{col}(A_0H)^\perp$
of dimension $r-N$ by (A2), (A5); and $Cx=0$ iff $A_0x\in\operatorname{col}(A_0H)$
iff $x\in\operatorname{col}(H)+\ker A_0$.
\end{proof}

What the first reading uses is visible in \eqref{eq:C}: of $B_T$ only the
subspace $\operatorname{row}(B_T)$ — a point of the Grassmannian
$\operatorname{Gr}(N,r)$, $N(r-N)$ numbers — and of $A_T$ only its projection
off that subspace. The coefficient matrix $P_T$ is removed by the projection.
The equations it imposes on candidate data are $CH(w)=0$: $C$ has rank $r-N$,
so these are $(r-N)N$ scalar equations, $r-N$ per image; Rev~9 Thm~3 and
Prop~4 show they pin $w$ locally iff $k<r-N$. No recipe and no solver are
used.

\subsubsection*{Second reading: keep the seed and solve for it}

\begin{definition}[the attacker's map]\label{def:rho}
The attacker knows $W_0$, $\varphi$, $\psi$, $E$, the recipe $(\eta,s,T)$
and the release $(A_T,B_T)$; the unknowns are $w\in\mathbb R^{Nk}$ and, by
Corollary~\ref{cor:seed}, $X\in\mathbb R^{r\times N}$.\footnote{$Y$ is not an
unknown: it enters neither $B_T$ nor $A_TU$, and \eqref{eq:release} shows it
passes to the release unchanged — in the true basis $V$, which the attacker
does not know and does not need.}
For a candidate $(w,X)$ let $P_T(w,X)$, $M_T(w,X)$ be the output of
\eqref{eq:rec} run with $H=H(w)$, $G=G(w)$, $Q=R_H(w)^{\!\top}X^{\!\top}XR_H(w)$,
and $\Omega(w,X)=I+R_H(w)M_TR_H(w)^{\!\top}$. The residual map is
\begin{equation}\label{eq:rho}
  \rho(w,X):=\Bigl(\;\frac{P_T(w,X)\,R_H(w)^{\!\top}X^{\!\top}-B_T}{\|B_T\|}\;,\;\;
  \frac{X\,\Omega(w,X)-A_T\,U_c(w)}{\|A_T\|}\;\Bigr)
  \in\mathbb R^{m\times r}\times\mathbb R^{r\times N}.
\end{equation}
The attack is $\min_{w,X}\|\rho(w,X)\|^2$. The true $(w^\star,X^\star)$ is a
zero of $\rho$; the second reading asks whether it is an isolated zero.
\end{definition}

The target $A_TU_c(w)$ in the second block moves with the candidate, because
the attacker does not know $U$ and must form $U_c(w)$; its $w$-derivative
$A_T\,\partial U_c/\partial w$ is part of $D\rho$.\footnote{Writing the target
as the fixed $A_TU$ instead drops this term and undercounts the
$w$-dependence; the budget one then obtains is refuted by the measured full
column rank at $k=24>m$ (Example~\ref{ex:T1}). The normalisers $\|B_T\|$,
$\|A_T\|$ are fixed numbers — deliberately not $\|A_TU_c(w)\|$, which the
optimiser could shrink — and scale the spectrum of $D\rho$, not its rank.}

\begin{proposition}[the softmax head constrains $B_T$]\label{prop:simplex}
Assume (A2), (A3). Then $\mathbf 1^{\!\top}P_t=0$ and hence
$\mathbf 1^{\!\top}B_t=0$ for every $t$.
\end{proposition}
\begin{proof}
Every column of $\operatorname{softmax}(Z)$ and of $E$ sums to one, so
$\mathbf 1^{\!\top}D(Z)=0$. In \eqref{eq:rec},
$\mathbf 1^{\!\top}P_{t+1}=\mathbf 1^{\!\top}P_t-\eta s\,(\mathbf 1^{\!\top}D_t)(I+GM_t^{\!\top})=\mathbf 1^{\!\top}P_t$,
and $P_0=0$. Then $\mathbf 1^{\!\top}B_t=\mathbf 1^{\!\top}P_t(A_0H)^{\!\top}=0$.
\end{proof}

\begin{theorem}[capacity: the count is necessary]\label{thm:cap}
Assume (A1)--(A3), (A6) and $N\le m-1$. Let
\[
  \mathcal B:=\{B\in\mathbb R^{m\times r}:\ \operatorname{rank}B=N,\ \mathbf 1^{\!\top}B=0\},
  \qquad \dim\mathcal B=N(r-N)+N(m-1)=N\bigl((m-1)+r-N\bigr).
\]
Then, on each connected component of $\{(w,X):\operatorname{rank}H(w)=N\}$
that contains a point with $\operatorname{rank}P_T(w,X)=N$, the maximal rank
of $D\rho$ is at most $N\bigl((m-1)+r-N\bigr)+rN$. Consequently, if
\begin{equation}\label{eq:cap}
  k\;\ge\;m+r-N\qquad\bigl(\text{equivalently } Nk+rN>\dim\mathcal B+rN\bigr),
\end{equation}
then for almost every true $(w^\star,X^\star)$ the set of candidates that
reproduce the release exactly, $\rho^{-1}(0)$, contains a submanifold through
$(w^\star,X^\star)$ of dimension at least $N\bigl(k-(m-1+r-N)\bigr)\ge N$.
\end{theorem}
\begin{proof}
\emph{Dimension of $\mathcal B$.} Let $\Theta\in\mathbb R^{m\times(m-1)}$ be
an orthonormal basis of $\mathbf 1^\perp$; $B\mapsto\Theta^{\!\top}B$ is a
rank-preserving isometry of $\{\mathbf 1^{\!\top}B=0\}$ onto
$\mathbb R^{(m-1)\times r}$. A rank-$N$ matrix there is determined by its row
space $S\in\operatorname{Gr}(N,r)$ and its coefficient matrix in a basis of
$S$, an element of $\mathbb R^{(m-1)\times N}$ of full rank; the dimensions
are $N(r-N)$ and $N(m-1)$.

\emph{Rank bound.} On the open set $\mathcal O$ where
$\operatorname{rank}P_T(w,X)=N$, Theorem~\ref{thm:traj} and
Proposition~\ref{prop:simplex} put the first block of $\rho$, plus the
constant $B_T/\|B_T\|$, in $\mathcal B/\|B_T\|$, a manifold of dimension
$\dim\mathcal B$, so the derivative of that block has rank at most
$\dim\mathcal B$ there; the second block has $rN$ rows; so
$\operatorname{rank}D\rho\le\dim\mathcal B+rN$ on $\mathcal O$. On a
connected component the maximal rank of $D\rho$ is attained on an open dense
set (the $\rho$ of \S\ref{sec:objects} is real-analytic there), which meets
the non-empty open set $\mathcal O$; hence the maximal rank on that component
is at most $\dim\mathcal B+rN$. (Under (A4) the truth lies in $\mathcal O$.)

\emph{Fibre.} On the component, the set where $\operatorname{rank}D\rho$ is
below its maximum is a proper analytic subvariety — closed, with empty
interior and Lebesgue measure zero. Off it the rank is locally constant and
the constant-rank theorem makes $\rho^{-1}(\rho(p))$ a submanifold through
$p$ of dimension $(Nk+rN)-\operatorname{rank}D\rho(p)\ge Nk+rN-\dim\mathcal
B-rN$. This holds at Lebesgue-almost every $(w^\star,X^\star)$; for the seed
that is a.e.\ under (A2), and for the data it is a.e.\ in the chart
coordinates, and $\rho(w^\star,X^\star)=0$ at the truth.
\end{proof}

\begin{remark}\label{rem:distance}
Theorem~\ref{thm:cap} says that past the line the truth is not an isolated
zero of $\rho$: a continuum of release-consistent candidates passes through
it. It says nothing about how far that continuum extends in image space:
non-isolation is not distance. And below the line, identifiability is not
fidelity: the coordinates $w$ are pinned, and the image returned, $\psi(\hat
w)$, is only as good as the chart $\psi$. Section~\ref{sec:measured} measures
both.
\footnote{Rank deficiency of $D\rho$ \emph{at} the truth alone would not give
non-isolation ($f(x)=x^2$ has $f'(0)=0$ and an isolated zero); the proof uses
that the rank bound holds at every candidate, and concludes only almost
everywhere.}
\end{remark}

\begin{theorem}[generic local identifiability below the line]\label{thm:suff}
Assume (A1)--(A3), (A6) and $k\le(m-1)+r-N$. Let
$\mathcal D:=\{(w,X):\operatorname{rank}H(w)=N\}$, an open set, and let
$\mathcal D_0$ be any of its connected components. Exactly one of the
following holds: $D\rho$ has full column rank $Nk+rN$ at almost every point
of $\mathcal D_0$, or at no point of $\mathcal D_0$. In the first case every
true $(w^\star,X^\star)\in\mathcal D_0$ outside a null set is an isolated
zero of $\rho$; so one point of full column rank in $\mathcal D_0$ gives
isolation for almost every seed (under (A2)) and Lebesgue-almost every chart
coordinate in $\mathcal D_0$.
\end{theorem}
\begin{proof}
The $(Nk+rN)$-minors of $D\rho$ are real-analytic on the connected
real-analytic manifold $\mathcal D_0$; their common zero set is an analytic
subvariety of $\mathcal D_0$, hence either all of $\mathcal D_0$ or of
measure zero (identity theorem). At a point of full column rank the
inverse-function theorem makes the zero isolated (Rev~9 Prop~8).
\end{proof}

The statement is per component because a real-analytic subvariety of
codimension one can disconnect: $\mathcal D$ is connected — and one witness
then serves globally — exactly when the rank-deficiency locus
$\{w:\operatorname{rank}H(w)<N\}$ has real codimension at least two, which is
the generic situation for $n\ge N+1$ but is not verified here. The theorem
reduces genericity to one witness of full column rank in the truth's
component, and at $T=1$ the map is explicit enough to exhibit one.

\begin{example}[$T=1$, closed form]\label{ex:T1}
From \eqref{eq:rec}, $P_1=-\eta s\,D_0$ and $M_1=0$, with
$D_0=D(W_0H(w))$ depending on $w$ only through the base model's logits.
Hence $A_1=A_0$, $\Omega=I$, and \eqref{eq:rho} reads (unnormalised)
\[
  \rho_{T=1}(w,X)=\bigl(-\eta s\,D_0(w)\,R_H(w)^{\!\top}X^{\!\top}-B_T,\;\;X-A_TU_c(w)\bigr).
\]
The second block gives $X=A_TU_c(w)$ exactly, and substituting,
$U_c(w)R_H(w)=H(w)$,
\begin{equation}\label{eq:T1}
  B_T\;=\;-\eta s\;D_0(w)\,H(w)^{\!\top}A_T^{\!\top}:
\end{equation}
an equation in $w$ alone whose right side ranges over $\mathcal B$ — the
$N((m-1)+r-N)$ constraints of Theorem~\ref{thm:cap} on the $Nk$ unknowns,
with no unrolling and the seed eliminated in closed form; the budget obtained
by eliminating $X$ here equals the one obtained by counting $X$ as an unknown
in Theorem~\ref{thm:cap}, as it must.\footnote{Eliminating
$X$ does not lower the count: the count is the dimension of the manifold that
$B_T$ lives on, not a tally of which block ``pays for'' which unknown. That is
why the intuition ``at $T=1$ the $A$-block hands $X$ over for free, so
$k\le m$'' is wrong; it is refuted by measurement at $k=24>m=20$.}
Measured$^\dagger$ (job 479587, $N=8$): $D\rho_{T=1}$ has full column rank
at $k=16,20,24$ with $\sigma_{\min}=7.26\times10^{-4},\,1.21\times10^{-4},\,3.15\times10^{-5}$,
identical to three figures with the $T=2$ values, and collapses at $k=28=m+r-N$
($3.3\times10^{-19}$). These are witnesses of Theorem~\ref{thm:suff} at
$T=1$ for the charts $\psi_k$ with $k=16,20,24$ — each $k$ is its own chart
and its own dichotomy. For the working point $T=400$ the witnesses are
direct: $D\rho$ at the truth has full column rank at every $k\le27$ (jobs
467914, 479587); the $T=1$/$T=2$ agreement to three figures is a consistency
check on the closed form, not the witness for $T>1$.
\end{example}

\subsubsection*{The two readings side by side}

\begin{corollary}[the two channels]\label{cor:two}
Under (A1)--(A3) the release $(A_T,B_T)$ has the form \eqref{eq:release}.
Reading it by Theorem~\ref{thm:quot} uses $\operatorname{row}(B_T)\in\operatorname{Gr}(N,r)$
and $\Pi_{\operatorname{row}(B_T)}A_T$, imposes $N(r-N)$ equations on $Nk$
unknowns, and needs (A4), (A5) but no recipe. Reading it by
Definition~\ref{def:rho} uses all of $B_T\in\mathcal B$ and $A_TU_c$,
imposes $N(r-N)+N(m-1)+rN$ equations on $Nk+rN$ unknowns, and needs (A6),
a solver, and the recipe — which \S\ref{sec:measured} shows can be checked
against the floor, fitted jointly, or probed. The $rN$ equations of the second block pay exactly
for the $rN$ unknowns of $X$; the difference between the channels is the
$N(m-1)$ coefficients of $P_T$ in a basis of $\operatorname{row}(B_T)$, which
the projection in \eqref{eq:C} removes and which \eqref{eq:rho} keeps:
\[
  \text{certificate: } k<r-N\qquad\qquad \text{exact: } k<m+r-N .
\]
\end{corollary}
\begin{proof}
Assemble Theorem~\ref{thm:quot}, the proof of Theorem~\ref{thm:cap}, and
Theorem~\ref{thm:suff}.
\end{proof}

\begin{center}\small
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{0.2\linewidth}>{\raggedright\arraybackslash}p{0.27\linewidth}>{\raggedright\arraybackslash}p{0.13\linewidth}>{\raggedright\arraybackslash}p{0.32\linewidth}@{}}\toprule
 & reads & equations per image & needs\\ \midrule
certificate (Thm~\ref{thm:quot}) & $\operatorname{row}(B_T)$; $\Pi_{\operatorname{row}(B_T)}A_T$ & $r-N$ & (A1)--(A5)\\
exact (Thms~\ref{thm:cap}, \ref{thm:suff}) & all of $B_T$; $A_TU_c(w)$ & $(m-1)+r-N$ & (A1)--(A3), (A6); the recipe, checked or fitted (\S\ref{sec:measured}); a solver and a start\\ \bottomrule
\end{tabular}
\end{center}
Both capacities are statements about the chart coordinates $w\in\mathbb R^{Nk}$
of Definition~\ref{def:setup}: the unknown is $w$, the theorems pin $w$, and
the attack returns $\psi(\hat w)$. Nothing in them constrains the chart
$\psi$, which is the attacker's to choose.

\begin{corollary}[the count for a general adapted layer]\label{cor:general}
The proof of Theorem~\ref{thm:cap} uses only that the simulated $B_T$ has
rank at most $\min(N,r)$ and, under a softmax head, zero column sums; it
does not use what the layer computes downstream. Hence, with $m$ the output
width of the adapted layer, the necessary condition on the chart dimension
reads
\begin{center}\small
\begin{tabular}{@{}lll@{}}\toprule
channel & $N\le r$ & $N\ge r$\\ \midrule
certificate (Theorem~\ref{thm:quot}) & $k<r-N$ & — (no certificate)\\
exact, softmax head & $k<m+r-N$ & $k\le (m-1)r/N$\\
exact, any other head & $k<m+r-N+1$ & $k\le mr/N$\\ \bottomrule
\end{tabular}
\end{center}
\end{corollary}
\begin{proof}
For $N\le r$ replace $\mathcal B$ by the rank-$N$ matrices in
$\mathbb R^{m\times r}$ (dimension $N(m+r-N)$) when there is no column-sum
constraint. For $N\ge r$ and $m\ge r$ (so that $\operatorname{rank}B_T=\min(\operatorname{rank}P_T,r)$
can reach $r$), $B_T=P_T(A_0H)^{\!\top}$ lies in $\mathbb R^{m\times r}$
(or in its $(m-1)r$-dimensional column-sum-zero subspace), while Corollary~\ref{cor:seed} still reduces the
seed to $X\in\mathbb R^{r\times N}$; the count is $mr+rN\ge Nk+rN$
(resp.\ $(m-1)r+rN\ge Nk+rN$). The two regimes agree at $N=r$.
\end{proof}
This is the realistic picture for a transformer: the first adapted
projection has $m$ equal to the layer's output width, hundreds to thousands
of coordinates per representation, against $r-N$ for the certificate; and
once $N$ exceeds $r$ — tokens, not images — the budget per representation is
$mr/N$. The $N\ge r$ regime is derived here and untested; every measurement
in \S\ref{sec:measured} has $N<r$.

\paragraph{Eliminating the seed in closed form.} By Corollary~\ref{cor:seed}
the coefficients depend on $X$ only through the Gram $S:=X^{\!\top}X\in
\mathrm{Sym}^+_N$, so $\Omega=\Omega(w,S)$, and the second block of
\eqref{eq:rho} solves for $X$ outright: $X=A_TU_c(w)\,\Omega(w,S)^{-1}$
wherever $\Omega$ is invertible. Substituting into the first block and
adding the consistency condition $X^{\!\top}X=S$ leaves unknowns
$(w,S)\in\mathbb R^{Nk}\times\mathrm{Sym}^+_N$ ($S$ is a Gram matrix, so it
lies in the positive-semidefinite cone, whose boundary
$\operatorname{rank}X<N$ is where the parametrisation degenerates) — $Nk+N(N+1)/2$ numbers, $36$
in place of $rN=128$ at $N=8$, $r=16$ — against $\dim\mathcal B+N(N+1)/2$
equations: the same capacity, a search space smaller by $rN-N(N+1)/2$. Since
\S\ref{sec:opens} finds the initialiser, not the information, to be what
stops the attack, this is the first thing to implement for efficiency. Two caveats: the
count transfers but the reduced problem's local identifiability near
$\partial\mathrm{Sym}^+_N$ is untested, and a matched cell measured no basin
widening \textnormal{[VERIFIED†]} (job 605718: $(w,X)$ recovers $0$ of $3$ at
a $1.5\times$ perturbation against $1$ of $3$ for the $(w,S)$ form, both $0$
of $3$ beyond), so it is a smaller search space, not a bigger basin.

\paragraph{Where the law stops applying: a second channel-closing
mechanism.} Identifiability does fail below the line on a strong encoder,
and it is the imprint law of the next paragraph acting rather than a gap in
the theorems: $\operatorname{rank}B_T$ counts the examples recorded above the
floor, and when that count falls below $N$, hypothesis (A4) fails, $D\rho$
is rank-deficient, and the channel closes \emph{below its own capacity line}
— for reasons of the model rather than the count. On the $97.9\%$ encoder
this happens in nine of the ten batches measured, with
$\operatorname{rank}B_T$ between $1$ and $6$.

What decides which examples survive is measured, and it is the margin
\emph{ordering}. Across the nine rank-deficient batches$^\dagger$ (job
631392), the examples whose imprint lies below the floor are \emph{exactly}
the highest-margin ones, in all nine, with drop counts of $2,2,2,4,5,5,7,7$
and $7$ — so the match is not carried by the cheap two-of-eight cases. Under
random assignment the joint probability of that is $4\times10^{-13}$, and it
reproduces in a second measurement of the same batches (job 650891, nine of
nine).

There is no absolute cut. Below-floor margins run $30$ to $95$ while
above-floor margins reach $74$: an example at margin $44$ drops in one batch
while one at $74$ survives in another. Recording is therefore decided by an
example's \emph{rank within its batch}, not by any threshold on its own
confidence — which is both more surprising than a threshold and more
consequential, since it means a defender cannot secure an example by making
the model confident about it in isolation. Neither its label nor any
coupling between examples enters the ordering, and same-label batches settle
that directly: eight images sharing one label give full rank $8$ when the
model is wrong about all of them (the new-class cells) and rank $1$ when it
is wrong about one and confident about seven. Identical label multiset,
opposite outcomes, distinguished only by the margins.

One caution before that is read as a defence. The floor is relative to each
batch's largest imprint, so a badly fit example added to a batch demotes the
confident ones out of the \emph{count} — but the count is not the same as
recoverability, which turns on the absolute imprint against numerical
precision. And the absolute imprints point the other way: high-margin
examples sit at $10^{-41}$--$10^{-38}$ in an all-confident batch and at
$10^{-26}$--$10^{-25}$ in batches containing a badly fit one, fifteen orders
higher. Those are different image sets, so the comparison is uncontrolled and
proves nothing; it is exactly the measurement named as missing in
\S\ref{sec:opens}, and until it is made, the rank should be read as a count
of what dominates the release and not as a certificate about the rest.

That the identity is a measurement and not a definition matters here, and it
holds: the rank of $B_T$ read from its own singular spectrum, the rank of the
stacked imprints, and the count above the threshold are three separate
computations, and they agree in all forty batches. In the rank-deficient
cells the deficiency is visible in the release's own spectrum
($\sigma_N/\sigma_1$ of $1.5\times10^{-19}$ and $1.0\times10^{-16}$), so an
attacker sees it without the decomposition.

Two earlier readings of this paragraph were wrong and are worth recording,
because both were natural. Encoder quality alone was blamed, by quoting a
single strong rung of $2.0\times10^{-15}$ that was the geometric mean of two
populations seven orders apart — a value no measurement lies within three
orders of. Then the label multiset was blamed, on the observation that the
distinct-label draw keeps rank $8$ on-chart where the repeated-label draw
falls to $6$. That comparison is confounded: the two draws are not the same
images relabelled but different images, and the repeated one happens to
contain examples of margin $46$ and $37$ against a maximum of $23$ in the
other. On raw digits both draws give rank $6$. Chart projection is the one
factor that genuinely moves the count, and it moves it \emph{upward} for the
attacker: projecting onto a $16$-dimensional chart degrades the image, the
model becomes less certain, and more is recorded.

\paragraph{What the release records, per example.} The gradient in $B$ is
$\nabla_BL=s\,D\,(AH)^{\!\top}=s\sum_i D[:,i]\,(Ah_i)^{\!\top}$, a sum of
rank-one pieces, one per private example. Under (A2) and (A3) the release is
therefore
\begin{equation}\label{eq:imprint}
  B_T=\sum_{i=1}^N C_i,\qquad
  C_i=-\eta s\sum_{t<T} D_t[:,i]\,(A_th_i)^{\!\top},\qquad
  \|C_i\|\;\le\;\eta s\max_t\|A_th_i\|\;\sum_{t<T}\bigl\|D_t[:,i]\bigr\|,
\end{equation}
The split is additive with one term per example and no cross-index term —
exactly what the retracted reading of $P_T$ was not. Within the bound, the
error factor $\sum_t\|D_t[:,i]\|$ involves example $i$ alone, while the
scale factor $\max_t\|A_th_i\|$ contains the shared adapter and so depends
on the whole batch through the trajectory. That is not a caveat but the
content: the measurement below finds that common factor near-constant across
examples and batches, and its constancy is what earns the per-example
reading. Since $B_0=0$ starts the trajectory at $W_0$, an example the base
model already classifies with margin $M$ enters at the scale of its softmax
error there, $\mathrm e^{-M}$.

Two things are measured, and the second is the substantive one. First, in
all forty batches $\operatorname{rank}B_T$ equals the number of examples
whose imprint clears the floor$^\dagger$ (job 631392, gated at
$\|\sum_iC_i-B_T\|/\|B_T\|<4\times10^{-12}$): the rank of the release is a
count of the examples it records. Second, and this is the non-obvious part,
the \emph{ratio} $\|C_i\|/\sum_t\|D_t[:,i]\|$ is tight — median
$3.0\times10^{-3}$ overall, and within a single batch it varies by a median
of $0.4$ and at worst $0.95$ of a decade. Since $C_i$ is a sum over $T=400$
steps of terms each carrying $D_t[:,i]$, the per-step contributions could
have cancelled to any degree; that they do not, uniformly across encoders
and batches, is a dynamical fact and not an algebraic one.

The rank correlation between $\|C_i\|$ and $\sum_t\|D_t[:,i]\|$ is then
$971$ of $1120$ concordant pairs (Kendall $\tau\approx0.73$), but it should
be read per encoder rather than pooled,
because one population contributes nothing:

The random encoder is at chance for a reason that is itself quantitative
rather than an excuse: its eight examples differ in accumulated error by
only $4$ to $21\%$, which is smaller than the ratio's own within-batch
scatter of a third to a whole decade. So the ordering is resolvable exactly
when the spread in error exceeds the scatter in the ratio — a criterion that
says in advance where the law will and will not order examples, and which
the three trained encoders satisfy comfortably.\footnote{Because $C_i$
contains $D_t[:,i]$ as a factor, a log-log plot of $\|C_i\|$ against
$\sum_t\|D_t[:,i]\|$ is close to plotting a quantity against its own bound,
and a slope-one line is largely forced by \eqref{eq:imprint}. The evidence
is therefore the tightness of the ratio and the rank identity, not the
correlation; Figure~\ref{fig:imprint} is an illustration of the former, not
the argument for it.}

Two consequences follow, and both are basis-free. \emph{A private set the
model already fits is not recorded at all}$^\dagger$ (job 627166): for eight
digits of margin $56$--$95$ under the strong model, every imprint is below
$1.5\times10^{-24}$ in absolute terms — the release is numerically empty,
and a rank read off it at a relative threshold is meaningless. The same eight
digits through the random, weak and mid encoders give eight imprints of order
one. \emph{Where the model is wrong, the release is sharply selective}: in a
batch containing one misclassified digit (margin $-3.9$) the relative
imprints run $1$, $7.6\times10^{-2}$, $8.2\times10^{-7}$,
$4.8\times10^{-8}$ for margins $-3.9$, $4.1$, $15.6$, $17.7$, and the four
examples with margins above $39$ sit between $3\times10^{-17}$ and
$10^{-30}$. What leaks is what the model had to learn.

For the attacker, the capacity $m+r-N$ is an upper bound that a confident
model does not spend, and $\operatorname{rank}B_T$ read off the release says
how many examples were recorded at all. For the defender, the per-example
accumulated residual during their own fine-tuning — a quantity already
computed in the course of training — is the meter. Batch composition cannot
enter the \emph{form} of the bound in \eqref{eq:imprint}, and it enters its
\emph{value} only weakly: the accumulated error is trajectory-dependent and
the trajectory is shared, but holding the images fixed and permuting the
label multiset moves the accumulated error by $0.1$--$4\%$ and the imprint
by about $5\%$. That is composition-independence tested against label
permutation only; the case that would test it properly — the same example in
a batch of confident examples, then beside a badly fit one — is not in the
design (\S\ref{sec:opens}).\footnote{An earlier version of this section attributed per-example
content to the columns of $P_T$ and reported a coupling between batch-mates.
Column $i$ of $P_T=-\eta s(\sum_tD_t)R_H^{\!\top}$ mixes every example
$j\ge i$ in batch order through the triangular factor of $H=U_cR_H$: it is
basis- and order-dependent, and permuting the batch changes it. A gated
trace of the training loop refuted the coupling from the other side (job
628731: the margins it was supposed to move do not move — a margin-$71$
digit ends at $71.1$). Equation~\eqref{eq:imprint} is the basis-free
replacement, and every rank statement above was unaffected by the
retraction.}

\paragraph{A new category is the opposite extreme, and one shared label
costs conditioning but not identifiability.} Fine-tuning a class the model
has never had is the maximal case of the mechanism: the head is extended by
a zero row, so the model assigns the new examples no probability at all and
is maximally wrong about every one of them. Measured on a CIFAR-10
multilayer perceptron (best test accuracy $53\%$ — the weak regime) whose
head is extended to $m=11$, with eight CIFAR-100 flowers all labelled into
the new class$^\dagger$ (job 656205): every imprint lies within a factor
$2.5$ of the largest (relative $0.4$--$1.0$) and $\operatorname{rank}B_T=8$.
The new category is recorded in full.

The eight imprints are also \emph{aligned}, which is a second mechanism and
not the same one: the Gram of the stacked imprints has
$\sigma_N/\sigma_1=0.145$ (raw) and $0.097$ (on-chart) with mean pairwise
cosine $0.50$--$0.57$, because every error points toward the same corner of
the simplex. That is the mirror image of the confident-batch case above,
where the imprints were \emph{absent}. Both depress $\sigma_{\min}(D\rho)$,
and what distinguishes them is magnitude against angle — which is why the
imprint Gram is worth recording beside the imprint norms. Alignment,
however, is not a wall: $\sigma_{\min}$ at the truth is $2.4\times10^{-6}$
and the inversion reaches the residual floor in $62$ iterations, recovering
the chart coordinates to $7\times10^{-14}$. A shared label costs
conditioning and nothing more.

Two details support the reading. The base model files the eight flowers as
frog, cat, truck, truck, frog, automobile, frog and deer — several classes,
so their errors are only moderately aligned, which is the within-cell version
of the same prediction: had they all been assigned to one class the
alignment would have been stronger. And an attacker's prior built from
public flowers draws them to $0.305$. This cell isolates the new-class and
shared-label structure rather than replicating the encoder-quality ladder,
since its base model is weak by construction; the same construction on the
$97.9\%$ digit model, with an EMNIST letter as the new class, is running and
will put both extremes of the mechanism on a single encoder.

\paragraph{On a strong model, ordinary private data collapses to one
recorded example.} The imprint spectrum can be read off the release without
any private knowledge, and $\sigma_2/\sigma_1(B_T)<10^{-3}$ is the
attacker's test for ``one example carries this''. Its frequency over the
forty batches$^\dagger$ (job 650891): $9$ of $10$ on the strong encoder
($\sigma_2/\sigma_1$ from $4.8\times10^{-16}$ to $0.07$; ranks
$1,1,1,3,3,4,6,6,6,8$), and $0$ of $10$ on both the weak and the mid
encoders ($0.021$--$0.58$ and $0.005$--$0.69$). Both \emph{ordinary} random
draws trigger on the strong model ($2.1$ and $2.9\times10^{-5}$ raw), and
the single non-trigger there is the batch built to contain two low-margin
examples. What the test detects, though, is \emph{one dominant direction} and
not \emph{one recorded example}, and the two are different claims: of the
nine triggering batches the rank is $1$ in three, $3$ in two, $4$ in one,
$6$ in three and $8$ in one — a full-rank release with
$\sigma_2/\sigma_1=2.4\times10^{-4}$. So a single dominant imprint is what
ordinary private data produces on this $97.9\%$ model and never on the
weaker ones; a literal reduction to one recorded example happens in three of
the ten.\footnote{The test can fire for the wrong
reason. The random encoder produces one false positive
($\sigma_2/\sigma_1=8.7\times10^{-4}$ at full rank $8$) on a batch of eight
$1$s, all at residual $\approx0.95$: there the spectrum reflects the near
collinearity of the features, not a difference in what was recorded. A
spectral gap is evidence about the release, not automatically about the
margins, and a negative control that acts on a non-triggered read is what
makes the test load-bearing.}

\paragraph{Private data the model gets wrong is recorded in full, whatever
its provenance.} The mechanism predicts that data the base model
misclassifies is recorded in full, and a separate argument predicts that
data unlike the model's own domain is drawn badly by an attacker's prior.
Two foreign sets were used to test both$^\dagger$ (job 644062, strong
backbone, raw digits, $8$ private examples; the in-distribution control is a
random distinct-label draw from the MNIST test split through the identical
pipeline). The first prediction holds. The second was not tested at all, for
a reason worth recording.
\begin{center}\small
\begin{tabular}{@{}lcccc@{}}\toprule
private set & base accuracy on it & median margin & $\operatorname{rank}B_T$ & the MNIST chart's error\\ \midrule
MNIST (control) & $8/8$ & $23.8$ & $6$ & $0.518$\\
rendered fonts & $8/8$ & $8.9$ & $6$ ($4$ at $10^{-8}$) & $0.383$\\
UCI optdigits scans & $4/8$ & $-2.0$ & \textbf{8} & $0.317$\\ \bottomrule
\end{tabular}
\end{center}
The recording half runs through the margin, not through foreignness: the
rendered fonts are foreign and yet classified $8/8$ correctly, and their rank
is $6$ rather than full, because three bold glyphs are recognised with
margins $31$--$47$. It is the scanned digits, misclassified half the time by
a $97.9\%$ model, that reach full rank $8$ with every imprint of order one,
where the in-distribution control sits at rank $6$. Foreign data tends to be
recorded in full because a model tends to get it wrong, and the imprint law
holds on data of a kind the model has never seen.

The chart half looked like a refutation — the MNIST chart appearing to draw
both foreign sets \emph{better} than MNIST — and it is an artefact of
resolution. Blurring or downsampling the same eight control digits and
re-measuring on the same $k=16$ basis$^\dagger$ (job 667729) gives
$0.354$, $0.343$, $\mathbf{0.320}$ and $0.268$ through $14$, $10$, $8$ and
$6$ pixels, against optdigits' $0.317$; a $2000$-digit reference moves from
$0.521$ to $0.336$ the same way, and the fonts' $0.383$ sits where blurred
MNIST sits. So a low-dimensional linear chart represents smooth images
better whatever data fitted it, MNIST at optdigits' own resolution is
indistinguishable from optdigits, and the provenance of the chart contributed
nothing measurable. The claim that both axes favour the attacker is
withdrawn; the recording half stands on its own. A provenance test that would
survive this needs foreign digits at \emph{matched} resolution — USPS at
$16\times16$ upsampled exactly as the control is downsampled — and has not
been run.

\paragraph{The certificate is the recipe-free face of the same fact.}
Theorem~\ref{thm:quot} builds $C=\Pi_{\mathrm{row}(B_T)}A_T$ and gives
$CH=0$ with $\operatorname{rank}C=r-N$ under (A4). The imprint law says what
happens when (A4) fails, and it is not a degradation but a sharpening. By
Theorem~\ref{thm:traj}, $\mathrm{row}(B_T)$ is spanned by the vectors
$A_0h_i$ weighted by each example's imprint, so when only $N'$ examples are
recorded it is spanned by \emph{their} directions alone: $\operatorname{rank}
C=r-N'$, and $Ch_i=0$ for a recorded example while $Ch_i$ is $O(1)$ for an
invisible one. Measured$^\dagger$ (job 701679, strong backbone,
in-distribution draw): $\operatorname{rank}B_T=6$ with
$\operatorname{rank}C=10=r-N'$ exactly, and the per-image residuals
$\|Ch_i\|/\|A_Th_i\|$ are $8.8\times10^{-15}$, $\mathbf{0.84}$,
$3.7\times10^{-10}$, $6.1\times10^{-10}$, $9.9\times10^{-14}$,
$\mathbf{0.49}$, $3.7\times10^{-10}$, $8.5\times10^{-13}$ — the two bold
entries are exactly the two invisible examples, eight orders of magnitude
clear of the largest recorded one. On the chart-projected version, where all
eight are recorded, all eight residuals fall below $3\times10^{-8}$.

Two things make this worth having beyond elegance. It is \emph{recipe-free}:
$C$ needs no $\eta$, no $T$ and no labels, so an attacker can test any
candidate image for membership without knowing how the adapter was trained.
And the condition $C\varphi(\psi(w))=0$ is linear in the features and needs
no unrolled dynamics at all, a far simpler landscape than the full residual,
at the certificate line $k<r-N'$ rather than $k<m+r-N'$. That line comes from
the kernel rather than from $C$'s rank: $\ker C=\mathrm{span}\{h_i:\text{recorded}\}
\oplus\ker A_0$ has dimension $N'+(n-r)$, so it is of codimension $r-N'$ in
feature space, and a $k$-dimensional chart image meets it in generically
$k-(r-N')$ dimensions. Below the line the recorded examples are isolated
solutions and generically the only ones; above it each lies on a
positive-dimensional family that satisfies the certificate without being the
image. When several examples are recorded they are all equally valid
solutions, so a search started at random reaches \emph{one} of them with no
way to steer and no way to tell from the objective which — the $N'=1$ case is
the one where recovering a solution and recovering \emph{the} example
coincide.
Two caveats belong with it. The vanishing is approximate rather than exact
below full rank — the recorded residuals sit at $10^{-10}$, not at machine
precision, because the invisible examples' contribution to $M_T$ does not
identically vanish — so the test needs a threshold, and the gap it must
straddle is eight orders wide here. And $Ch=0$ is membership of a subspace,
not of the private set: any candidate whose features lie in the span of the
recorded ones passes, which is tight when $N'=1$ and a $6$-dimensional
subspace of $\mathbb R^{1000}$ when $N'=6$. Under Adam
$\operatorname{rank}B_T=r$, $C\equiv0$, and none of this applies.

\subsubsection*{What an attacker gets from it}

\paragraph{The image finishes long before the residual does.} On the
$95.1\%$ backbone every below-line cell stops at residual
$10^{-13}$--$10^{-15}$ after $300$ iterations — fifteen orders above the
reproduction floor, so by the classification of \S\ref{sec:measured} these
are search failures. Yet the median image error already equals the chart's
own representation error to within $10^{-6}$--$10^{-3}$ at
$k=6,10,14,16$$^\dagger$ (job 614344). So on this backbone, at this budget, the median image error stood within
$0.4\%$ of the chart's floor while the residual stood fifteen orders above
its own — which is the attacker-relevant statement: no tight residual was
needed for the best available image here. Three things it is not. Neither
quantity finished: all five cells stopped at the iteration cap, so the image
was never shown to have \emph{stopped} moving either. The gap is not a
function of the residual, so it does not lift off this backbone — pooling
the twenty-four cells on disk that record both, at residual
$\approx2\times10^{-14}$ the gap runs from $4\times10^{-8}$ to
$1.2\times10^{-3}$, five orders at matched residual, with rank correlation
only $+0.43$; the obvious first-order model $\sqrt{\text{residual}}/
\sigma_{\min}$ lifts that only to $+0.53$, so conditioning does not explain
the scatter either. And what those last fifteen orders would do was not
observed at all, since no cell reached the floor — and off-chart there is reason to think they could do harm rather
than nothing. Driving the residual down moves the coordinates toward the
release-optimal point, which for data not lying in the chart is not the
representation-optimal point; continuing could therefore move the image away
from the chart's best. That is the same distinction that organises the whole
section, now on the convergence axis, and it is untested: one cell driven to
the floor off-chart would settle whether an attacker should stop early, and
recording the gap and the residual every twenty-five iterations along a
single cell would settle the plateau question directly, without the
cross-cell scatter. For
locating the boundary the residual remains the right instrument, since it is
what separates an alias from a search failure; for judging what an attacker
obtains it is not, and the image-space gap of this section is.



\paragraph{A free convergence certificate in image space.} The identity above
is not only a result; it is an instrument. On an on-chart cell a converged
inversion \emph{must} land on the chart's own best point, so
$\bigl|\,\text{median image error}-\text{chart representation error}\,\bigr|$
is zero at convergence and grows as the solver falls short — an image-space
diagnostic that costs nothing, needs no floor known in advance, and is
independent of the residual. Measured across twenty-two on-chart rows it
tracks the residual over fourteen orders$^\dagger$: $0$ at residuals
$10^{-31}$, $10^{-13}$ at $5\times10^{-17}$, $10^{-7}$ at $10^{-14}$,
$1.6\times10^{-3}$ at $2.9\times10^{-9}$. Two uses. At the fixed $80$-iteration
budget of the trained-model sweep it gives the shortfall as a curve in $k$ —
$3.8\times10^{-8}$, $7.2\times10^{-5}$, $4.1\times10^{-4}$,
$1.5\times10^{-3}$, $2.4\times10^{-3}$ at $k=14,16,17,18,20$ — for cells that
never reached the floor at all. And it flags the least converged member of a
comparison independently of the residual: among the four charts of
\S\ref{sec:opens} it singles out the ReLU-decoder VAE ($1.6\times10^{-3}$
against $10^{-13}$--$10^{-6}$ for the rest), the same cell whose
solved $\sigma_{\min}$ is three times its value at the truth. Separately,
the VAE charts' conditioning penalty is largely their decoder's own geometry
rather than anything about the encoder: the VAE decoder's Jacobian has
$\sigma_{\min}=0.008$ against $\sigma_{\max}=6.7$ (condition number
${\sim}830$) where a PCA chart's is exactly $1$, of the right order to
account for most of the ${\sim}2000\times$ gap in $\sigma_{\min}(D\rho)$
between them$^\dagger$ (job 626565). That verdict
carries the same caveat as the comparison itself: for a VAE chart the
reference $c_i$ comes from the inner solve, so the flag conflates outer
shortfall with inner slack and is unresolved in the same way.\footnote{An asymmetry to record while using it: for a PCA chart the
projection onto the chart is exact (orthonormal basis, closed form), while
for a VAE chart it is a $300$-step Adam solve with no convergence criterion
(vae\_chart.py:85). Between families this biases conservatively for the
richness ordering — an under-converged inner projection overstates a VAE's
error and so understates its richness. \emph{Within} the VAE family it need
not, because that same inner solve returns $W_{\mathrm{true}}$, which is both
the point the adapter is trained on and the point at which $\sigma_{\min}$ is
evaluated. Measured$^\dagger$ (job 626565), the concern does not bite for the
GELU decoder: at $300$, $1000$ and $3000$ inner iterations the inner loss is
$90.5$ to four digits and $\sigma_{\min}$ at the truth is
$1.13\times10^{-11}$ at every level and at three perturbations of the truth,
so the truth does not move with the inner budget there. Untested for the
ReLU decoder and the $\beta$ arms. The GELU/ReLU pair
therefore compares two separately fitted models differing in encoder,
decoder and inner starting point at once, not one activation swap, and its
$1.04\times$ in richness against $1.15\times$ in $\sigma_{\min}$ is
unresolved on three counts: a single $A_0$ draw, below the replicate spread,
and an unmeasured inner solve in both coordinates.}

\paragraph{The boundary is about the chart, not the image.} Same digits,
same $N=8$, $m=10$, $r=16$, line $k<18$; three charts at matched $k$\textsuperscript{$\dagger$}
(job 574169): the principal subspace (PCA), the same subspace under a
non-linear warp, and an \emph{oracle-built} chart constructed from the eight
private digits themselves so that it contains them exactly — a control on
what the count can and cannot see, not an attack. Two predictions of Theorems~\ref{thm:cap}--\ref{thm:suff} come
apart:

blind to how the chart is built. What comes back is not: at $k=17$ every
chart recovers its own representable image to $10^{-14}$, and against the
real digit the three differ by thirteen orders. The boundary is therefore a
statement about the chart coordinates, not about the images. Below it the
coordinates $w$ are locally identifiable and the attack returns
$\psi(\hat w)$; how close that is to the private image depends entirely on
whether the chart can represent it. The right reading is a capacity one, and it is chart-\emph{independent}:
the release exposes exactly $m+r-N$ independent coordinates per example, in
any basis whatever. Reconstruction then factorises as
\emph{capacity} $\times$ \emph{chart quality} — this law fixes the first
exactly, and a prior supplies the second. Equivalently, and this is the
honest conditional form: exact reconstruction holds if and only if the
private data lies in an attacker-buildable chart of dimension below
$m+r-N$; the capacity is proved, and building that chart is the open
problem. What the law does not bound is the fraction of the image an
attacker can reach, so it is not a leakage bound: nothing constrains the
attacker's chart to be data-agnostic, and the oracle-built chart — which
spans the private images and so has $k=N=8$, far below the line — returns
them at $10^{-14}$ (last rows); an attacker does not have that chart, and
what the row shows is what a chart that happened to contain the images would
give. What the boundary limits is the search space, and
the search space is the attacker's to choose — which is the counting
argument handing the generative-prior direction of Rev~9 its role: a prior
is a chart-builder (Figure~\ref{fig:chart}).

\paragraph{A richer chart is not harder to invert.} Since the count caps the
chart's dimension and not its quality, the natural worry is that a chart
which draws better is harder to invert — that expressiveness is paid for in
conditioning. It is not, within a family. Varying only the $\beta$ of a VAE
at fixed latent dimension, architecture and training data$^\dagger$ (job
626565, weak encoder): $\beta=0.25$ represents the private digits markedly
better than $\beta=1$ (error $0.222$ against $0.339$) at the same
conditioning ($\sigma_{\min}$ at the truth $1.15\times10^{-11}$ against
$1.13\times10^{-11}$; three $A_0$ draws span $0.99$--$1.28\times10^{-11}$).
Across \emph{families} the differences are large — a VAE chart is
${\sim}2000\times$ worse conditioned than a PCA chart on this encoder — but
that is accounted for by the decoder's own Jacobian (condition number
${\sim}830$ against exactly $1$ for a linear chart), not by what the chart
can draw. An ordering across four charts that appeared monotone in richness
was withdrawn on this evidence: it does not survive a within-family sweep,
and the linear pair inverts on two of the four encoders. The practical
reading is favourable to the attacker and to the defender's
threat model alike: \emph{spend the $k$ coordinates on the best chart
available}, because richness is free in conditioning terms provided the
decoder itself is well conditioned.

\paragraph{Past the line: what the alternative point is.} On the synthetic
test-bed — whose chart $\psi$ is the generator that produced the images, so
it contains the truth exactly, the most favourable case of the previous
paragraph — the aliases of the sharpness table sit at image error
$1.7\times10^{-3}$ ($N=4$, $k=34,38$), $2.6$--$9.3\times10^{-3}$ ($N=8$,
$k=28$ to $44$), $1.2$--$1.7\times10^{-2}$ ($N=12$, $k=26,30$) and
$6.4\times10^{-3}$--$1.4\times10^{-2}$ ($N=14$, $k=22$--$24$). The error
increases overall with distance past the line at every $N$, not
monotonically ($N=8$: $5.7\times10^{-3}$ at $k=30$, $5.3\times10^{-3}$ at
$k=32$); $9$ of the $13$ cells have every image inside the study's $10^{-2}$
tolerance, and the four outside it are all at $N=12$ and $N=14$ — at $N=8$
even the cell sixteen units past the line stays inside. The split is by
$N$, the defender's variable, not by $k$, the attacker's. On real digits
(job 568095) the same rows are worse: the past-line aliases at $r=16$ and
$r=32$ — residual at the floor — sit at $2.3$--$6.8\times10^{-2}$,
increasing with distance ($r=16$: $2.7$, $5.4$, $6.8\times10^{-2}$ at
$k=18,22,26$), and five of the eight past-line cells have no image inside
tolerance; the three $r=8$ past-line rows stop at residuals
$10^{-16}$--$9\times10^{-15}$, search failures rather than aliases, at
$3.5$--$10\times10^{-2}$. The continuum of
Theorem~\ref{thm:cap} was then walked$^\dagger$ (job 482338): from the truth,
step along the right singular vector of $\sigma_{\min}(D\rho)$, retract onto
$\rho^{-1}(0)$ by a few LM steps, repeat (Figure~\ref{fig:quadrant}).

One procedure, one solver — the control ran a $12$-step budget against
$24$, and below the line the numerical null space is empty ($n_{\mathrm{null}}=0$),
so its step is along the smallest non-null direction, which the retraction
undoes — and the image moves by $10^{-14}$ on one side and by percent on the
other.
This is Theorem~\ref{thm:cap} and Theorem~\ref{thm:suff} as a single
experiment: the set of exact reproductions is a point below the line and a
walkable manifold at and past it. Each walk follows one null direction
(of $14$ and $40$), so the errors reached bound the extent from below; the
at-line walk had not stopped rising. On the paths measured the alternative
solutions lie within $1.4$--$3.8\%$ of the private images, in a chart that
contains them; $3.8\%$ is a visible degradation, and whether the images
remain recognisable was not assessed. Hence $k<m+r-N$ is the boundary of
\emph{exact} identifiability of the chart coordinates, and it is not a
leakage bound: below it the image reached is only as good as the chart, and
past it the alternatives \emph{found so far} lie a few percent from the
truth — sub-percent to $1.7\%$ in the synthetic chart that contains the
truth, $2$--$7\%$ on real digits, worse at larger $N$ — and every such
figure is a lower bound on the fibre's extent, not a ceiling: the cells
start next to the truth and find the nearest branch, and both walks were
stopped by budget along one direction. Whether the alternatives stay
recognisable further out is not established.

\subsubsection*{The recipe}

\paragraph{How much of the recipe the release itself gives up.}
Definition~\ref{def:rho} takes $(\eta,s,T)$ and the labels as known. Under a
\emph{weights-only} release — the attacker has $(A_T,B_T)$, $W_0$, $\varphi$,
$\psi$ and nothing else — three measurements$^\dagger$ at $k=12$, $N=8$,
$T=400$, $\eta=0.01$ (job 484255) show that a wrong recipe is detected (R1)
and that $\eta$, the $(\eta,T)$ split and the labels are fitted from the
release alone (R2, R3). A fourth (R4) needs more access and is scoped as
such.

\emph{(R1) None of seven wrong recipes reached the floor.} With the control first:
\begin{center}\small
\begin{tabular}{@{}lll@{}}\toprule
hypothesis simulated & residual & image error\\ \midrule
correct (control) & $5.3\times10^{-31}$, at the floor & $2.8\times10^{-15}$\\
$T+1$ step ($401$ of $400$) & $6.0\times10^{-8}$ & $4.1\times10^{-3}$\\
$\eta\times1.01$ & $9.1\times10^{-7}$ & $1.8\times10^{-2}$\\
$T+25\%$ \;/\; $T-25\%$ & $1.6\times10^{-4}$ \;/\; $2.9\times10^{-3}$ & $0.30$ \;/\; $0.40$\\
$\eta\times2$ \;/\; $\eta/2$ & $1.6\times10^{-3}$ \;/\; $3.4\times10^{-2}$ & $0.54$ \;/\; $0.46$\\
wrong optimiser (SGD release inverted as Adam) & $4.9$ & $0.26$\\ \bottomrule
\end{tabular}
\end{center}
The true recipe is the unique floor-reacher, separated from the nearest
wrong hypothesis — one step in four hundred — by twenty-three orders. The
residual is \emph{not} monotone in the size of the recipe error
($\eta\times2$ ends lower than $\eta/2$; $T+25\%$ is eighteen times lower
than $T-25\%$), so it is a detector, not a graded objective; it is computed
from the release and the candidate alone, never from the private images.
Seven hypotheses, one cell, one seed.

\emph{(R2) The learning rate can be fitted jointly with the data.} Carrying
$\eta$ as a free unknown beside $(w,X)$ from starts at $1\times$, $2\times$,
$0.5\times$ the truth, the fit returns $\eta$ to relative error
$5.2\times10^{-16}$ and the images to $3\times10^{-15}$ at the floor in all
three cases — one extra unknown against a slack of $120$ in \eqref{eq:cap}.

\emph{(R3) $\eta T$ is not a degeneracy of discrete SGD, and labels are
identifiable.} Holding $\eta T=4$ and varying the split, $(T,\eta)=(100,0.04)$,
$(200,0.02)$, $(800,0.005)$ stop at residuals $3.1\times10^{-7}$,
$3.5\times10^{-8}$, $8.8\times10^{-9}$ while the true $(400,0.01)$ reaches
$6.4\times10^{-31}$: the product does not determine the release, so $\eta$ and
$T$ are separately identifiable — the degeneracy is a gradient-flow statement
that finite steps break, and this is the weakest separation in the study
(twenty-two orders), as it should be if the degeneracy is approached in the
small-step limit. The wrong splits still return images within
$0.1$--$0.7\%$. Labels: the true assignment reaches the floor
($7.2\times10^{-31}$); swapping two examples ($2.5\times10^{-3}$) or shifting
all of them ($7.5\times10^{-6}$) does not.

\emph{(R5) The batch size is absorbed into the step, and is not
identifiable.} This one is a corollary of \eqref{eq:rec} rather than a
measurement. If the recipe minimises the \emph{mean} rather than the sum, as
implementations normally do, then Lemma~\ref{lem:grad} carries a factor $1/N$
and every recurrence in \eqref{eq:rec} depends on $(\eta,s,N)$ only through
the single scalar $\eta s/N$. Three consequences follow. First, a simulation
handed a subset of $N'$ examples reproduces the release only at step $\eta
N'/N$: at the original step it is a different recipe, and its residual at the
truth will not reach the floor, which is (R1) turned on the analyst. The
subset simulation is legitimate because the omitted examples contribute no
gradient --- that is what invisibility means in \eqref{eq:imprint} --- but the
normalisation must be corrected for the ones that remain. Second, the attacker
never needs $N$: they fit one scalar, which is precisely the fit (R2) measures
succeeding to $5.2\times10^{-16}$. Third, and this is the defender-side
reading, $N$ cannot be recovered from the release at all. Only $N'$, the
number of examples recorded, can be, and $\operatorname{rank}B_T$ returns it;
and the two halves of that statement hold to different strengths. That $\eta$
and $N$ are inseparable, and that neither is needed, is exact and
unconditional: it is a statement about the parametrisation of a fixed batch.
That a batch of eight of which two are invisible is indistinguishable from a
batch of six at a proportionally smaller step is exact only if the two
contribute \emph{identically} zero gradient at every step, and invisibility is
a floor rather than a zero, so the two releases agree to the floor and not
beyond. They differ through two channels, not one: the omitted imprints
$\sum_{i\,\mathrm{inv}}C_i$ directly, and the invisible rows of $M_T$, which
enter $A_t$ and so perturb the \emph{recorded} examples' imprints as well ---
the same second channel that keeps the certificate residuals at $10^{-10}$
rather than at machine precision. The second can be the larger of the two, so
a measured discrepancy above the omitted-imprint scale is the trajectory
channel and not a refutation. The exception is instructive. Under
decoupled weight decay the step contributes $-\eta\,\mathrm{wd}\,B_t$ as well,
so the release determines the two combinations $\eta s/N$ and
$\eta\,\mathrm{wd}$; a defender whose weight decay is nonzero \emph{and}
published thereby publishes their batch size.

\emph{(R4) With continued-training access, a decaying schedule leaks the
step count that a constant one hides.} This result assumes an access model
the others do not: the attacker submits a batch of their own choosing,
$H_p=\varphi(\psi(W_p))$ with labels they choose, and observes the
\emph{victim's} training process take one more step on it with its hidden
$\eta$, weight decay and scheduler state — a checkpoint that includes
optimiser and scheduler state, or a fine-tuning service that continues the
job. An attacker who runs the step themselves sets $\eta$ and learns nothing;
under weights-only release (R4) does not apply, and neither the schedule nor
$T$ is recoverable by it. Within that access model no private data enters:
every regressor is computable from $(A_T,B_T)$, $W_0$ and the probe batch.
One step under decoupled weight decay is $\Delta B=-\eta\nabla_BL-\eta\,\mathrm{wd}\,B_T$, linear in
$(\eta,\eta\,\mathrm{wd})$: the two-parameter least squares returns $\eta$ to
$5\times10^{-16}$ and $\mathrm{wd}=10^{-3},10^{-2}$ to $10^{-11}$ relative
(job 487290), and
$\cos(\Delta B,-\nabla_BL)=1$ under SGD against $0.43$ under Adam separates
the optimiser family (job 485912). The same probe returns the per-step $\eta$ at the continuation steps
to $3\times10^{-14}$ or better (jobs 487290, 488314). Under a constant rate that is all it
returns, and $T$ stays hidden. Under a cosine schedule of known family, a
global fit of $(\text{base},T_{\max},T)$ to six probe steps (eliminate the
base by ratios, grid the two integers, solve the base in closed form) returns
all three exactly — $T=400$, $T_{\max}=1000$, base $0.0100$ — and identically
from windows of $20$, $60$ and $150$ steps, and for releases at $T=200$ and
$T=700$ as well$^\dagger$ (job 591848, a global grid over $(t,T_{\max})$
with the base in closed form, fit residual $\le1.3\times10^{-16}$, applied to
the probed values of job 488314). A schedule encodes position on its own
curve — and with checkpoint access $T$ is usually in the metadata anyway, so
the content of (R4) is the fine-tuning-service case.\footnote{A gradient fit
of the same three parameters had landed $30$--$40\%$ off and was retracted;
it was a local minimum of a multimodal objective, not a degeneracy, and the
global fit was reproduced independently by an auditor before the script was
committed.} The family is assumed known;
selecting it is a model-selection problem not run here.

\subsection{What is open}
\label{sec:opens}

\begin{enumerate}\itemsep3pt
\item \textbf{Local, not global.} Theorem~\ref{thm:suff} isolates the truth
  in a neighbourhood; injectivity of the release map on the whole manifold
  is open, as it is for the certificate in Rev~9. No alias has appeared below
  the line$^\dagger$; that is evidence. The continuation walks one null
  direction per cell — a multi-direction version, which would say whether the
  extent is direction-dependent, has not been run — and the at-line cell was
  still rising when its budget ended. ``Recognisable'' is measured to $3.8\%$
  and no further. And since the boundary is about the chart, the question
  that matters for privacy is which chart an attacker can build
  \emph{without} the private images — a population prior — and how far below
  the line it lands; that is the generative-prior direction, now with a
  precise specification. Two priors compose here. A chart \emph{local to the
  label}, $\psi_y(w)=\mu_y+V_yw$ or a class-conditional generator, spends
  all $k$ coordinates on within-class variation (the label is given by the
  task or identifiable from the release, R3), and is better conditioned than
  a global chart. And the \emph{encoder} sets a ceiling above the count: two
  images with the same features $\varphi(x)$ are indistinguishable to any
  attack on this adapter, so the chart may as well be built in feature space
  and decoded by a separate feature-inversion prior; detail the encoder
  discards is lost whatever the chart. Cells comparing global PCA, local PCA
  and a conditional VAE at equal $k$ on the trained backbone are queued.
\item \textbf{The recipe family, and the access model.} What
  \S\ref{sec:measured} leaves open is the \emph{family}: the schedule fit
  assumes cosine, and choosing among families is model selection, which the
  floor test of (R1) would presumably rank but which has not been run; and
  (R4) holds only under continued-training access — under weights-only
  release the schedule and $T$ are not recovered by anything here. The count \eqref{eq:cap} bounds the recipe
  as it bounds the data: $p$ unknown recipe scalars are identifiable only if
  $p\le N\bigl((m-1)+r-N\bigr)-Nk$, which is $120$ at the standard cell. A
  parametric schedule ($p=2$--$4$) sits far inside; a free per-step sequence
  ($p=T$) is identifiable only for $T\le120$ and not at $T=400$ — a defence
  that works by counting, not by obscurity. The certificate remains the
  recipe-free channel at the price $k<r-N$; Rev~9 \S3.6--3.8's curvature
  mixture remains the recipe-unknown approximation.
\item \textbf{Adam.} Its entrywise normalisation is not of the form in
  (A3): Theorem~\ref{thm:traj} does not hold, Corollary~\ref{cor:seed} does
  not reduce the seed to $X$, and the unknown seed is all of $A_0$ —
  $Nk+rn$ unknowns against $mr+rn$ equations. Measured$^\dagger$ (jobs
  466915, 467622, 479684): $D\rho$ has $536=Nk+rn$ columns at $n=32$ and
  $1632$ at $n=96$ against SGD's $88$ and $224$, is full column rank at the
  truth in every Adam cell, and is $\sim\!400\times$ worse conditioned at
  $n=96$ ($\kappa=8\times10^{5}$ against $2\times10^{3}$) with a much smaller
  basin. Adam keeps identifiability and loses the $rN$ reduction; and because
  it also breaks Proposition~\ref{prop:simplex}, it hands the attacker the
  $N$ dimensions the softmax head had removed (the prediction above).
\item \textbf{One Jacobian, three readings.} $D\rho$ at the truth is one
  object: its rank is Theorem~\ref{thm:suff}; its spectrum is the local
  resolution ellipsoid $\{h:h^{\!\top}(J^{\!\top}\Sigma^{-1}J)h<\tau\}$ of image
  changes the release cannot distinguish; its whitened form is the
  detectability $d^2=2\,\mathrm{KL}$ of the dataset-sensitivity programme,
  whose valley widths are one-dimensional slices of the ellipsoid
  (Figure~\ref{fig:spectrum}). Rendering its leading directions as image
  edits is the natural next object.
\item \textbf{Encoder quality: measured, and it ends the channel before the
  count does.} This was an open item; it is now the ladder in
  \S\ref{sec:measured}. The conditioning cost grows monotonically with
  accuracy, and at $97.9\%$ hypothesis (A4) fails outright, so the capacity
  law has nothing to say about that regime — not because the count is wrong
  but because the adapter stops recording the easy examples at all. What is
  open now is the boundary between the two: at what accuracy, or what margin
  distribution, does $\operatorname{rank}P_T$ begin to fall? That a
  deliberately high-margin fine-tuning set suppresses the release is now
  measured (\S\ref{sec:measured}, the confident batch); what is not is how
  the suppression degrades as the batch is diluted with harder examples. The
  measurement that settles it is the same example's imprint in a confident
  batch and then beside a badly fit one, which no cell yet contains; the
  question is about \eqref{eq:imprint}'s value through the shared trajectory,
  not about the feature coupling that was withdrawn.
\item \textbf{LoRA on more than one layer.} Adapting the first layer moves
  its own successors' inputs, so (A1) fails downstream: the closure of
  Theorem~\ref{thm:traj} does not hold, several adapters are released, and
  neither the count nor identifiability is derived. Measured as an empirical
  cell$^\dagger$ (job 626564, all three layers): with the seeds \emph{known}
  as an oracle ($112$ unknowns) the solver reaches residual
  $2.2\times10^{-8}$; with them unknown ($22{,}384$ unknowns),
  $7.7\times10^{-7}$ — both twenty orders off the floor, and the oracle arm
  nearly as bad as the attack. The obstruction at this budget is the
  three-layer unroll itself, not the unknown seeds, so the cell supports no
  identifiability statement in either direction; the fact that more numbers
  are released than are unknown ($38{,}352$ against $22{,}384$) is not
  evidence either way: a count of released entries is not a count of
  independent constraints, which is the whole content of \eqref{eq:cap}. It stays outside
  the theorems.
\item \textbf{What stops the attack today is neither line.} Inside the
  capacity region, LM from perturbed-truth starts recovers up to $86\%$ start
  error, with one budget-limited failure at $81\%$ (residual $8\times10^{-5}$
  at the iteration cap, still descending) bracketed by successes at $77\%$
  and $86\%$\textsuperscript{$\dagger$} (job 459111). From the $20$ starts an attacker can build
  from the release alone (random, span, certificate, span-anchor; jobs
  408560--63) it reaches the floor in \emph{none}: one run ends inside the
  $10^{-2}$ image tolerance at residual $9\times10^{-7}$ with its restart
  budget exhausted — by the classification above a search failure that
  happened to land near the truth — and the other nineteen end at residuals
  $5\times10^{-3}$ to $0.4$. The binding constraint is the initialiser — the
  slot the framework reserves for a learned or population prior — not the
  information in the release.
\end{enumerate}

\subsection{What to say to Gal (replaces the lead of Rev~9 \S16)}
\label{sec:gal}

Lead with two theorems, each with a sharp measurement. The first says how
many numbers per private example a released adapter can hold; the second
says which examples are in it at all. Together they make the adapter a
\emph{measurable channel}: two computable quantities, one of which the
defender can evaluate during their own training without simulating any
attack. The reconstructions are consequences, and the $49/49$ grid should
not lead — it sits inside the capacity region and measures existence, not
rate.

\begin{enumerate}\itemsep3pt
\item \emph{The chain the theorems stand on.} Theorem~\ref{thm:traj} writes
  the adapter's whole trajectory in closed form; Corollary~\ref{cor:seed}
  shows the random initialisation enters it through only $rN$ numbers, the
  rest being released verbatim. Projecting those away is Rev~9's
  certificate, $r-N$ equations per image (Theorem~\ref{thm:quot}); solving
  for them is the exact channel, and the difference between the two is one
  identifiable object (Corollary~\ref{cor:two}).

\item \textbf{Theorem I — capacity.} The released $B_T$ lies on a manifold
  of dimension $N((m-1)+r-N)$, the $-1$ being the softmax head
  (Proposition~\ref{prop:simplex}), while $A_TU_c$ pays exactly for the
  seed. Hence $k<m+r-N$ coordinates per example (Theorem~\ref{thm:cap}), and
  full column rank almost everywhere below it (Theorem~\ref{thm:suff}, with
  the closed-form $T=1$ witness). Measured sharp to one unit of $k$ at
  $N=4,8,14$, at five head widths, at three ranks on real MNIST digits, and
  on a \emph{trained} network with unseen test digits — where the line holds
  unmoved while the conditioning needed to reach it worsens a thousandfold.
  Both terms of the cap are attained by the measured block ranks.

\item \textbf{Theorem II — imprint.} The gradient in $B$ is a sum of
  rank-one terms, one per private example, so
  $B_T=\sum_iC_i$ with $\|C_i\|$ bounded by \emph{that example's own}
  accumulated error and no cross term (\eqref{eq:imprint}). Measured on
  $320$ example-batch pairs across four encoders: one proportional line over
  forty orders of magnitude, $971$ of $1120$ concordant pairs,
  and $\operatorname{rank}B_T$ equal to the number of examples recorded above
  the floor in all forty batches (Figure~\ref{fig:imprint}).

\item \emph{What the two theorems give the defender — this is the tool.}
  What leaks is what the model had to learn. A private batch its base model
  already fits leaves every imprint below $1.5\times10^{-24}$: the release is
  numerically empty. Conditioning worsens ten orders from random weights to
  $97.9\%$ accuracy at fixed architecture, and at $97.9\%$ hypothesis (A4)
  fails outright — $\operatorname{rank}P_T<N$ — so the channel closes
  \emph{below its own capacity line}, for reasons of the model rather than
  the count. The exposure meter is therefore the per-example accumulated
  residual during the defender's own fine-tuning: already computed in the
  course of training, and requiring no attack to interpret.

\item \emph{What they give the attacker.} $\operatorname{rank}B_T$ is read
  off the release with no private knowledge and says how many examples were
  recorded; on the $97.9\%$ model that is essentially one, in $9$ of $10$
  ordinary batches. Private data from a different distribution is recorded
  in full — scanned digits misclassified $4/8$ by that model give every
  imprint order one and full rank — — foreign data tends to be
  recorded in full because a model tends to get it wrong.

\item \emph{Why a prior is the missing piece, precisely.} The theorems pin
  the chart \emph{coordinates}; the attack returns $\psi(\hat w)$. At one
  fixed $k$, three charts each recover their own representable image to
  $10^{-14}$ and differ by thirteen orders against the real digit. So
  reconstruction factorises as capacity $\times$ chart quality: this work
  fixes the first exactly, and a prior supplies the second. Richness is free
  in conditioning terms, so the instruction is to spend the coordinates on
  the best chart available.

\item \emph{What stops the attack today.} At $N=8$, none of $20$
  attacker-buildable starts reaches the residual floor, while from perturbed
  truth the solver recovers from $86\%$ away — so the obstruction is the
  initialiser, not the information. The hypothesis under test is that this
  obstruction scales with the number of \emph{recorded} examples rather than
  with $N$, which would make it the model's choice: on a strong model the
  release is often dominated by one example, and one example is a far smaller
  search. That arm is running and unverified. It also carries a tension worth
  owning rather than hiding: model confidence \emph{reduces} what is
  recorded, which favours the defender, while making the little that remains
  a smaller problem to solve, which favours the attacker. Both are true, and
  which dominates is not yet measured.

\item \emph{Open, and said in the same breath.} Global uniqueness below the
  line; how far the release-consistent set extends past it (one direction of
  forty was walked, and one cell was still rising); the accuracy at which
  (A4) begins to fail; LoRA on more than one layer, where the closure fails
  and the measured cells are dominated by the unrolled optimisation rather
  than by identifiability; Adam, which keeps identifiability, loses the seed
  reduction, destroys the certificate and is predicted to sit one unit higher
  on the line; and the schedule family under a weights-only release.
\end{enumerate}
