\section{The exact channel: from the quotient to the full release}
\label{sec:exact-channel}

A released LoRA adapter is a deterministic function of the private data, the
public recipe and the random initialisation. This section writes that
function out (Theorem~\ref{thm:traj}), shows that the initialisation enters
it only through $rN$ numbers (Corollary~\ref{cor:seed}), and then reads the
same function in two ways: projecting the initialisation away gives Rev~9's
certificate with $r-N$ equations per image (Theorem~\ref{thm:quot}); keeping
it and solving for the $rN$ numbers gives $m-1+r-N$ equations per image
(Theorems~\ref{thm:cap}--\ref{thm:suff}). The difference between the two is
one identifiable object — the coefficient matrix of the released $B_T$ —
and Corollary~\ref{cor:two} states it. Section~\ref{sec:measured} reports
what was measured, on the synthetic test-bed and on real digits, and shows
that the boundary is a statement about the coordinates the attacker searches
in, not about the images.

%----------------------------------------------------------------------------
\subsection{Objects, hypotheses, and the gradient}
\label{sec:objects}

\begin{definition}[the layer, the data, the adapter]\label{def:setup}
Fix integers $m,n,r,N$ with $N\le r\le n$. A frozen weight
$W_0\in\mathbb R^{m\times n}$ receives the representations
$H=[h_1,\dots,h_N]\in\mathbb R^{n\times N}$, $\operatorname{rank}H=N$, of the $N$ private examples;
$E\in\{0,1\}^{m\times N}$ holds their one-hot labels. A LoRA adapter is a
pair $A\in\mathbb R^{r\times n}$, $B\in\mathbb R^{m\times r}$ with a fixed
scale $s>0$; the layer computes logits and loss
\[
  Z(A,B):=(W_0+sBA)H\in\mathbb R^{m\times N},\qquad
  L(A,B):=\sum_{i=1}^N \ell\bigl(z_i,e_i\bigr),
\]
$\ell$ the softmax cross-entropy. Write $D(Z):=\operatorname{softmax}(Z)-E$
(column-wise softmax), $G:=H^{\!\top}H$, and for any $A_0$,
$Q:=(A_0H)^{\!\top}(A_0H)$; both Grams are $N\times N$.
\end{definition}

\begin{lemma}[gradients]\label{lem:grad}
$\nabla_B L=s\,D\,H^{\!\top}A^{\!\top}$ and $\nabla_A L=s\,B^{\!\top}D\,H^{\!\top}$,
with $D=D(Z(A,B))$.
\end{lemma}
\begin{proof}
$dL=\langle D,dZ\rangle=\langle D,\;s\,dB\,AH+s\,B\,dA\,H\rangle
 =\langle s\,D H^{\!\top}A^{\!\top},dB\rangle+\langle s\,B^{\!\top}DH^{\!\top},dA\rangle$.
\end{proof}

\paragraph{Hypotheses.} Each is a formula; each theorem lists the ones it uses.
\begin{itemize}\itemsep2pt
\item[(A1)] \emph{Fixed inputs.} The same $H$ enters the adapter at every
  step: no augmentation, no dropout, no trainable block upstream.
\item[(A2)] \emph{Initialisation.} $B_0=0$; $A_0$ is drawn from a law with a
  density on $\mathbb R^{r\times n}$ (so, with $r\le n$,
  $\operatorname{rank}A_0=r$ a.s.).
\item[(A3)] \emph{Recipe.} Plain gradient descent with step $\eta$ for $T$
  steps,
  \[
    B_{t+1}=B_t-\eta\,\nabla_BL(A_t,B_t),\qquad
    A_{t+1}=A_t-\eta\,\nabla_AL(A_t,B_t),\qquad t=0,\dots,T-1,
  \]
  and the release is $(A_T,B_T)$.\footnote{Everything below survives any
  update that is a scalar linear filter of past gradients and iterates
  (momentum, scalar weight decay — which multiplies $A_0$ by a scalar $c_t$
  in Theorem~\ref{thm:traj}, and nothing else changes) and any matrix map of
  the form $M\mapsto p(MM^{\!\top})M=M\,p(M^{\!\top}M)$ applied to the
  gradient. Adam's entrywise normalisation is not of this form; \S\ref{sec:opens}, item~3.}
\item[(A4)] $\operatorname{rank}P_T=N$ for the matrix $P_T$ of
  Theorem~\ref{thm:traj}. (Needs $m\ge N$ and $N$ linearly independent
  accumulated error trajectories; it fails for duplicated or already-fit
  examples.)
\item[(A5)] $r\ge N+1$.
\item[(A6)] \emph{Canonical basis.} For any full-column-rank $H$, $U(H)$
  denotes the $Q$-factor of the QR decomposition $H=UR_H$ with
  $\operatorname{diag}R_H>0$. This makes $H\mapsto U(H)$ real-analytic; a
  QR without the sign convention is discontinuous at every sign change of a
  diagonal entry.
\end{itemize}

\paragraph{The two coordinates of the seed.} Let $U=U(H)$ and let
$V\in\mathbb R^{n\times(n-N)}$ be an orthonormal basis of $\operatorname{col}(H)^\perp$.
Define
\begin{equation}\label{eq:XY}
\begin{aligned}
  X&:=A_0U\in\mathbb R^{r\times N},\qquad Y:=A_0V\in\mathbb R^{r\times(n-N)},\\
  \text{so that}\qquad A_0&=XU^{\!\top}+YV^{\!\top},\qquad A_0H=XR_H,\qquad
  Q=R_H^{\!\top}X^{\!\top}XR_H .
\end{aligned}
\end{equation}

\paragraph{The private data as a manifold.} Each private image is
$x_i=\psi(w_i)$ for a generator $\psi:\mathbb R^k\to\mathbb R^{d}$, and its
representation is $h_i=\varphi(x_i)$; write $H(w)$, $R_H(w)$, $U_c(w):=U(H(w))$
for a candidate $w=(w_1,\dots,w_N)\in\mathbb R^{Nk}$. Both $\varphi$ and
$\psi$ are real-analytic ($\tanh$ networks in the test-bed).

\subsection{The trajectory, written out}
\label{sec:trajectory}

\begin{theorem}[closure of the factor dynamics]\label{thm:traj}
Assume (A1)--(A3). Then for every $t\le T$
\begin{equation}\label{eq:closure}
  B_t=P_t\,(A_0H)^{\!\top},\qquad A_t=A_0\bigl(I_n+H\,M_t\,H^{\!\top}\bigr),
\end{equation}
where $P_t\in\mathbb R^{m\times N}$, $M_t\in\mathbb R^{N\times N}$ are given
by $P_0=0$, $M_0=0$ and
\begin{equation}\label{eq:rec}
  D_t=D\bigl(W_0H+s\,P_tQ(I_N+M_tG)\bigr),\qquad
  P_{t+1}=P_t-\eta s\,D_t\,(I_N+GM_t^{\!\top}),\qquad
  M_{t+1}=M_t-\eta s\,P_t^{\!\top}D_t .
\end{equation}
\end{theorem}
\begin{proof}
Induction on $t$; $t=0$ is (A2). Assume \eqref{eq:closure} at $t$. Then
$B_tA_tH=P_t(A_0H)^{\!\top}A_0(H+HM_tG)=P_t\,Q\,(I+M_tG)$, which gives $D_t$.
By Lemma~\ref{lem:grad},
\begin{align*}
  \nabla_BL&=sD_tH^{\!\top}A_t^{\!\top}=sD_t\bigl(H^{\!\top}+G^{\!\top}M_t^{\!\top}H^{\!\top}\bigr)A_0^{\!\top}
  =sD_t(I+GM_t^{\!\top})(A_0H)^{\!\top}\qquad(G^{\!\top}=G),\\
  \nabla_AL&=sB_t^{\!\top}D_tH^{\!\top}=A_0H\,(sP_t^{\!\top}D_t)\,H^{\!\top},
\end{align*}
so the update of (A3) keeps both shapes and produces \eqref{eq:rec}.
\end{proof}

\begin{corollary}[where the seed enters]\label{cor:seed}
Assume (A1)--(A3). The coefficient trajectory $(P_t,M_t,D_t)_{t\le T}$ is a
function of $W_0H$, $G$, $Q$, $E$ and $(\eta,s,T)$ only; the initialisation
$A_0$ enters it through $Q=R_H^{\!\top}X^{\!\top}XR_H$ and nowhere else. The release is
\begin{equation}\label{eq:release}
  B_T=P_T\,R_H^{\!\top}X^{\!\top},\qquad
  A_T=X\,\Omega\,U^{\!\top}+Y\,V^{\!\top},\qquad
  \Omega:=I_N+R_HM_TR_H^{\!\top},
\end{equation}
so $A_TU=X\Omega$ and $A_TV=Y$: the block $Y$ of the seed is released
unchanged, and $B_T$, $A_TU$ depend on the seed only through $X$. In
particular, for every orthogonal $O\in\mathrm O(r)$ the seed $OA_0$ produces
the same $(P_t,M_t,D_t)_{t\le T}$ and $\Omega$, and the release
$(OA_T,\;B_TO^{\!\top})$: the coefficient trajectory is invariant under the
LoRA gauge $BA=(BO^{\!\top})(OA)$, and the release transforms covariantly.
\end{corollary}
\begin{proof}
Substitute \eqref{eq:XY} into \eqref{eq:closure}--\eqref{eq:rec}:
$A_T=A_0+A_0HM_TH^{\!\top}=XU^{\!\top}+YV^{\!\top}+XR_HM_TR_H^{\!\top}U^{\!\top}$.
For the gauge, $Q$ is unchanged by $X\mapsto OX$, so \eqref{eq:rec} is
unchanged, and \eqref{eq:release} transforms as stated.
\end{proof}
The gauge statement was tested rather than assumed (job 487882 — a log,
not a jsonl row; the third shape ran at $\eta=0.03$):
at three shapes — $(n,k,N,T)$ equal to $(96,12,8,400)$, to $(32,6,4,200)$
and to $(96,10,12,150)$ — the relative discrepancies
\[
  \frac{\|A_T'-OA_T\|}{\|A_T\|},\qquad \frac{\|B_T'-B_TO^{\!\top}\|}{\|B_T\|},\qquad
  \frac{\|\Omega'-\Omega\|}{\|\Omega\|}
\]
are all between $5.8\times10^{-16}$ and $2.1\times10^{-15}$. The coefficients see only $Q$; the release sees $X$; the
gauge moves both, so nothing is lost by the attacker and the invariance is a
statement about the trajectory, not about identifiability — the release
still determines $X$, and $\rho$ is not degenerate.

\subsection{Two readings of the same release}
\label{sec:read}

\subsubsection*{First reading: project the seed away}

\begin{theorem}[the quotient certificate]\label{thm:quot}
Assume (A1)--(A5). Then $\operatorname{row}(B_T)=\operatorname{col}(A_0H)$
as subspaces of $\mathbb R^r$, and with $\Pi_S$ the orthogonal projector of
$\mathbb R^r$ onto $S^\perp$,
\begin{equation}\label{eq:C}
\begin{aligned}
  C&:=\Pi_{\operatorname{row}(B_T)}A_T=\Pi_{\operatorname{col}(A_0H)}A_0,\\
  CH&=0,\qquad \operatorname{rank}C=r-N,\qquad
  \ker C=\operatorname{col}(H)+\ker A_0 .
\end{aligned}
\end{equation}
\end{theorem}
\begin{proof}
By \eqref{eq:closure} the rows of $B_T$ are the $m$ combinations of the $N$
columns of $A_0H$ with coefficient rows $P_T$; under (A4) they span
$\operatorname{col}(A_0H)$. Projecting \eqref{eq:closure} for $A_T$:
$\Pi_{\operatorname{col}(A_0H)}A_T=\Pi_{\operatorname{col}(A_0H)}A_0+\Pi_{\operatorname{col}(A_0H)}(A_0H)\,M_TH^{\!\top}$,
and the second term is zero. Hence $CH=\Pi_{\operatorname{col}(A_0H)}(A_0H)=0$;
the image of $C$ is $\Pi_{\operatorname{col}(A_0H)}(\operatorname{col}A_0)=\operatorname{col}(A_0H)^\perp$
of dimension $r-N$ by (A2), (A5); and $Cx=0$ iff $A_0x\in\operatorname{col}(A_0H)$
iff $x\in\operatorname{col}(H)+\ker A_0$.
\end{proof}

What the first reading uses is visible in \eqref{eq:C}: of $B_T$ only the
subspace $\operatorname{row}(B_T)$ — a point of the Grassmannian
$\operatorname{Gr}(N,r)$, $N(r-N)$ numbers — and of $A_T$ only its projection
off that subspace. The coefficient matrix $P_T$ is removed by the projection.
The equations it imposes on candidate data are $CH(w)=0$: $C$ has rank $r-N$,
so these are $(r-N)N$ scalar equations, $r-N$ per image; Rev~9 Thm~3 and
Prop~4 show they pin $w$ locally iff $k<r-N$. No recipe and no solver are
used.

\subsubsection*{Second reading: keep the seed and solve for it}

\begin{definition}[the attacker's map]\label{def:rho}
The attacker knows $W_0$, $\varphi$, $\psi$, $E$, the recipe $(\eta,s,T)$
and the release $(A_T,B_T)$; the unknowns are $w\in\mathbb R^{Nk}$ and, by
Corollary~\ref{cor:seed}, $X\in\mathbb R^{r\times N}$.\footnote{$Y$ is not an
unknown: it enters neither $B_T$ nor $A_TU$, and \eqref{eq:release} shows it
passes to the release unchanged — in the true basis $V$, which the attacker
does not know and does not need.}
For a candidate $(w,X)$ let $P_T(w,X)$, $M_T(w,X)$ be the output of
\eqref{eq:rec} run with $H=H(w)$, $G=G(w)$, $Q=R_H(w)^{\!\top}X^{\!\top}XR_H(w)$,
and $\Omega(w,X)=I+R_H(w)M_TR_H(w)^{\!\top}$. The residual map is
\begin{equation}\label{eq:rho}
  \rho(w,X):=\Bigl(\;\frac{P_T(w,X)\,R_H(w)^{\!\top}X^{\!\top}-B_T}{\|B_T\|}\;,\;\;
  \frac{X\,\Omega(w,X)-A_T\,U_c(w)}{\|A_T\|}\;\Bigr)
  \in\mathbb R^{m\times r}\times\mathbb R^{r\times N}.
\end{equation}
The attack is $\min_{w,X}\|\rho(w,X)\|^2$. The true $(w^\star,X^\star)$ is a
zero of $\rho$; the second reading asks whether it is an isolated zero.
\end{definition}

The target $A_TU_c(w)$ in the second block moves with the candidate, because
the attacker does not know $U$ and must form $U_c(w)$; its $w$-derivative
$A_T\,\partial U_c/\partial w$ is part of $D\rho$.\footnote{Writing the target
as the fixed $A_TU$ instead drops this term and undercounts the
$w$-dependence; the budget one then obtains is refuted by the measured full
column rank at $k=24>m$ (Example~\ref{ex:T1}). The normalisers $\|B_T\|$,
$\|A_T\|$ are fixed numbers — deliberately not $\|A_TU_c(w)\|$, which the
optimiser could shrink — and scale the spectrum of $D\rho$, not its rank.}

\begin{proposition}[the softmax head constrains $B_T$]\label{prop:simplex}
Assume (A2), (A3). Then $\mathbf 1^{\!\top}P_t=0$ and hence
$\mathbf 1^{\!\top}B_t=0$ for every $t$.
\end{proposition}
\begin{proof}
Every column of $\operatorname{softmax}(Z)$ and of $E$ sums to one, so
$\mathbf 1^{\!\top}D(Z)=0$. In \eqref{eq:rec},
$\mathbf 1^{\!\top}P_{t+1}=\mathbf 1^{\!\top}P_t-\eta s\,(\mathbf 1^{\!\top}D_t)(I+GM_t^{\!\top})=\mathbf 1^{\!\top}P_t$,
and $P_0=0$. Then $\mathbf 1^{\!\top}B_t=\mathbf 1^{\!\top}P_t(A_0H)^{\!\top}=0$.
\end{proof}

\begin{theorem}[capacity: the count is necessary]\label{thm:cap}
Assume (A1)--(A3), (A6) and $N\le m-1$. Let
\[
  \mathcal B:=\{B\in\mathbb R^{m\times r}:\ \operatorname{rank}B=N,\ \mathbf 1^{\!\top}B=0\},
  \qquad \dim\mathcal B=N(r-N)+N(m-1)=N\bigl((m-1)+r-N\bigr).
\]
Then, on each connected component of $\{(w,X):\operatorname{rank}H(w)=N\}$
that contains a point with $\operatorname{rank}P_T(w,X)=N$, the maximal rank
of $D\rho$ is at most $N\bigl((m-1)+r-N\bigr)+rN$. Consequently, if
\begin{equation}\label{eq:cap}
  k\;\ge\;m+r-N\qquad\bigl(\text{equivalently } Nk+rN>\dim\mathcal B+rN\bigr),
\end{equation}
then for almost every true $(w^\star,X^\star)$ the set of candidates that
reproduce the release exactly, $\rho^{-1}(0)$, contains a submanifold through
$(w^\star,X^\star)$ of dimension at least $N\bigl(k-(m-1+r-N)\bigr)\ge N$.
\end{theorem}
\begin{proof}
\emph{Dimension of $\mathcal B$.} Let $\Theta\in\mathbb R^{m\times(m-1)}$ be
an orthonormal basis of $\mathbf 1^\perp$; $B\mapsto\Theta^{\!\top}B$ is a
rank-preserving isometry of $\{\mathbf 1^{\!\top}B=0\}$ onto
$\mathbb R^{(m-1)\times r}$. A rank-$N$ matrix there is determined by its row
space $S\in\operatorname{Gr}(N,r)$ and its coefficient matrix in a basis of
$S$, an element of $\mathbb R^{(m-1)\times N}$ of full rank; the dimensions
are $N(r-N)$ and $N(m-1)$.

\emph{Rank bound.} On the open set $\mathcal O$ where
$\operatorname{rank}P_T(w,X)=N$, Theorem~\ref{thm:traj} and
Proposition~\ref{prop:simplex} put the first block of $\rho$, plus the
constant $B_T/\|B_T\|$, in $\mathcal B/\|B_T\|$, a manifold of dimension
$\dim\mathcal B$, so the derivative of that block has rank at most
$\dim\mathcal B$ there; the second block has $rN$ rows; so
$\operatorname{rank}D\rho\le\dim\mathcal B+rN$ on $\mathcal O$. On a
connected component the maximal rank of $D\rho$ is attained on an open dense
set (the $\rho$ of \S\ref{sec:objects} is real-analytic there), which meets
the non-empty open set $\mathcal O$; hence the maximal rank on that component
is at most $\dim\mathcal B+rN$. (Under (A4) the truth lies in $\mathcal O$.)

\emph{Fibre.} On the component, the set where $\operatorname{rank}D\rho$ is
below its maximum is a proper analytic subvariety — closed, with empty
interior and Lebesgue measure zero. Off it the rank is locally constant and
the constant-rank theorem makes $\rho^{-1}(\rho(p))$ a submanifold through
$p$ of dimension $(Nk+rN)-\operatorname{rank}D\rho(p)\ge Nk+rN-\dim\mathcal
B-rN$. This holds at Lebesgue-almost every $(w^\star,X^\star)$; for the seed
that is a.e.\ under (A2), and for the data it is a.e.\ in the chart
coordinates, and $\rho(w^\star,X^\star)=0$ at the truth.
\end{proof}

\begin{remark}\label{rem:distance}
Theorem~\ref{thm:cap} says that past the line the truth is not an isolated
zero of $\rho$: a continuum of release-consistent candidates passes through
it. It says nothing about how far that continuum extends in image space:
non-isolation is not distance. And below the line, identifiability is not
fidelity: the coordinates $w$ are pinned, and the image returned, $\psi(\hat
w)$, is only as good as the chart $\psi$. Section~\ref{sec:measured} measures
both.
\footnote{Rank deficiency of $D\rho$ \emph{at} the truth alone would not give
non-isolation ($f(x)=x^2$ has $f'(0)=0$ and an isolated zero); the proof uses
that the rank bound holds at every candidate, and concludes only almost
everywhere.}
\end{remark}

\begin{theorem}[generic local identifiability below the line]\label{thm:suff}
Assume (A1)--(A3), (A6) and $k\le(m-1)+r-N$. Let
$\mathcal D:=\{(w,X):\operatorname{rank}H(w)=N\}$, an open set, and let
$\mathcal D_0$ be any of its connected components. Exactly one of the
following holds: $D\rho$ has full column rank $Nk+rN$ at almost every point
of $\mathcal D_0$, or at no point of $\mathcal D_0$. In the first case every
true $(w^\star,X^\star)\in\mathcal D_0$ outside a null set is an isolated
zero of $\rho$; so one point of full column rank in $\mathcal D_0$ gives
isolation for almost every seed (under (A2)) and Lebesgue-almost every chart
coordinate in $\mathcal D_0$.
\end{theorem}
\begin{proof}
The $(Nk+rN)$-minors of $D\rho$ are real-analytic on the connected
real-analytic manifold $\mathcal D_0$; their common zero set is an analytic
subvariety of $\mathcal D_0$, hence either all of $\mathcal D_0$ or of
measure zero (identity theorem). At a point of full column rank the
inverse-function theorem makes the zero isolated (Rev~9 Prop~8).
\end{proof}

The statement is per component because a real-analytic subvariety of
codimension one can disconnect: $\mathcal D$ is connected — and one witness
then serves globally — exactly when the rank-deficiency locus
$\{w:\operatorname{rank}H(w)<N\}$ has real codimension at least two, which is
the generic situation for $n\ge N+1$ but is not verified here. The theorem
reduces genericity to one witness of full column rank in the truth's
component, and at $T=1$ the map is explicit enough to exhibit one.

\begin{example}[$T=1$, closed form]\label{ex:T1}
From \eqref{eq:rec}, $P_1=-\eta s\,D_0$ and $M_1=0$, with
$D_0=D(W_0H(w))$ depending on $w$ only through the base model's logits.
Hence $A_1=A_0$, $\Omega=I$, and \eqref{eq:rho} reads (unnormalised)
\[
  \rho_{T=1}(w,X)=\bigl(-\eta s\,D_0(w)\,R_H(w)^{\!\top}X^{\!\top}-B_T,\;\;X-A_TU_c(w)\bigr).
\]
The second block gives $X=A_TU_c(w)$ exactly, and substituting,
$U_c(w)R_H(w)=H(w)$,
\begin{equation}\label{eq:T1}
  B_T\;=\;-\eta s\;D_0(w)\,H(w)^{\!\top}A_T^{\!\top}:
\end{equation}
an equation in $w$ alone whose right side ranges over $\mathcal B$ — the
$N((m-1)+r-N)$ constraints of Theorem~\ref{thm:cap} on the $Nk$ unknowns,
with no unrolling and the seed eliminated in closed form; the budget obtained
by eliminating $X$ here equals the one obtained by counting $X$ as an unknown
in Theorem~\ref{thm:cap}, as it must.\footnote{Eliminating
$X$ does not lower the count: the count is the dimension of the manifold that
$B_T$ lives on, not a tally of which block ``pays for'' which unknown. That is
why the intuition ``at $T=1$ the $A$-block hands $X$ over for free, so
$k\le m$'' is wrong; it is refuted by measurement at $k=24>m=20$.}
Measured$^\dagger$ (job 479587, $N=8$): $D\rho_{T=1}$ has full column rank
at $k=16,20,24$ with $\sigma_{\min}=7.26\times10^{-4},\,1.21\times10^{-4},\,3.15\times10^{-5}$,
identical to three figures with the $T=2$ values, and collapses at $k=28=m+r-N$
($3.3\times10^{-19}$). These are witnesses of Theorem~\ref{thm:suff} at
$T=1$ for the charts $\psi_k$ with $k=16,20,24$ — each $k$ is its own chart
and its own dichotomy. For the working point $T=400$ the witnesses are
direct: $D\rho$ at the truth has full column rank at every $k\le27$ (jobs
467914, 479587); the $T=1$/$T=2$ agreement to three figures is a consistency
check on the closed form, not the witness for $T>1$.
\end{example}

\subsubsection*{The two readings side by side}

\begin{corollary}[the two channels]\label{cor:two}
Under (A1)--(A3) the release $(A_T,B_T)$ has the form \eqref{eq:release}.
Reading it by Theorem~\ref{thm:quot} uses $\operatorname{row}(B_T)\in\operatorname{Gr}(N,r)$
and $\Pi_{\operatorname{row}(B_T)}A_T$, imposes $N(r-N)$ equations on $Nk$
unknowns, and needs (A4), (A5) but no recipe. Reading it by
Definition~\ref{def:rho} uses all of $B_T\in\mathcal B$ and $A_TU_c$,
imposes $N(r-N)+N(m-1)+rN$ equations on $Nk+rN$ unknowns, and needs (A6),
a solver, and the recipe — which \S\ref{sec:measured} shows can be checked
against the floor, fitted jointly, or probed. The $rN$ equations of the second block pay exactly
for the $rN$ unknowns of $X$; the difference between the channels is the
$N(m-1)$ coefficients of $P_T$ in a basis of $\operatorname{row}(B_T)$, which
the projection in \eqref{eq:C} removes and which \eqref{eq:rho} keeps:
\[
  \text{certificate: } k<r-N\qquad\qquad \text{exact: } k<m+r-N .
\]
\end{corollary}
\begin{proof}
Assemble Theorem~\ref{thm:quot}, the proof of Theorem~\ref{thm:cap}, and
Theorem~\ref{thm:suff}.
\end{proof}

\begin{tabular}{@{}>{\raggedright\arraybackslash}p{0.2\linewidth}>{\raggedright\arraybackslash}p{0.27\linewidth}>{\raggedright\arraybackslash}p{0.13\linewidth}>{\raggedright\arraybackslash}p{0.32\linewidth}@{}}\toprule

certificate (Thm~\ref{thm:quot}) & $\operatorname{row}(B_T)$; $\Pi_{\operatorname{row}(B_T)}A_T$ & $r-N$ & (A1)--(A5)\\
exact (Thms~\ref{thm:cap}, \ref{thm:suff}) & all of $B_T$; $A_TU_c(w)$ & $(m-1)+r-N$ & (A1)--(A3), (A6); the recipe, checked or fitted (\S\ref{sec:measured}); a solver and a start\\ \bottomrule
\end{tabular}
\end{center}
Both capacities are statements about the chart coordinates $w\in\mathbb R^{Nk}$
of Definition~\ref{def:setup}: the unknown is $w$, the theorems pin $w$, and
the attack returns $\psi(\hat w)$. Nothing in them constrains the chart
$\psi$, which is the attacker's to choose.

\subsection{What was measured}
\label{sec:measured}

Test-bed: FP64, $m=20$, $r=16$, $n=96$, $\tanh$ generator $\psi$ to a $64$-d
image, $\tanh$ encoder $\varphi$, softmax head, plain SGD; the attack is
Levenberg--Marquardt on \eqref{eq:rho} with an autograd Jacobian. Every run is
classified by two numbers, the final residual $\|\rho\|^2$ and the worst
image error: \emph{recovered} — residual at the $10^{-30}$ floor and error at
machine precision; \emph{alias} — residual at the floor at a point other than
the truth, with $\sigma_{\min}(D\rho)$ collapsed; \emph{search failure} —
residual above the floor. Two scopings hold for every capacity, alias and
recipe cell below: each is one seed, and each is solved from a start at the
truth plus $10\%$ latent noise, which an attacker cannot build (the starts an
attacker can build are the last item of \S\ref{sec:opens}). The residual and
$\sigma_{\min}$ are attacker-computable; the image-error column is not.

the search space is the attacker's to choose — which is the counting
argument handing the generative-prior direction of Rev~9 its role: a prior

grows with distance past the line at every $N$; $9$ of the $13$ cells have
every image inside the study's $10^{-2}$ tolerance, and the four outside it
are the cells furthest past the line. On real digits (job 568095) the same
rows are worse: the past-line aliases at $r=16$ and $r=32$ — residual at the
floor — sit at $2.3$--$6.8\times10^{-2}$, again growing with distance

step along the right singular vector of $\sigma_{\min}(D\rho)$, retract onto

This is Theorem~\ref{thm:cap} and Theorem~\ref{thm:suff} as a single
experiment: the set of exact reproductions is a point below the line and a
walkable manifold at and past it. Each walk follows one null direction
(of $14$ and $40$), so the errors reached bound the extent from below; the
at-line walk had not stopped rising. On the paths measured the alternative
solutions lie within $1.4$--$3.8\%$ of the private images, in a chart that
contains them; $3.8\%$ is a visible degradation, and whether the images
remain recognisable was not assessed. Hence $k<m+r-N$ is the boundary of
\emph{exact} identifiability of the chart coordinates, and it is not a
leakage bound: below it the image reached is only as good as the chart, and
past it the release-consistent alternative lies within a few percent of the
truth on the evidence measured — sub-percent to $1.7\%$ in the synthetic
chart that contains the truth, $2$--$7\%$ on real digits — with the error
growing with distance past the line in both. Whether the alternatives stay
recognisable further out is not established, and the trend with distance
does not favour it.

\paragraph{The recipe need not be handed to the attacker.} Definition~\ref{def:rho}
takes $(\eta,s,T)$ and the labels as known. Four measurements$^\dagger$ at
$k=12$, $N=8$, $T=400$, $\eta=0.01$ (job 484255) show that they can be
checked, fitted, and in part read off instead.

\emph{(R1) None of seven wrong recipes reached the floor.} With the control first:
\begin{center}\small
\begin{tabular}{@{}lll@{}}\toprule
hypothesis simulated & residual & image error\\ \midrule
correct (control) & $5.3\times10^{-31}$, at the floor & $2.8\times10^{-15}$\\
$T+1$ step ($401$ of $400$) & $6.0\times10^{-8}$ & $4.1\times10^{-3}$\\
$\eta\times1.01$ & $9.1\times10^{-7}$ & $1.8\times10^{-2}$\\
$T+25\%$ \;/\; $T-25\%$ & $1.6\times10^{-4}$ \;/\; $2.9\times10^{-3}$ & $0.30$ \;/\; $0.40$\\
$\eta\times2$ \;/\; $\eta/2$ & $1.6\times10^{-3}$ \;/\; $3.4\times10^{-2}$ & $0.54$ \;/\; $0.46$\\
wrong optimiser (SGD release inverted as Adam) & $4.9$ & $0.26$\\ \bottomrule
\end{tabular}
\end{center}
The true recipe is the unique floor-reacher, separated from the nearest
wrong hypothesis — one step in four hundred — by twenty-three orders. The
residual is \emph{not} monotone in the size of the recipe error
($\eta\times2$ ends lower than $\eta/2$; $T+25\%$ is eighteen times lower
than $T-25\%$), so it is a detector, not a graded objective; it is computed
from the release and the candidate alone, never from the private images.
Seven hypotheses, one cell, one seed.

\emph{(R2) The learning rate can be fitted jointly with the data.} Carrying
$\eta$ as a free unknown beside $(w,X)$ from starts at $1\times$, $2\times$,
$0.5\times$ the truth, the fit returns $\eta$ to relative error
$5.2\times10^{-16}$ and the images to $3\times10^{-15}$ at the floor in all
three cases — one extra unknown against a slack of $120$ in \eqref{eq:cap}.

\emph{(R3) $\eta T$ is not a degeneracy of discrete SGD, and labels are
identifiable.} Holding $\eta T=4$ and varying the split, $(T,\eta)=(100,0.04)$,
$(200,0.02)$, $(800,0.005)$ stop at residuals $3.1\times10^{-7}$,
$3.5\times10^{-8}$, $8.8\times10^{-9}$ while the true $(400,0.01)$ reaches
$6.4\times10^{-31}$: the product does not determine the release, so $\eta$ and
$T$ are separately identifiable — the degeneracy is a gradient-flow statement
that finite steps break, and this is the weakest separation in the study
(twenty-two orders), as it should be if the degeneracy is approached in the
small-step limit. The wrong splits still return images within
$0.1$--$0.7\%$. Labels: the true assignment reaches the floor
($7.2\times10^{-31}$); swapping two examples ($2.5\times10^{-3}$) or shifting
all of them ($7.5\times10^{-6}$) does not.

\emph{(R4) A decaying schedule leaks the step count that a constant one
hides.} The \emph{probe}: the attacker continues training the released
adapter on a batch of their own choosing, $H_p=\varphi(\psi(W_p))$ with
labels they choose, so every regressor is computable from $(A_T,B_T)$ and the
public $W_0$ and no private data enters. One step under decoupled weight
decay is $\Delta B=-\eta\nabla_BL-\eta\,\mathrm{wd}\,B_T$, linear in
$(\eta,\eta\,\mathrm{wd})$: the two-parameter least squares returns $\eta$ to
$5\times10^{-16}$ and $\mathrm{wd}=10^{-3},10^{-2}$ to $10^{-11}$ relative
(job 487290), and
$\cos(\Delta B,-\nabla_BL)=1$ under SGD against $0.43$ under Adam separates
the optimiser family (job 485912). The same probe returns the per-step $\eta$ at the continuation steps
to $3\times10^{-14}$ or better (jobs 487290, 488314). Under a constant rate that is all it
returns, and $T$ stays hidden. Under a cosine schedule of known family, a
global fit of $(\text{base},T_{\max},T)$ to six probe steps (eliminate the
base by ratios, grid the two integers, solve the base in closed form) returns
all three exactly — $T=400$, $T_{\max}=1000$, base $0.0100$ — and identically
from windows of $20$, $60$ and $150$ steps (the probed values are job
488314's). A schedule encodes position on its own curve.\footnote{A gradient
fit of the same three parameters had landed $30$--$40\%$ off and was
retracted; it was a local minimum of a multimodal objective, not a
degeneracy. The global fit is a re-analysis of the saved probe values,
reproduced independently by an auditor from the same rows; the fitting
script and its result row are not yet committed, so this number does not
yet carry a dagger.} The family is assumed known;
selecting it is a model-selection problem not run here.

\subsection{What is open}
\label{sec:opens}

\begin{enumerate}\itemsep3pt
\item \textbf{Local, not global.} Theorem~\ref{thm:suff} isolates the truth
  in a neighbourhood; injectivity of the release map on the whole manifold
  is open, as it is for the certificate in Rev~9. No alias has appeared below
  the line$^\dagger$; that is evidence. The continuation walks one null
  direction per cell — a multi-direction version, which would say whether the
  extent is direction-dependent, has not been run — and the at-line cell was
  still rising when its budget ended. ``Recognisable'' is measured to $3.8\%$
  and no further. And since the boundary is about the chart, the question
  that matters for privacy is which chart an attacker can build
  \emph{without} the private images — a population prior — and how far below
  the line it lands; that is the generative-prior direction, now with a
  precise specification.
\item \textbf{The recipe family.} What \S\ref{sec:measured} leaves open is
  the \emph{family}: the schedule fit assumes cosine, and choosing among
  families is model selection, which the floor test of (R1) would presumably
  rank but which has not been run. The count \eqref{eq:cap} bounds the recipe
  as it bounds the data: $p$ unknown recipe scalars are identifiable only if
  $p\le N\bigl((m-1)+r-N\bigr)-Nk$, which is $120$ at the standard cell. A
  parametric schedule ($p=2$--$4$) sits far inside; a free per-step sequence
  ($p=T$) is identifiable only for $T\le120$ and not at $T=400$ — a defence
  that works by counting, not by obscurity. The certificate remains the
  recipe-free channel at the price $k<r-N$; Rev~9 \S3.6--3.8's curvature
  mixture remains the recipe-unknown approximation.
\item \textbf{Adam.} Its entrywise normalisation is not of the form in
  (A3): Theorem~\ref{thm:traj} does not hold, Corollary~\ref{cor:seed} does
  not reduce the seed to $X$, and the unknown seed is all of $A_0$ —
  $Nk+rn$ unknowns against $mr+rn$ equations. Measured$^\dagger$ (jobs
  466915, 467622, 479684): $D\rho$ has $536=Nk+rn$ columns at $n=32$ and
  $1632$ at $n=96$ against SGD's $88$ and $224$, is full column rank at the
  truth in every Adam cell, and is $\sim\!400\times$ worse conditioned at
  $n=96$ ($\kappa=8\times10^{5}$ against $2\times10^{3}$) with a much smaller
  basin. Adam keeps identifiability and loses the $rN$ reduction; and because
  it also breaks Proposition~\ref{prop:simplex}, it hands the attacker the
  $N$ dimensions the softmax head had removed (the prediction above).
\item \textbf{One Jacobian, three readings.} $D\rho$ at the truth is one
  object: its rank is Theorem~\ref{thm:suff}; its spectrum is the local
  resolution ellipsoid $\{h:h^{\!\top}(J^{\!\top}\Sigma^{-1}J)h<\tau\}$ of image
  changes the release cannot distinguish; its whitened form is the
  detectability $d^2=2\,\mathrm{KL}$ of the dataset-sensitivity programme,
  whose valley widths are one-dimensional slices of the ellipsoid
  (Figure~\ref{fig:spectrum}). Rendering its leading directions as image
  edits is the natural next object.
\item \textbf{What stops the attack today is neither line.} Inside the
  capacity region, LM from perturbed-truth starts recovers up to $86\%$ start
  error, with one budget-limited failure at $81\%$ (residual $8\times10^{-5}$
  at the iteration cap, still descending) bracketed by successes at $77\%$
  and $86\%$\textsuperscript{$\dagger$} (job 459111). From the $20$ starts an attacker can build
  from the release alone (random, span, certificate, span-anchor; jobs
  408560--63) it reaches the floor in \emph{none}: one run ends inside the
  $10^{-2}$ image tolerance at residual $9\times10^{-7}$ with its restart

  slot the framework reserves for a learned or population prior — not the
  information in the release.
\end{enumerate}

\subsection{What to say to Gal (replaces the lead of Rev~9 \S16)}
\label{sec:gal}

In this order: the law, because it is a theorem with a derivation and a sharp
measurement, on synthetic data and on digits; the chart, because it is what
the law means; the walk, because it turns the law into a set one can move
along; the initialiser, because it is what stops the attack today; the
recipe, because ``the attacker must be handed it'' did not survive
measurement. Not the $49/49$ grid first: it lies inside the capacity
region and measures existence, not rate.
\begin{enumerate}\itemsep2pt
\item \emph{The chain.} Theorem~\ref{thm:traj} writes the adapter's
  trajectory in closed form; Corollary~\ref{cor:seed} shows the seed enters
  through $rN$ numbers $X$ and the rest of it is released verbatim. Projecting
  $X$ away is Rev~9's certificate, $r-N$ equations per image
  (Theorem~\ref{thm:quot}); solving for it is the exact channel.
\item \emph{The law.} The released $B_T$ lives on a manifold of dimension
  $N((m-1)+r-N)$ — the $-1$ is the softmax head, Proposition~\ref{prop:simplex}
  — and $A_TU_c$ pays exactly for $X$. So $k<m+r-N$ (Theorem~\ref{thm:cap}),
  $m-1$ per image above the certificate, and the difference is precisely the
  coefficient matrix the certificate projects away (Corollary~\ref{cor:two}).
  Measured sharp to one unit of $k$ at two values of $N$ synthetically and
  at three ranks on real digits; both terms of the cap attained by the block
  ranks.
\item \emph{The chart.} The theorems pin the coordinates $w$; the attack
  returns $\psi(\hat w)$. At the same $k=17$, three charts each recover their
  own image to $10^{-14}$ and differ by thirteen orders against the real
  digit; a chart containing the digits recovers them at $k=N$, far below the
  line. The law bounds the dimension of the search, not the image reached,
  and the search space is the attacker's to choose: the counting gives the
  generative prior its job.
\item \emph{The walk.} Past the line the truth is not isolated
  (Theorem~\ref{thm:cap}); one continuation procedure moves the image by
  $10^{-14}$ below the line and by $1.4$--$3.8\%$ at and past it, at the
  residual floor throughout. Where the solver lands past the line, the
  alternative is within $1.7\%$ in the synthetic chart that contains the
  truth and $2$--$7\%$ on real digits, growing with distance; recognisability
  is not assessed. So the line bounds \emph{exact} identifiability and is
  not a leakage bound; how far the manifold reaches is open — one direction
  of forty was walked, and one cell was still rising.
\item \emph{The theorem behind the sharpness.} Below the line $D\rho$ is
  full column rank almost everywhere or nowhere (Theorem~\ref{thm:suff});
  the $T=1$ closed form \eqref{eq:T1} is the witness.
\item \emph{What stops the attack.} Not information: the initialiser
  (none of $20$ attacker-buildable starts reaches the floor; from perturbed
  truth the solver recovers up to $86\%$ away).
\item \emph{The recipe.} None of seven wrong recipes reached the floor —
  one step in four hundred is twenty-three orders away — so the residual
  checks it;
  $\eta$ fits jointly with the data to machine precision; $\eta T$ is not a
  degeneracy of discrete steps; labels are identifiable; and a decaying
  schedule of known family hands over $T$ from six probe steps on the
  attacker's own data. What the count leaves the defender is a free per-step
  schedule with $T>120$.
\item \emph{Open.} The schedule family (model selection, not run); global
  uniqueness; direction dependence of the manifold's extent; Adam, which keeps
  identifiability, loses the $rN$ reduction, destroys the certificate, and is
  predicted to sit one unit higher on the capacity line.
\end{enumerate}
